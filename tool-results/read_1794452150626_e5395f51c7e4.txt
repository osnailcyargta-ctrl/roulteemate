   112→function shuffled<T>(arr: T[]): T[] {
   113→  const a = [...arr];
   114→  for (let i = a.length - 1; i > 0; i--) {
   115→    const j = Math.floor(Math.random() * (i + 1));
   116→    [a[i], a[j]] = [a[j], a[i]];
   117→  }
   118→  return a;
   119→}

   217→  /** Live buyability of one offer (slots & cash can change any moment). */
   218→  private offerView(o: ShopOffer): ShopOfferView {
   219→    const st = this.logic.state;
   220→    let disabled = false;
   221→    let reason: string | null = null;
   222→    if (o.sold) {
   223→      reason = "SOLD";
   224→    } else if (this.activeSkillCount() >= MAX_SKILLS) {
   225→      disabled = true;
   226→      reason = `SLOTS FULL (${MAX_SKILLS})`;
   227→    } else if (this.money < o.price) {
   228→      disabled = true;
   229→      reason = "TOO POOR";
   230→    }
   231→    void st;
   232→    return { ...o, disabled, reason };

   317→  /** Dice intro: 2 rolls for the player (their only skills for the match),
   318→   *  then the enemy silently receives 1/2/3 skills depending on difficulty. */
   319→  private async diceIntro() {
   320→    const diff = this.logic.state.difficulty;
   321→    this.logic.setPhase("dice");
   322→    this.notify();
   323→    await sleep(400);
   324→    this.scene.enemySetMood("grin");
   325→    this.scene.enemySay(["Dice first.", "Your fate is decided here."], 2.8);
   326→    gameAudio.gasp();
   327→
   328→    const granted = shuffled(SKILL_IDS).slice(0, 2);
   329→
   330→    this.scene.diceVisible(true);
   331→    for (const id of granted) {
   332→      gameAudio.diceShake();
   333→      await sleep(360);
   334→      await this.diceTumble(id);
   335→      this.logic.grantSkill("player", id);
   336→      gameAudio.stamp();
   337→      this.onEvent({ type: "skillGet", id });
   338→      this.notify();
   339→      await sleep(820);
   340→    }
   341→    this.scene.diceVisible(false);
   342→
   343→    // the enemy quietly takes its skills
   344→    const n = enemySkillCount(diff);
   345→    for (const id of shuffled(SKILL_IDS).slice(0, n)) this.logic.grantSkill("enemy", id);
   346→
   347→    this.scene.enemySetMood("laugh");
   348→    gameAudio.laugh();
   349→    this.scene.enemySay([pick("gameStart")], 2.4);
   350→    await sleep(1200);
   351→  }

   353→  /** One dice tumble animation in front of the camera. */
   354→  private diceTumble(finalId: SkillId): Promise<void> {
   355→    const { main, ghosts } = this.scene.diceMeshes();
   356→    let lastSwitch = 0;
   357→    let cycle = Math.floor(Math.random() * SKILL_IDS.length);
   358→    const setFace = (id: SkillId) => this.scene.diceSetFace(id, texDieFace(id));
   359→    const dur = 1400;
   360→    const start = performance.now();
   361→    gameAudio.diceRoll(); // physical tumbling knocks while the die spins
   362→    return new Promise<void>((resolve) => {
   363→      const step = () => {
   364→        const t = Math.min(1, (performance.now() - start) / dur);
   365→        main.rotation.z = Math.pow(1 - t, 1.4) * 10;
   366→        const bounce = Math.abs(Math.sin(t * Math.PI * 3.4)) * (1 - t) * 0.3;
   367→        main.position.y = -0.02 + bounce;
   368→        main.scale.setScalar(0.7 + t * 0.3);
   369→        if (t > 0.7) {
   370→          if (main.userData.face !== finalId) {
   371→            setFace(finalId);
   372→            main.userData.face = finalId;
   373→          }
   374→        } else if (performance.now() - lastSwitch > 40 + t * t * 240) {
   375→          lastSwitch = performance.now();
   376→          cycle = (cycle + 1) % SKILL_IDS.length;
   377→          setFace(SKILL_IDS[cycle]);
   378→          main.userData.face = SKILL_IDS[cycle];
   379→        }
   380→        ghosts.forEach((g, i) => {
   381→          g.rotation.z = main.rotation.z - 0.05 * (i + 1);
   382→          g.position.y = main.position.y + 0.012 * (i + 1);
   383→          g.material.opacity = (0.13 / (i + 1)) * (1 - t * 0.4);
   384→          g.material.map = main.material.map;
   385→        });
   386→        if (t < 1) {
   387→          requestAnimationFrame(step);
   388→        } else {
   389→          gameAudio.diceLand();
   390→          this.scene.flash(0.3);
   391→          // squash & settle
   392→          const s0 = performance.now();
   393→          const squash = () => {
   394→            const k = Math.min(1, (performance.now() - s0) / 220);
   395→            main.scale.set(1 + Math.sin(k * Math.PI) * 0.12, 1 - Math.sin(k * Math.PI) * 0.18, 1);
   396→            main.rotation.z = (1 - k) * 0.08;
   397→            if (k < 1) requestAnimationFrame(squash);
   398→            else resolve();
   399→          };
   400→          squash();
   401→        }
   402→      };
   403→      step();
   404→    });
   405→  }

   412→    this.logic.setPhase("reloading");
   413→    this.notify();

   415→
   416→    this.scene.setLampBoost(0.4);
   417→    await sleep(380);
   418→    this.scene.gunAim("enemy", true);
   419→    this.scene.gunOpenCylinder(true);
   420→    gameAudio.reloadOpen();
   421→    await sleep(620);
   422→
   423→    const total = 6;
   424→    for (let i = 0; i < total; i++) {
   425→      gameAudio.reloadLoad(i);
   426→      this.onEvent({ type: "reloadStep", loaded: i + 1, total });
   427→      await sleep(190);
   428→    }
   429→    gameAudio.spinCylinder();
   430→    await sleep(560);
   431→    this.scene.gunOpenCylinder(false);
   432→    gameAudio.reloadClose();
   433→    this.scene.setLampBoost(1);
   434→    this.scene.flash(0.28);
   435→    await sleep(420);
   436→    this.scene.gunAim(null);
   437→
   438→    const live = this.logic.liveRemaining;
   439→    const blank = this.logic.blankRemaining;
   440→    this.onEvent({
   441→      type: "banner",
   442→      title: `ROUND ${this.logic.state.round}`,
   443→      sub: `LIVE ${live} — BLANK ${blank}`,
   444→    });
   445→    gameAudio.hammerCock();
   446→    await sleep(1400);
   447→    this.busy = false;
   448→  }

   458→      this.toast("THE BOARD FADED — memorize your shells!", "bad");

   465→  async playerShoot(target: Side) {
   466→    const st = this.logic.state;
   467→    if (this.busy || st.phase !== "playerTurn" || st.turn !== "player") return;
   468→    if (st.cylinder.length === 0) return;
   469→    this.busy = true;
   470→    this.notify();
   471→
   472→    // LOW-HP TENSION: below 3 HP a self-shot gets 0.7s of pure dread first.
   473→    const tension = target === "player" && st.hp.player < 3;
   474→
   475→    this.scene.gunAim(target === "player" ? "self" : "enemy");
   476→    gameAudio.hammerCock();
   477→    gameAudio.riser(tension ? 1.5 : 0.55);
   478→    await sleep(800);
   479→    if (tension) {
   480→      // 0.7s: the trigger finger trembles... 240 + 240 + 220
   481→      this.scene.addTrauma(0.14);
   482→      await sleep(240);
   483→      this.scene.addTrauma(0.18);
   484→      await sleep(240);
   485→      this.scene.addTrauma(0.22);
   486→      await sleep(220);
   487→    }
   488→
   489→    const res = await this.shootSequence("player", target, tension);
   490→    this.scene.gunAim(null);
   491→    await this.afterShot("player", res);
   492→  }

   494→  async playerUseSkill(id: SkillId) {
   495→    const st = this.logic.state;
   496→    if (this.busy || st.phase !== "playerTurn" || st.turn !== "player") return;
   497→    if (!this.logic.canUseSkill("player", id)) return;
   498→    this.busy = true;
   499→    this.notify();
   500→
   501→    const oc = this.logic.useSkill("player", id);
   502→    if (!oc.ok) {
   503→      this.busy = false;
   504→      this.notify();
   505→      return;
   506→    }
   507→    // a spent TRIAL starts its 2-turn fade away
   508→    const sk = st.skills[id];
   509→    if (sk.trial && sk.charges <= 0) this.registerGhost(id);
   510→    this.onEvent({ type: "skillFlash", id });
   511→    await this.playSkillFx("player", id, oc);
   512→    this.notify();
   513→    this.checkHealthFx();
   514→    await sleep(450);
   515→    this.busy = false;
   516→    this.notify();
   517→  }

   519→  /* ---------------- enemy turn ---------------- */

   521→  private async enemyTurnStart() {
   522→    const st = this.logic.state;
   523→    if (st.hp.enemy <= 0 || st.hp.player <= 0) return;
   524→    st.phase = "enemyTurn";
   525→    const wasPublic = st.countdown > 0;
   526→    this.logic.startTurnTick();
   527→    if (wasPublic && st.countdown <= 0 && st.cylinder.length > 0) {
   528→      this.toast("THE BOARD FADED — memorize your shells!", "bad");
   529→      gameAudio.ghostFade();
   530→    }
   531→    if (this.logic.lastSlip) {
   532→      // the Dealer lost count — and says so out loud
   533→      this.scene.enemySay([pick("aiForget")], 2.2);
   534→    }
   535→    this.busy = true;
   536→    this.notify();
   537→
   538→    this.scene.enemySetMood("idle2");
   539→    this.scene.enemySay([pick("enemyTurn")], 2.2);
   540→    await sleep(rnd(1000, 1700));
   541→
   542→    let skillsUsed = 0;
   543→    let guard = 0;
   544→    let decision: AIDecision | null = null;
   545→    // AI loop: burn skills first, then shoot
   546→    for (;;) {
   547→      if (st.hp.enemy <= 0 || st.hp.player <= 0) return;
   548→      if (this.logic.isEmpty || guard++ > 24) break;
   549→      const d = this.ai.decide();
   550→      if (d.kind === "skill" && d.skill && skillsUsed < 3) {
   551→        const oc = this.logic.useSkill("enemy", d.skill);
   552→        if (!oc.ok) continue;
   553→        skillsUsed++;
   554→        await this.playSkillFx("enemy", d.skill, oc);
   555→        this.notify();
   556→        await sleep(rnd(500, 850));
   557→        continue;
   558→      }
   559→      decision = d;
   560→      break;
   561→    }
   562→
   563→    if (decision && decision.kind === "shoot" && decision.target && !this.logic.isEmpty) {
   564→      this.scene.enemySetMood("aim");
   565→      gameAudio.riser(1.0);
   566→      await sleep(rnd(850, 1400));
   567→      const res = await this.shootSequence("enemy", decision.target);
   568→      this.scene.enemySetMood("idle1");
   569→      if (res === "dead") {
   570→        await this.endGame();
   571→        return;
   572→      }
   573→    }
   574→
   575→    if (this.logic.isEmpty) {
   576→      await this.maybeShopAndReload();
   577→    }
   578→    const next = this.passTurn();
   579→    this.notify();
   580→    if (next === "player") {
   581→      await this.playerTurnStart();
   582→    } else {
   583→      // the player is cuffed
   584→      this.toast("You're cuffed! The DEALER goes again.", "bad");
   585→      await sleep(800);
   586→      await this.enemyTurnStart();
   587→    }
   588→  }

   590→  /* ---------------- shot resolution ---------------- */
   591→
   592→  /** Fire one shot and resolve EVERYTHING that follows it:
   593→   *  the voodoo curse (fresh shell + forced shot) and the global
   594→   *  self-blank rule (one automatic free shot at the opponent). */
   595→  private async shootSequence(shooter: Side, target: Side, tensionSelf = false, forced = false): Promise<"pass" | "dead"> {
   596→    const out = this.logic.shoot(shooter, target, forced);
   597→    const res = await this.resolveShot(out, tensionSelf);
   598→    if (res === "dead") return "dead";
   599→
   600→    const opp: Side = shooter === "player" ? "enemy" : "player";
   601→
   602→    if (out.voodooReroll) {
   603→      // VOODOO CURSE: blank was rerolled — the shooter MUST fire once more,
   604→      // and the gun swings toward the opponent.
   605→      this.notify();
   606→      await sleep(300);
   607→      const out2 = this.logic.shoot(shooter, opp, true);
   608→      const res2 = await this.resolveShot(out2, false);
   609→      if (res2 === "dead") return "dead";
   610→    } else if (out.bonusShot && !this.logic.isEmpty) {
   611→      // GLOBAL RULE: shot yourself with a blank -> ONE automatic shot at the enemy.
   612→      this.notify();
   613→      if (shooter === "player") {
   614→        this.toast("BLANK! Free shot at the DEALER!", "good");
   615→        this.scene.gunAim("enemy");
   616→        gameAudio.hammerCock();
   617→        await sleep(750);
   618→      } else {
   619→        this.scene.enemySetMood("aim");
   620→        this.toast("DEALER blanks himself — free shot at you!", "bad");
   621→        await sleep(750);
   622→      }
   623→      const out2 = this.logic.shoot(shooter, opp);
   624→      const res2 = await this.resolveShot(out2, false);
   625→      if (res2 === "dead") return "dead";
   626→    }
   627→    return "pass";
   628→  }

   630→  /** Animation + effects for ONE trigger pull (both sides). */
   631→  private async resolveShot(out: ShotOutcome, tensionSelf = false): Promise<"pass" | "dead"> {

   633→

   666→      gameAudio.gunshot();

   668→        this.scene.muzzleFlash();
   669→        this.scene.gunKick();
   670→      } else {
   671→        this.scene.addTrauma(0.5);
   672→        this.scene.flash(0.4);
   673→      }
   674→      // bullet-exit delay: at low HP a self-shot lingers 0.2s longer
   675→      await sleep(tensionSelf ? 340 : 140);
   676→
   677→      if (out.reflected) {
   678→        // voodoo reflected the pain back to the shooter
   679→        gameAudio.voodooReflect();
   680→        this.scene.flash(0.55);
   681→        this.onEvent({ type: "dmgPop", side: out.damageTo!, text: "REFLECTED!", tone: "reflect" });

   686→        } else {
   687→          this.scene.bloodBurst("player");
   688→          this.onEvent({ type: "screenSplat" });
   689→        }
   690→      } else if (out.damageTo === "player") {
   691→        gameAudio.hitPlayer();
   692→        this.scene.addTrauma(0.85);
   693→        this.scene.setAberration(2.4);
   694→        setTimeout(() => this.scene.setAberration(0), 380);
   695→        this.onEvent({ type: "screenSplat" });
   696→        this.scene.bloodBurst("player");
   697→        if (!shooterIsPlayer) {
   698→          this.scene.enemySetMood("laugh");
   699→          this.onEvent({ type: "dmgPop", side: "player", text: "-1 HP", tone: "hit" });
   700→        } else {
   701→          this.scene.enemySetMood("laugh");
   702→          this.scene.enemySay([pick("playerHitSelf")], 2.4);
   703→          this.onEvent({ type: "dmgPop", side: "player", text: "-1 HP", tone: "hit" });
   704→        }
   705→      } else if (out.damageTo === "enemy") {
   706→        gameAudio.hitEnemy();
   707→        this.scene.addTrauma(0.4);
   708→        this.scene.bloodBurst("enemy", !!out.death);
   709→        this.scene.enemySetMood("hit");
   710→        this.onEvent({ type: "dmgPop", side: "enemy", text: "-1 HP", tone: "hit" });
   711→        await sleep(650);
   712→        if (out.death !== "enemy") {
   713→          this.scene.enemySetMood("grin");
   714→          this.scene.enemySay([pick("playerHitsEnemy")], 2.2);
   715→        }
   716→      }
   717→      await sleep(520);
   718→    } else {
   719→      // blank
   720→      gameAudio.blankClick();
   721→      if (shooterIsPlayer) {
   722→        this.scene.dustPuffAtCamera();
   723→      } else {
   724→        this.scene.enemySetMood("scared");
   725→      }
   726→      this.onEvent({ type: "dmgPop", side: out.target, text: "CLICK! blank", tone: "blank" });
   727→      const who =
   728→        !shooterIsPlayer && out.target === "enemy"
   729→          ? pick("enemyBlankSelf")
   730→          : shooterIsPlayer && out.target === "player"
   731→            ? pick("playerBlankSelf")
   732→            : pick("playerShootsBlank");
   733→      if (!shooterIsPlayer) {
   734→        this.scene.enemySay([who], 2);
   735→      }
   736→      await sleep(500);
   737→    }
   738→
   739→    if (out.death) return "dead";
   740→    this.notify();
   741→    this.checkHealthFx();
   742→    return "pass";
   743→  }

   928→    }
   929→    this.ghosts = this.ghosts.filter((g) => g.turns > 0);
   930→  }

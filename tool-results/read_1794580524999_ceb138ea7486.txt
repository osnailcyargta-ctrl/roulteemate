   132→function shuffled<T>(arr: T[]): T[] {
   133→  const a = [...arr];
   134→  for (let i = a.length - 1; i > 0; i--) {
   135→    const j = Math.floor(Math.random() * (i + 1));
   136→    [a[i], a[j]] = [a[j], a[i]];
   137→  }
   138→  return a;
   139→}

   242→  /** Live buyability of one offer (slots & cash can change any moment). */
   243→  private offerView(o: ShopOffer): ShopOfferView {
   244→    const st = this.logic.state;
   245→    let disabled = false;
   246→    let reason: string | null = null;
   247→    if (o.sold) {
   248→      reason = "SOLD";
   249→    } else if (this.activeSkillCount() >= MAX_SKILLS) {
   250→      disabled = true;
   251→      reason = `SLOTS FULL (${MAX_SKILLS})`;
   252→    } else if (this.money < o.price) {
   253→      disabled = true;
   254→      reason = "TOO POOR";
   255→    }
   256→    void st;
   257→    return { ...o, disabled, reason };
   258→  }

   289→  /* ---------------- main flow ---------------- */
   290→
   291→  /** The SHOP pool for this table. EPIC cards live here ONLY — and only on
   292→   *  NORMAL and above. LEGEND needs INSANE / ENDLESS (+ the allow-legend
   293→   *  switch). The mutator card pool gates everything further. */
   294→  private poolFor(diff: Difficulty, mode: Mode): SkillId[] {
   295→    const mut = this.mut;
   296→    let pool = ALL_SKILL_IDS;
   297→    if (mut) pool = pool.filter((id) => mut.pool.includes(id));
   298→    const epicOk = diff !== "easy"; // EPIC: shop-only, NORMAL+
   299→    const legendOk = (diff === "insane" || mode === "endless") && (mut ? mut.allowLegend : true);
   300→    return pool.filter((id) => {
   301→      const r = RARITY[id];
   302→      if (r === "epic") return epicOk;
   303→      if (r === "legend") return legendOk;
   304→      return true;
   305→    });
   306→  }
   307→
   308→  /** The DICE pool: strictly BASE cards. The dice NEVER hands out EPIC or
   309→   *  LEGEND — epic cards are bought in the shop on NORMAL+ tables only. */
   310→  private dicePool(): SkillId[] {
   311→    return this.poolFor(
   312→      this.logic.state.difficulty,
   313→      this.logic.state.mode,
   314→    ).filter((id) => RARITY[id] === "base");
   315→  }
   316→
   317→  /** Draw n skills from the pool. The DEALER keeps the stacking record;
   318→   *  the player's dice grants go through diceIntro (which manages slots). */
   319→  private drawSkills(side: Side, pool: SkillId[], n: number) {
   320→    const allowDup = this.mut?.allowDup ?? false;
   321→    const st = this.logic.state;
   322→    const bag = allowDup
   323→      ? pool
   324→      : pool.filter((id) => !st.enemySkills[id].owned || side === "player");
   325→    const src = bag.length ? bag : pool;
   326→    for (let k = 0; k < n; k++) {
   327→      const id = src[Math.floor(Math.random() * src.length)];
   328→      this.logic.grantSkill(side, id);
   329→    }
   330→  }

   332→  async startGame(diff: Difficulty | "endless", mut?: MutConfig) {
   333→    gameAudio.ensure();
   334→    gameAudio.startAmbient();
   335→    this.lowHpTaunted = false;
   336→    this.money = START_MONEY;
   337→    this.ghosts = [];
   338→    this.shopResolve = null;
   339→    this.tStart = performance.now();
   340→    this.tRoundStart = this.tStart;
   341→    this.tEnd = 0;
   342→    // ENDLESS counts as INSANE: same sharp brain, legends allowed in the shop.
   343→    const mode: Mode = diff === "endless" ? "endless" : "standard";
   344→    const baseDiff: Difficulty = diff === "endless" ? "insane" : diff;
   345→    this.mode = mode;
   346→    this.mut = mut ?? null;
   347→    if (mut) {
   348→      const filled: MutConfig = { ...MUT_DEFAULTS, ...mut };
   349→      this.mut = filled;
   350→      this.logic = new RouletteLogic(baseDiff, mode, {
   351→        hpPlayer: filled.hpPlayer,
   352→        hpEnemy: filled.hpEnemy,
   353→      });
   354→    } else {
   355→      this.logic = new RouletteLogic(baseDiff, mode);
   356→    }
   357→    this.ai = new EnemyAI(this.logic, SMART[this.mut?.ai ?? baseDiff]);
   358→    await this.setupScripts();
   359→    this.scene.clearSpentShells();
   360→    this.scene.dealerCuffed(false);
   361→    this.scene.dollyHome();
   362→    this.scene.voodooEndFX("player");
   363→    this.scene.voodooEndFX("enemy");
   364→    this.notify();
   365→    if (mode === "endless") {
   366→      this.toast(t("endlessIntro"), "info");
   367→    } else {
   368→      this.toast(t("startCash", { m: START_MONEY }), "info");
   369→    }
   370→
   371→    await sleep(300);
   372→    await this.diceIntro();
   373→    await this.reloadSequence();
   374→    await this.playerTurnStart();
   375→  }

   377→  /** Compile the mutator's ENGINE C program to WebAssembly and wire it up.
   378→   *  The script runs NATIVELY in the browser's WASM engine — not in JS. */
   379→  private async setupScripts() {
   380→    const src = this.mut?.rules ?? "";
   381→    this.script = null;
   382→    if (!src.trim()) return;
   383→    const compiled = compileScript(src);
   384→    for (const e of compiled.errors.slice(0, 4)) this.toast(`[ENGINE C] ${e}`, "bad");
   385→    if (!compiled.ok || !compiled.bytes) return;
   386→    const host: ScriptHost = {
   387→      runSkill: (id, target) => void this.forceSkillFx(id, target),
   388→      heal: (side, n) => {
   389→        const did = this.logic.healSide(side, n);
   390→        if (did) this.onEvent({ type: "dmgPop", side, text: `+${n} HP`, tone: "reflect" });
   391→      },
   392→      damage: (side, n) => {
   393→        const dead = this.logic.damageSide(side, n);
   394→        if (side === "enemy") {
   395→          this.scene.bloodBurst("enemy");
   396→          if (this.logic.state.mode === "endless") this.onEvent({ type: "dmgPop", side, text: `SCORE +${n}`, tone: "hit" });
   397→          else this.onEvent({ type: "dmgPop", side, text: `-${n} HP`, tone: "hit" });
   398→        } else {
   399→          this.onEvent({ type: "screenSplat" });
   400→          this.scene.addTrauma(0.4);
   401→          if (!dead) this.onEvent({ type: "dmgPop", side, text: `-${n} HP`, tone: "hit" });
   402→        }
   403→      },
   404→      say: (text) => this.toast(text, "info"),
   405→      stateRef: () => ({
   406→        hp: this.logic.state.hp,
   407→        score: this.logic.state.score,
   408→        round: this.logic.state.round,
   409→      }),
   410→    };
   411→    try {
   412→      this.script = await ScriptRuntime.instantiate(compiled.bytes, host, (msg) => this.toast(msg, "bad"));
   413→    } catch (e) {
   414→      this.toast(`[ENGINE C] ${(e as Error).message}`, "bad");
   415→      return;
   416→    }
   417→    this.script.fire("setup");
   418→    if (this.logic.state.hp.player <= 0) {
   419→      // a setup rule killed the player before the first round — mercy: undo it
   420→      this.logic.state.hp.player = 1;
   421→      this.toast("[custom rules] the setup killed you already... you start at 1 HP instead.", "bad");
   422→    }
   423→  }

   434→  /** A script FORCED a skill's effect — apply it with a matching mini-FX. */
   435→  private async forceSkillFx(id: SkillId | "all", target: Side) {
   436→    const real: SkillId =
   437→      id === "all"
   438→        ? (ALL_SKILL_IDS[Math.floor(Math.random() * ALL_SKILL_IDS.length)] as SkillId)
   439→        : id;
   440→    this.logic.forceSkill(real, target);
   441→    this.onEvent({ type: "skillFlash", id: real });
   442→    const label = SKILL_META[real].name;
   443→    switch (real) {
   444→      case "cuffs":
   445→        gameAudio.cuffs();
   446→        if (target === "enemy") {
   447→          this.scene.dealerCuffed(true);
   448→          this.toast(`CUSTOM RULE: ${label} — the DEALER is cuffed!`, "good");
   449→        } else {
   450→          this.toast(`CUSTOM RULE: ${label} — YOUR wrists are locked!`, "bad");
   451→        }
   452→        break;
   453→      case "coffee":
   454→        gameAudio.heal();
   455→        void this.scene.coffeeDrinkFX(target);
   456→        this.toast(`CUSTOM RULE: ${label} — ${target === "player" ? "you" : "the DEALER"} take a sip.`, target === "player" ? "good" : "bad");
   457→        break;
   458→      case "reroll":
   459→        gameAudio.spinCylinder();
   460→        this.scene.addTrauma(0.2);
   461→        this.toast(`CUSTOM RULE: ${label} — the chamber spins!`, "info");
   462→        break;
   463→      case "voodoo":
   464→        gameAudio.voodooCast();
   465→        this.scene.voodooCastFX(target);
   466→        this.toast(`CUSTOM RULE: ${label} curse on ${target === "player" ? "YOU" : "the DEALER"}!`, "voodoo");
   467→        break;
   468→      case "peek":
   469→        gameAudio.scanDing();
   470→        this.toast(`CUSTOM RULE: ${label} — ${target === "player" ? "you" : "the DEALER"} inspect the chamber.`, "info");
   471→        break;
   472→      case "eject":
   473→        gameAudio.ejectShell();
   474→        this.toast(`CUSTOM RULE: ${label} — a shell is tossed!`, "info");
   475→        break;
   476→      case "gamble": {
   477→        // the REAL semi-3D coin: tossed up from the table, lands on the face
   478→        // the script rolled — then the effect applies.
   479→        const win = Math.random() < 0.5;
   480→        this.toast(`CUSTOM RULE: ${label} — a coin flips!`, "info");
   481→        await this.scene.coinTossFX(win);
   482→        this.logic.forceSkill("gamble", target, { win });
   483→        if (target === "player") {
   484→          if (win) {
   485→            gameAudio.heal();
   486→            this.onEvent({ type: "dmgPop", side: "player", text: "+1 HP", tone: "reflect" });
   487→          } else {
   488→            gameAudio.hitPlayer();
   489→            this.scene.addTrauma(0.5);
   490→            this.onEvent({ type: "dmgPop", side: "player", text: "-1 HP", tone: "hit" });
   491→          }
   492→        }
   493→        break;
   494→      }
   495→      default:
   496→        this.toast(`CUSTOM RULE: ${label} fires!`, "info");
   497→    }
   498→    this.notify();
   499→  }

   501→  /** Playtime in the current round and in the whole match (ms). */
   502→  getTimes() {
   503→    const now = this.tEnd || (this.tStart > 0 ? performance.now() : 0);

   505→      roundMs: Math.max(0, now - this.tRoundStart),
   506→      totalMs: Math.max(0, now - this.tStart),
   507→    };

   585→  /** One dice tumble animation in front of the camera. */
   586→  private diceTumble(finalId: SkillId): Promise<void> {
   587→    const { main, ghosts } = this.scene.diceMeshes();
   588→    let lastSwitch = 0;
   589→    let cycle = Math.floor(Math.random() * SKILL_IDS.length);
   590→    const setFace = (id: SkillId) => this.scene.diceSetFace(id, texDieFace(id));
   591→    const dur = 1400;
   592→    const start = performance.now();
   593→    gameAudio.diceRoll(); // physical tumbling knocks while the die spins
   594→    return new Promise<void>((resolve) => {
   595→      const step = () => {
   596→        const t = Math.min(1, (performance.now() - start) / dur);
   597→        main.rotation.z = Math.pow(1 - t, 1.4) * 10;
   598→        const bounce = Math.abs(Math.sin(t * Math.PI * 3.4)) * (1 - t) * 0.3;
   599→        main.position.y = -0.02 + bounce;
   600→        main.scale.setScalar(0.7 + t * 0.3);
   601→        if (t > 0.7) {
   602→          if (main.userData.face !== finalId) {
   603→            setFace(finalId);
   604→            main.userData.face = finalId;
   605→          }
   606→        } else if (performance.now() - lastSwitch > 40 + t * t * 240) {
   607→          lastSwitch = performance.now();
   608→          cycle = (cycle + 1) % SKILL_IDS.length;
   609→          setFace(SKILL_IDS[cycle]);
   610→          main.userData.face = SKILL_IDS[cycle];
   611→        }
   612→        ghosts.forEach((g, i) => {
   613→          g.rotation.z = main.rotation.z - 0.05 * (i + 1);
   614→          g.position.y = main.position.y + 0.012 * (i + 1);
   615→          g.material.opacity = (0.13 / (i + 1)) * (1 - t * 0.4);
   616→          g.material.map = main.material.map;
   617→        });
   618→        if (t < 1) {
   619→          requestAnimationFrame(step);
   620→        } else {
   621→          gameAudio.diceLand();
   622→          this.scene.flash(0.3);
   623→          // squash & settle
   624→          const s0 = performance.now();
   625→          const squash = () => {
   626→            const k = Math.min(1, (performance.now() - s0) / 220);
   627→            main.scale.set(1 + Math.sin(k * Math.PI) * 0.12, 1 - Math.sin(k * Math.PI) * 0.18, 1);
   628→            main.rotation.z = (1 - k) * 0.08;
   629→            if (k < 1) requestAnimationFrame(squash);
   630→            else resolve();
   631→          };
   632→          squash();
   633→        }
   634→      };
   635→      step();
   636→    });
   637→  }

   737→    const st = this.logic.state;
   738→    if (this.busy || st.phase !== "playerTurn" || st.turn !== "player") return;

   748→      // from the table, and only when it lands does fate apply.

   770→    } else {
   771→      oc = this.logic.useSkill("player", uid);
   772→      if (!oc.ok) {
   773→        this.busy = false;
   774→        this.notify();
   775→        return;
   776→      }

   782→    if (spent?.trial && spent.charges <= 0) this.registerGhost(spent);

   789→      await sleep(600);
   790→      await this.endGame();
   791→      return;
   792→    }
   793→
   794→    // EJECT on the LAST shell empties the cylinder — nothing left to fire,
   795→    // so the round ends RIGHT HERE: shop break, reload, pass the turn.
   796→    if (this.logic.isEmpty) {
   797→      this.toast(t("chamberEmpty"), "info");
   798→      await sleep(900);
   799→      await this.maybeShopAndReload();

   802→      if (next === "player") {
   803→        this.toast(t("dealerCuffedAgain"), "good");
   804→        await sleep(700);
   805→        await this.playerTurnStart();
   806→      } else {
   807→        await this.enemyTurnStart();
   808→      }
   809→      return;
   810→    }
   811→
   812→    await sleep(450);
   813→    this.busy = false;
   814→    this.notify();
   815→  }

   895→  /** Fire one shot and resolve EVERYTHING that follows it:
   896→   *  the voodoo curse (fresh shell + forced shot) and the global

   898→  private async shootSequence(shooter: Side, target: Side, tensionSelf = false, forced = false): Promise<"pass" | "dead"> {
   899→    const out = this.logic.shoot(shooter, target, forced);
   900→    const res = await this.resolveShot(out, tensionSelf);
   901→    if (res === "dead") return "dead";
   902→
   903→    const opp: Side = shooter === "player" ? "enemy" : "player";
   904→
   905→    if (out.voodooReroll) {
   906→      // VOODOO CURSE: blank was rerolled — the shooter MUST fire once more,
   907→      // and the gun swings toward the opponent.

   909→      await sleep(300);
   910→      const out2 = this.logic.shoot(shooter, opp, true);
   911→      const res2 = await this.resolveShot(out2, false);
   912→      if (res2 === "dead") return "dead";
   913→    } else if (out.bonusShot && !this.logic.isEmpty) {
   914→      // GLOBAL RULE: shot yourself with a blank -> ONE automatic shot at the enemy.
   915→      this.notify();

   923→        this.toast("DEALER blanks himself — free shot at you!", "bad");
   924→        await sleep(750);
   925→      }
   926→      const out2 = this.logic.shoot(shooter, opp);
   927→      const res2 = await this.resolveShot(out2, false);
   928→      if (res2 === "dead") return "dead";
   929→    }
   930→    return "pass";

  1349→  /** Every passed turn dissolves the ghosts a little more. */

  1351→    if (!this.ghosts.length) return;
  1352→    for (const g of this.ghosts) g.turns -= 1;
  1353→    const done = this.ghosts.filter((g) => g.turns <= 0);
  1354→    for (const g of done) {
  1355→      const st = this.logic.state;
  1356→      const idx = st.skills.findIndex((s) => s.uid === g.uid);
  1357→      if (idx >= 0) st.skills.splice(idx, 1);
  1358→      this.toast(`${SKILL_META[g.id].name} faded away.`, "info");
  1359→    }
  1360→    this.ghosts = this.ghosts.filter((g) => g.turns > 0);
  1361→  }

  1395→      .slice(0, 3)

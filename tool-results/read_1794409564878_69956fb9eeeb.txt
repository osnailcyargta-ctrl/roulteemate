    71→export function mulberry32(seed: number): RNG {
    72→  let a = seed >>> 0;
    73→  return () => {
    74→    a |= 0;
    75→    a = (a + 0x6d2b79f5) | 0;
    76→    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    77→    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    78→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
    79→  };
    80→}

    89→const cache = new Map<string, HTMLCanvasElement>();

    92→export function makeCanvas(
    93→  key: string,
    94→  w: number,
    95→  h: number,
    96→  draw: (ctx: CanvasRenderingContext2D, w: number, h: number, rng: RNG) => void,
    97→  seed = 1,
    98→): HTMLCanvasElement {
    99→  const hit = cache.get(key);
   100→  if (hit) return hit;
   101→  const cv = document.createElement("canvas");
   102→  cv.width = w;
   103→  cv.height = h;
   104→  const ctx = cv.getContext("2d")!;
   105→  draw(ctx, w, h, mulberry32(seed));
   106→  cache.set(key, cv);
   107→  return cv;
   108→}
   109→
   110→export function dataURL(key: string, cv: HTMLCanvasElement): string {
   111→  const hit = dataCache.get(key);
   112→  if (hit) return hit;
   113→  const url = cv.toDataURL("image/png");
   114→  dataCache.set(key, url);
   115→  return url;
   116→}

   118→function wobblePath(
   119→  ctx: CanvasRenderingContext2D,
   120→  pts: [number, number][],
   121→  rng: RNG,
   122→  jitter = 3,
   123→  close = false,
   124→) {
   125→  // subdivide
   126→  const dense: [number, number][] = [];
   127→  for (let i = 0; i < pts.length - 1; i++) {
   128→    const [x1, y1] = pts[i];
   129→    const [x2, y2] = pts[i + 1];
   130→    const d = Math.hypot(x2 - x1, y2 - y1);
   131→    const n = Math.max(2, Math.round(d / 16));
   132→    for (let k = 0; k < n; k++) {
   133→      const t = k / n;
   134→      dense.push([x1 + (x2 - x1) * t, y1 + (y2 - y1) * t]);
   135→    }
   136→  }
   137→  dense.push(pts[pts.length - 1]);
   138→  // jitter
   139→  const j = dense.map(([x, y]) => [
   140→    x + (rng() - 0.5) * jitter * 2,
   141→    y + (rng() - 0.5) * jitter * 2,
   142→  ]) as [number, number][];
   143→  ctx.beginPath();
   144→  ctx.moveTo(j[0][0], j[0][1]);
   145→  for (let i = 1; i < j.length - 1; i++) {
   146→    const mx = (j[i][0] + j[i + 1][0]) / 2;
   147→    const my = (j[i][1] + j[i + 1][1]) / 2;
   148→    ctx.quadraticCurveTo(j[i][0], j[i][1], mx, my);
   149→  }
   150→  const last = j[j.length - 1];
   151→  ctx.lineTo(last[0], last[1]);
   152→  if (close) ctx.closePath();
   153→}

   155→interface StrokeOpts {
   156→  width?: number;
   157→  jitter?: number;
   158→  close?: boolean;
   159→  color?: string;
   160→  passes?: number;
   161→}
   162→
   163→function sketchStroke(
   164→  ctx: CanvasRenderingContext2D,
   165→  pts: [number, number][],
   166→  rng: RNG,
   167→  opts: StrokeOpts = {},
   168→) {
   169→  const { width = 6, jitter = 3, close = false, color = INK, passes = 2 } = opts;
   170→  ctx.strokeStyle = color;
   171→  ctx.lineCap = "round";
   172→  ctx.lineJoin = "round";
   173→  for (let p = 0; p < passes; p++) {
   174→    ctx.globalAlpha = p === 0 ? 1 : 0.35;
   175→    ctx.lineWidth = p === 0 ? width : width * 0.7;
   176→    wobblePath(ctx, pts, rng, jitter, close);
   177→    ctx.stroke();
   178→  }
   179→  ctx.globalAlpha = 1;
   180→}

   182→function sketchLine(
   183→  ctx: CanvasRenderingContext2D,
   184→  x1: number,
   185→  y1: number,
   186→  x2: number,
   187→  y2: number,
   188→  rng: RNG,
   189→  opts?: { width?: number; jitter?: number; color?: string },
   190→) {
   191→  sketchStroke(ctx, [[x1, y1], [x2, y2]], rng, opts);
   192→}

   194→function sketchCircle(
   195→  ctx: CanvasRenderingContext2D,
   196→  cx: number,
   197→  cy: number,
   198→  r: number,
   199→  rng: RNG,
   200→  opts: { width?: number; jitter?: number; color?: string; fill?: string; passes?: number } = {},
   201→) {
   202→  if (opts.fill && r > 0.5) {
   203→    ctx.fillStyle = opts.fill;
   204→    ctx.beginPath();
   205→    ctx.ellipse(cx, cy, r, r * (0.97 + rng() * 0.05), rng() * 0.2, 0, Math.PI * 2);
   206→    ctx.fill();
   207→  }
   208→  if (r < 0.5) return;
   209→  const pts: [number, number][] = [];
   210→  const n = Math.max(10, Math.round(r / 3));
   211→  for (let i = 0; i <= n; i++) {
   212→    const a = (i / n) * Math.PI * 2;
   213→    pts.push([cx + Math.cos(a) * r, cy + Math.sin(a) * r]);
   214→  }
   215→  sketchStroke(ctx, pts, rng, { ...opts, close: true });
   216→}
   217→
   218→function sketchEllipse(
   219→  ctx: CanvasRenderingContext2D,
   220→  cx: number,
   221→  cy: number,
   222→  rx: number,
   223→  ry: number,
   224→  rng: RNG,
   225→  opts: { width?: number; jitter?: number; fill?: string; rot?: number } = {},
   226→) {
   227→  if (opts.fill) {
   228→    ctx.fillStyle = opts.fill;
   229→    ctx.beginPath();
   230→    ctx.ellipse(cx, cy, rx, ry, opts.rot ?? 0, 0, Math.PI * 2);
   231→    ctx.fill();
   232→  }
   233→  const pts: [number, number][] = [];
   234→  const n = Math.max(10, Math.round((rx + ry) / 4));
   235→  for (let i = 0; i <= n; i++) {
   236→    const a = (i / n) * Math.PI * 2;
   237→    const x = cx + Math.cos(a) * rx;
   238→    const y = cy + Math.sin(a) * ry;
   239→    const c = Math.cos(opts.rot ?? 0);
   240→    const s = Math.sin(opts.rot ?? 0);
   241→    pts.push([cx + (x - cx) * c - (y - cy) * s, cy + (x - cx) * s + (y - cy) * c]);
   242→  }
   243→  sketchStroke(ctx, pts, rng, { ...opts, close: true });
   244→}

   246→function hatch(
   247→  ctx: CanvasRenderingContext2D,
   248→  x: number,
   249→  y: number,
   250→  w: number,
   251→  h: number,
   252→  rng: RNG,
   253→  color = "rgba(34,26,18,0.5)",
   254→  gap = 9,
   255→) {
   256→  ctx.save();
   257→  ctx.beginPath();
   258→  ctx.rect(x, y, w, h);
   259→  ctx.clip();
   260→  ctx.strokeStyle = color;
   261→  ctx.lineWidth = 3;
   262→  for (let i = -h; i < w; i += gap) {
   263→    ctx.beginPath();
   264→    ctx.moveTo(x + i + (rng() - 0.5) * 3, y + h + (rng() - 0.5) * 3);
   265→    ctx.lineTo(x + i + h + (rng() - 0.5) * 3, y + (rng() - 0.5) * 3);
   266→    ctx.stroke();
   267→  }
   268→  ctx.restore();
   269→}
   270→
   271→function handText(
   272→  ctx: CanvasRenderingContext2D,
   273→  text: string,
   274→  x: number,
   275→  y: number,
   276→  size: number,
   277→  color = INK,
   278→  rot = 0,
   279→  font = '"Permanent Marker", cursive',
   280→) {
   281→  ctx.save();
   282→  ctx.translate(x, y);
   283→  ctx.rotate(rot);
   284→  ctx.font = `${size}px ${font}`;
   285→  ctx.textAlign = "center";
   286→  ctx.textBaseline = "middle";
   287→  ctx.fillStyle = color;
   288→  ctx.fillText(text, 0, 0);
   289→  ctx.restore();
   290→}

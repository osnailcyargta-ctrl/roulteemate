    71→    71→export function mulberry32(seed: number): RNG {
    72→    72→  let a = seed >>> 0;
    73→    73→  return () => {
    74→    74→    a |= 0;
    75→    75→    a = (a + 0x6d2b79f5) | 0;
    76→    76→    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    77→    77→    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    78→    78→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
    79→    79→  };
    80→    80→}

    92→    92→export function makeCanvas(
    93→    93→  key: string,
    94→    94→  w: number,
    95→    95→  h: number,
    96→    96→  draw: (ctx: CanvasRenderingContext2D, w: number, h: number, rng: RNG) => void,
    97→    97→  seed = 1,
    98→    98→): HTMLCanvasElement {
    99→    99→  const hit = cache.get(key);
   100→   100→  if (hit) return hit;
   101→   101→  const cv = document.createElement("canvas");
   102→   102→  cv.width = w;
   103→   103→  cv.height = h;
   104→   104→  const ctx = cv.getContext("2d")!;
   105→   105→  draw(ctx, w, h, mulberry32(seed));
   106→   106→  cache.set(key, cv);
   107→   107→  return cv;
   108→   108→}

   118→   118→function wobblePath(
   119→   119→  ctx: CanvasRenderingContext2D,
   120→   120→  pts: [number, number][],
   121→   121→  rng: RNG,
   122→   122→  jitter = 3,
   123→   123→  close = false,
   124→   124→) {
   125→   125→  // subdivide
   126→   126→  const dense: [number, number][] = [];
   127→   127→  for (let i = 0; i < pts.length - 1; i++) {
   128→   128→    const [x1, y1] = pts[i];
   129→   129→    const [x2, y2] = pts[i + 1];
   130→   130→    const d = Math.hypot(x2 - x1, y2 - y1);
   131→   131→    const n = Math.max(2, Math.round(d / 16));
   132→   132→    for (let k = 0; k < n; k++) {
   133→   133→      const t = k / n;
   134→   134→      dense.push([x1 + (x2 - x1) * t, y1 + (y2 - y1) * t]);
   135→   135→    }
   136→   136→  }
   137→   137→  dense.push(pts[pts.length - 1]);
   138→   138→  // jitter
   139→   139→  const j = dense.map(([x, y]) => [
   140→   140→    x + (rng() - 0.5) * jitter * 2,
   141→   141→    y + (rng() - 0.5) * jitter * 2,
   142→   142→  ]) as [number, number][];
   143→   143→  ctx.beginPath();
   144→   144→  ctx.moveTo(j[0][0], j[0][1]);
   145→   145→  for (let i = 1; i < j.length - 1; i++) {
   146→   146→    const mx = (j[i][0] + j[i + 1][0]) / 2;
   147→   147→    const my = (j[i][1] + j[i + 1][1]) / 2;
   148→   148→    ctx.quadraticCurveTo(j[i][0], j[i][1], mx, my);
   149→   149→  }
   150→   150→  const last = j[j.length - 1];
   151→   151→  ctx.lineTo(last[0], last[1]);
   152→   152→  if (close) ctx.closePath();
   153→   153→}

   163→   163→function sketchStroke(
   164→   164→  ctx: CanvasRenderingContext2D,
   165→   165→  pts: [number, number][],
   166→   166→  rng: RNG,
   167→   167→  opts: StrokeOpts = {},
   168→   168→) {
   169→   169→  const { width = 6, jitter = 3, close = false, color = INK, passes = 2 } = opts;
   170→   170→  ctx.strokeStyle = color;
   171→   171→  ctx.lineCap = "round";
   172→   172→  ctx.lineJoin = "round";
   173→   173→  for (let p = 0; p < passes; p++) {
   174→   174→    ctx.globalAlpha = p === 0 ? 1 : 0.35;
   175→   175→    ctx.lineWidth = p === 0 ? width : width * 0.7;
   176→   176→    wobblePath(ctx, pts, rng, jitter, close);
   177→   177→    ctx.stroke();
   178→   178→  }
   179→   179→  ctx.globalAlpha = 1;
   180→   180→}

   194→   194→function sketchCircle(
   195→   195→  ctx: CanvasRenderingContext2D,
   196→   196→  cx: number,
   197→   197→  cy: number,
   198→   198→  r: number,
   199→   199→  rng: RNG,
   200→   200→  opts: { width?: number; jitter?: number; color?: string; fill?: string; passes?: number } = {},
   201→   201→) {
   202→   202→  if (opts.fill && r > 0.5) {
   203→   203→    ctx.fillStyle = opts.fill;
   204→   204→    ctx.beginPath();
   205→   205→    ctx.ellipse(cx, cy, r, r * (0.97 + rng() * 0.05), rng() * 0.2, 0, Math.PI * 2);
   206→   206→    ctx.fill();
   207→   207→  }
   208→   208→  if (r < 0.5) return;
   209→   209→  const pts: [number, number][] = [];
   210→   210→  const n = Math.max(10, Math.round(r / 3));
   211→   211→  for (let i = 0; i <= n; i++) {
   212→   212→    const a = (i / n) * Math.PI * 2;
   213→   213→    pts.push([cx + Math.cos(a) * r, cy + Math.sin(a) * r]);
   214→   214→  }
   215→   215→  sketchStroke(ctx, pts, rng, { ...opts, close: true });
   216→   216→}

   218→   218→function sketchEllipse(
   219→   219→  ctx: CanvasRenderingContext2D,
   220→   220→  cx: number,
   221→   221→  cy: number,
   222→   222→  rx: number,
   223→   223→  ry: number,
   224→   224→  rng: RNG,
   225→   225→  opts: { width?: number; jitter?: number; fill?: string; rot?: number } = {},
   226→   226→) {
   227→   227→  if (opts.fill) {
   228→   228→    ctx.fillStyle = opts.fill;
   229→   229→    ctx.beginPath();
   230→   230→    ctx.ellipse(cx, cy, rx, ry, opts.rot ?? 0, 0, Math.PI * 2);
   231→   231→    ctx.fill();
   232→   232→  }
   233→   233→  const pts: [number, number][] = [];
   234→   234→  const n = Math.max(10, Math.round((rx + ry) / 4));
   235→   235→  for (let i = 0; i <= n; i++) {
   236→   236→    const a = (i / n) * Math.PI * 2;
   237→   237→    const x = cx + Math.cos(a) * rx;
   238→   238→    const y = cy + Math.sin(a) * ry;
   239→   239→    const c = Math.cos(opts.rot ?? 0);
   240→   240→    const s = Math.sin(opts.rot ?? 0);
   241→   241→    pts.push([cx + (x - cx) * c - (y - cy) * s, cy + (x - cx) * s + (y - cy) * c]);
   242→   242→  }
   243→   243→  sketchStroke(ctx, pts, rng, { ...opts, close: true });
   244→   244→}

   246→   246→function hatch(
   247→   247→  ctx: CanvasRenderingContext2D,
   248→   248→  x: number,
   249→   249→  y: number,
   250→   250→  w: number,
   251→   251→  h: number,
   252→   252→  rng: RNG,
   253→   253→  color = "rgba(34,26,18,0.5)",
   254→   254→  gap = 9,
   255→   255→) {
   256→   256→  ctx.save();
   257→   257→  ctx.beginPath();
   258→   258→  ctx.rect(x, y, w, h);
   259→   259→  ctx.clip();
   260→   260→  ctx.strokeStyle = color;
   261→   261→  ctx.lineWidth = 3;
   262→   262→  for (let i = -h; i < w; i += gap) {
   263→   263→    ctx.beginPath();
   264→   264→    ctx.moveTo(x + i + (rng() - 0.5) * 3, y + h + (rng() - 0.5) * 3);
   265→   265→    ctx.lineTo(x + i + h + (rng() - 0.5) * 3, y + (rng() - 0.5) * 3);
   266→   266→    ctx.stroke();
   267→   267→  }
   268→   268→  ctx.restore();
   269→   269→}

   271→   271→function handText(
   272→   272→  ctx: CanvasRenderingContext2D,
   273→   273→  text: string,
   274→   274→  x: number,
   275→   275→  y: number,
   276→   276→  size: number,
   277→   277→  color = INK,
   278→   278→  rot = 0,
   279→   279→  font = '"Permanent Marker", cursive',
   280→   280→) {
   281→   281→  ctx.save();
   282→   282→  ctx.translate(x, y);
   283→   283→  ctx.rotate(rot);
   284→   284→  ctx.font = `${size}px ${font}`;
   285→   285→  ctx.textAlign = "center";
   286→   286→  ctx.textBaseline = "middle";
   287→   287→  ctx.fillStyle = color;
   288→   288→  ctx.fillText(text, 0, 0);
   289→   289→  ctx.restore();
   290→   290→}

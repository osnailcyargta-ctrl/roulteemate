   132→   132→function shuffled<T>(arr: T[]): T[] {
   133→   133→  const a = [...arr];
   134→   134→  for (let i = a.length - 1; i > 0; i--) {
   135→   135→    const j = Math.floor(Math.random() * (i + 1));
   136→   136→    [a[i], a[j]] = [a[j], a[i]];
   137→   137→  }
   138→   138→  return a;
   139→   139→}

   297→   297→  private poolFor(diff: Difficulty, mode: Mode): SkillId[] {
   298→   298→    const mut = this.mut;
   299→   299→    let pool = ALL_SKILL_IDS;
   300→   300→    if (mut) pool = pool.filter((id) => mut.pool.includes(id));
   301→   301→    const epicOk = diff !== "easy";
   302→   302→    const legendOk = (diff === "insane" || mode === "endless") && (mut ? mut.allowLegend : true);
   303→   303→    return pool.filter((id) => {
   304→   304→      const r = RARITY[id];
   305→   305→      if (r === "epic") return epicOk;
   306→   306→      if (r === "legend") return legendOk;
   307→   307→      return true;
   308→   308→    });
   309→   309→  }

   326→   326→      this.logic.grantSkill(side, id);
   327→   327→    }

   375→   375→  /** Compile the mutator's custom-effect scripts and wire their hooks. */
   376→   376→  private setupScripts() {
   377→   377→    const rules = this.mut?.rules ?? [];
   378→   378→    this.script = null;
   379→   379→    if (!rules.length) return;
   380→   380→    let program: Program | null = null;
   381→   381→    const errors: string[] = [];
   382→   382→    for (const src of rules) {
   383→   383→      const res = parseScript(src);
   384→   384→      errors.push(...res.errors);
   385→   385→      if (res.errors.length) continue;
   386→   386→      if (!program) program = { setup: [], handlers: [] };
   387→   387→      program.setup.push(...res.program.setup);
   388→   388→      program.handlers.push(...res.program.handlers);
   389→   389→    }
   390→   390→    for (const e of errors.slice(0, 4)) this.toast(`[rule] ${e}`, "bad");
   391→   391→    if (!program) return;
   392→   392→    const rt = new ScriptRuntime(program, {
   393→   393→      runSkill: (id, target) => this.forceSkillFx(id, target),
   394→   394→      heal: (side, n) => {
   395→   395→        const did = this.logic.healSide(side, n);
   396→   396→        if (did) this.onEvent({ type: "dmgPop", side, text: `+${n} HP`, tone: "reflect" });
   397→   397→      },
   398→   398→      damage: (side, n) => {
   399→   399→        const dead = this.logic.damageSide(side, n);
   400→   400→        if (side === "enemy") {
   401→   401→          this.scene.bloodBurst("enemy");
   402→   402→          if (this.logic.state.mode === "endless") this.onEvent({ type: "dmgPop", side, text: `SCORE +${n}`, tone: "hit" });
   403→   403→          else this.onEvent({ type: "dmgPop", side, text: `-${n} HP`, tone: "hit" });
   404→   404→        } else {
   405→   405→          this.onEvent({ type: "screenSplat" });
   406→   406→          this.scene.addTrauma(0.4);
   407→   407→          if (!dead) this.onEvent({ type: "dmgPop", side, text: `-${n} HP`, tone: "hit" });
   408→   408→        }
   409→   409→      },
   410→   410→      say: (text) => this.toast(text, "info"),
   411→   411→    });
   412→   412→    rt.stateRef = () => ({ hp: this.logic.state.hp, score: this.logic.state.score, round: this.logic.state.round });
   413→   413→    this.script = rt;
   414→   414→    rt.setup();
   415→   415→    if (this.logic.state.hp.player <= 0) {
   416→   416→      // a setup rule killed the player before the first round — mercy: undo it
   417→   417→      this.logic.state.hp.player = 1;
   418→   418→      this.toast("[custom rules] the setup killed you already... you start at 1 HP instead.", "bad");
   419→   419→    }
   420→   420→  }

   431→   431→  /** A script FORCED a skill's effect — apply it with a matching mini-FX. */
   432→   432→  private async forceSkillFx(id: SkillId | "all", target: Side) {
   433→   433→    const real: SkillId =
   434→   434→      id === "all"
   435→   435→        ? (ALL_SKILL_IDS[Math.floor(Math.random() * ALL_SKILL_IDS.length)] as SkillId)
   436→   436→        : id;
   437→   437→    this.logic.forceSkill(real, target);
   438→   438→    this.onEvent({ type: "skillFlash", id: real });

   562→   562→  /** One dice tumble animation in front of the camera. */
   563→   563→  private diceTumble(finalId: SkillId): Promise<void> {
   564→   564→    const { main, ghosts } = this.scene.diceMeshes();
   565→   565→    let lastSwitch = 0;
   566→   566→    let cycle = Math.floor(Math.random() * SKILL_IDS.length);
   567→   567→    const setFace = (id: SkillId) => this.scene.diceSetFace(id, texDieFace(id));
   568→   568→    const dur = 1400;
   569→   569→    const start = performance.now();
   570→   570→    gameAudio.diceRoll(); // physical tumbling knocks while the die spins
   571→   571→    return new Promise<void>((resolve) => {
   572→   572→      const step = () => {
   573→   573→        const t = Math.min(1, (performance.now() - start) / dur);
   574→   574→        main.rotation.z = Math.pow(1 - t, 1.4) * 10;
   575→   575→        const bounce = Math.abs(Math.sin(t * Math.PI * 3.4)) * (1 - t) * 0.3;
   576→   576→        main.position.y = -0.02 + bounce;
   577→   577→        main.scale.setScalar(0.7 + t * 0.3);
   578→   578→        if (t > 0.7) {
   579→   579→          if (main.userData.face !== finalId) {
   580→   580→            setFace(finalId);
   581→   581→            main.userData.face = finalId;
   582→   582→          }
   583→   583→        } else if (performance.now() - lastSwitch > 40 + t * t * 240) {
   584→   584→          lastSwitch = performance.now();
   585→   585→          cycle = (cycle + 1) % SKILL_IDS.length;
   586→   586→          setFace(SKILL_IDS[cycle]);
   587→   587→          main.userData.face = SKILL_IDS[cycle];
   588→   588→        }
   589→   589→        ghosts.forEach((g, i) => {
   590→   590→          g.rotation.z = main.rotation.z - 0.05 * (i + 1);
   591→   591→          g.position.y = main.position.y + 0.012 * (i + 1);
   592→   592→          g.material.opacity = (0.13 / (i + 1)) * (1 - t * 0.4);
   593→   593→          g.material.map = main.material.map;
   594→   594→        });
   595→   595→        if (t < 1) {
   596→   596→          requestAnimationFrame(step);
   597→   597→        } else {
   598→   598→          gameAudio.diceLand();
   599→   599→          this.scene.flash(0.3);
   600→   600→          // squash & settle
   601→   601→          const s0 = performance.now();

   603→   603→            const k = Math.min(1, (performance.now() - s0) / 220);
   604→   604→            main.scale.set(1 + Math.sin(k * Math.PI) * 0.12, 1 - Math.sin(k * Math.PI) * 0.18, 1);
   605→   605→            main.rotation.z = (1 - k) * 0.08;
   606→   606→            if (k < 1) requestAnimationFrame(squash);
   607→   607→            else resolve();
   608→   608→          };
   609→   609→          squash();
   610→   610→        }
   611→   611→      };
   612→   612→      step();
   613→   613→    });
   614→   614→  }

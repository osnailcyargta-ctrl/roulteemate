   112→function shuffled<T>(arr: T[]): T[] {
   113→  const a = [...arr];
   114→  for (let i = a.length - 1; i > 0; i--) {
   115→    const j = Math.floor(Math.random() * (i + 1));
   116→    [a[i], a[j]] = [a[j], a[i]];
   117→  }
   118→  return a;
   119→}

   217→  /** Live buyability of one offer (slots & cash can change any moment). */
   218→  private offerView(o: ShopOffer): ShopOfferView {
   219→    const st = this.logic.state;
   220→    let disabled = false;
   221→    let reason: string | null = null;
   222→    if (o.sold) {
   223→      reason = "SOLD";
   224→    } else if (this.activeSkillCount() >= MAX_SKILLS) {
   225→      disabled = true;
   226→      reason = `SLOTS FULL (${MAX_SKILLS})`;
   227→    } else if (this.money < o.price) {
   228→      disabled = true;
   229→      reason = "TOO POOR";
   230→    }
   231→    void st;
   232→    return { ...o, disabled, reason };
   233→  }

   319→  /** Dice intro: 2 rolls for the player (their only skills for the match),
   320→   *  then the enemy silently receives 1/2/3 skills depending on difficulty. */
   321→  private async diceIntro() {
   322→    const diff = this.logic.state.difficulty;
   323→    this.logic.setPhase("dice");
   324→    this.notify();
   325→    await sleep(400);
   326→    this.scene.enemySetMood("grin");
   327→    this.scene.enemySay(["Dice first.", "Your fate is decided here."], 2.8);
   328→    gameAudio.gasp();
   329→
   330→    const granted = shuffled(SKILL_IDS).slice(0, 2);
   331→
   332→    this.scene.diceVisible(true);
   333→    for (const id of granted) {
   334→      gameAudio.diceShake();
   335→      await sleep(360);
   336→      await this.diceTumble(id);
   337→      this.logic.grantSkill("player", id);
   338→      gameAudio.stamp();
   339→      this.onEvent({ type: "skillGet", id });
   340→      this.notify();
   341→      await sleep(820);
   342→    }
   343→    this.scene.diceVisible(false);
   344→
   345→    // the enemy quietly takes its skills
   346→    const n = enemySkillCount(diff);
   347→    for (const id of shuffled(SKILL_IDS).slice(0, n)) this.logic.grantSkill("enemy", id);
   348→
   349→    this.scene.enemySetMood("laugh");
   350→    gameAudio.laugh();
   351→    this.scene.enemySay([pick("gameStart")], 2.4);
   352→    await sleep(1200);
   353→  }

   355→  /** One dice tumble animation in front of the camera. */
   356→  private diceTumble(finalId: SkillId): Promise<void> {
   357→    const { main, ghosts } = this.scene.diceMeshes();
   358→    let lastSwitch = 0;
   359→    let cycle = Math.floor(Math.random() * SKILL_IDS.length);
   360→    const setFace = (id: SkillId) => this.scene.diceSetFace(id, texDieFace(id));
   361→    const dur = 1400;
   362→    const start = performance.now();
   363→    gameAudio.diceRoll(); // physical tumbling knocks while the die spins
   364→    return new Promise<void>((resolve) => {
   365→      const step = () => {
   366→        const t = Math.min(1, (performance.now() - start) / dur);
   367→        main.rotation.z = Math.pow(1 - t, 1.4) * 10;
   368→        const bounce = Math.abs(Math.sin(t * Math.PI * 3.4)) * (1 - t) * 0.3;
   369→        main.position.y = -0.02 + bounce;
   370→        main.scale.setScalar(0.7 + t * 0.3);
   371→        if (t > 0.7) {
   372→          if (main.userData.face !== finalId) {
   373→            setFace(finalId);
   374→            main.userData.face = finalId;
   375→          }
   376→        } else if (performance.now() - lastSwitch > 40 + t * t * 240) {
   377→          lastSwitch = performance.now();
   378→          cycle = (cycle + 1) % SKILL_IDS.length;
   379→          setFace(SKILL_IDS[cycle]);
   380→          main.userData.face = SKILL_IDS[cycle];
   381→        }
   382→        ghosts.forEach((g, i) => {
   383→          g.rotation.z = main.rotation.z - 0.05 * (i + 1);
   384→          g.position.y = main.position.y + 0.012 * (i + 1);
   385→          g.material.opacity = (0.13 / (i + 1)) * (1 - t * 0.4);
   386→          g.material.map = main.material.map;
   387→        });
   388→        if (t < 1) {
   389→          requestAnimationFrame(step);
   390→        } else {
   391→          gameAudio.diceLand();
   392→          this.scene.flash(0.3);
   393→          // squash & settle
   394→          const s0 = performance.now();
   395→          const squash = () => {
   396→            const k = Math.min(1, (performance.now() - s0) / 220);
   397→            main.scale.set(1 + Math.sin(k * Math.PI) * 0.12, 1 - Math.sin(k * Math.PI) * 0.18, 1);
   398→            main.rotation.z = (1 - k) * 0.08;
   399→            if (k < 1) requestAnimationFrame(squash);
   400→            else resolve();
   401→          };
   402→          squash();
   403→        }
   404→      };
   405→      step();
   406→    });
   407→  }

   409→  /** Between-round reload: gun opens, shells drop in, cylinder spins. */
   410→  private async reloadSequence() {
   411→    this.logic.state.round += 1;
   412→    this.logic.loadCylinder();
   413→    this.tRoundStart = performance.now();
   414→    this.logic.setPhase("reloading");
   415→    this.notify();
   416→    this.busy = true;
   417→
   418→    // the Dealer sweeps the last round's casings off the table
   419→    this.scene.clearSpentShells();
   420→    this.scene.setLampBoost(0.4);
   421→    await sleep(380);
   422→    this.scene.gunAim("enemy", true);
   423→    this.scene.gunOpenCylinder(true);
   424→    gameAudio.reloadOpen();
   425→    await sleep(620);
   426→
   427→    const total = 6;
   428→    for (let i = 0; i < total; i++) {
   429→      gameAudio.reloadLoad(i);
   430→      this.onEvent({ type: "reloadStep", loaded: i + 1, total });
   431→      await sleep(190);
   432→    }
   433→    gameAudio.spinCylinder();
   434→    await sleep(560);
   435→    this.scene.gunOpenCylinder(false);
   436→    gameAudio.reloadClose();
   437→    this.scene.setLampBoost(1);
   438→    this.scene.flash(0.28);
   439→    await sleep(420);
   440→    this.scene.gunAim(null);
   441→
   442→    const live = this.logic.liveRemaining;
   443→    const blank = this.logic.blankRemaining;
   444→    this.onEvent({
   445→      type: "banner",
   446→      title: `ROUND ${this.logic.state.round}`,
   447→      sub: `LIVE ${live} — BLANK ${blank} — every fired shell lands on the table edge`,
   448→      long: true,
   449→    });
   450→    gameAudio.hammerCock();
   451→    await sleep(1400);
   452→    this.busy = false;
   453→  }

   457→  private async playerTurnStart() {
   458→    if (this.logic.state.hp.enemy <= 0 || this.logic.state.hp.player <= 0) return;
   459→    const wasPublic = this.logic.state.countdown > 0;
   460→    this.logic.setPhase("playerTurn");
   461→    this.logic.startTurnTick();
   462→    if (wasPublic && this.logic.state.countdown <= 0 && this.logic.state.cylinder.length > 0) {
   463→      this.toast("THE BOARD FADED — memorize your shells!", "bad");
   464→      gameAudio.ghostFade();
   465→    }
   466→    this.busy = false;
   467→    this.notify();
   468→  }

   470→  async playerShoot(target: Side) {
   471→    const st = this.logic.state;
   472→    if (this.busy || st.phase !== "playerTurn" || st.turn !== "player") return;
   473→    if (st.cylinder.length === 0) return;
   474→    this.busy = true;
   475→    this.notify();
   476→
   477→    // LOW-HP TENSION: below 3 HP a self-shot gets 0.7s of pure dread first.
   478→    const tension = target === "player" && st.hp.player < 3;
   479→
   480→    this.scene.gunAim(target === "player" ? "self" : "enemy");
   481→    gameAudio.hammerCock();
   482→    gameAudio.riser(tension ? 1.5 : 0.55);
   483→    await sleep(800);
   484→    if (tension) {
   485→      // 0.7s: the trigger finger trembles... 240 + 240 + 220
   486→      this.scene.addTrauma(0.14);
   487→      await sleep(240);
   488→      this.scene.addTrauma(0.18);
   489→      await sleep(240);
   490→      this.scene.addTrauma(0.22);
   491→      await sleep(220);
   492→    }
   493→
   494→    const res = await this.shootSequence("player", target, tension);
   495→    this.scene.gunAim(null);
   496→    await this.afterShot("player", res);
   497→  }

   499→  async playerUseSkill(id: SkillId) {
   500→    const st = this.logic.state;
   501→    if (this.busy || st.phase !== "playerTurn" || st.turn !== "player") return;
   502→    if (!this.logic.canUseSkill("player", id)) return;
   503→    this.busy = true;
   504→    this.notify();
   505→
   506→    const oc = this.logic.useSkill("player", id);
   507→    if (!oc.ok) {
   508→      this.busy = false;
   509→      this.notify();
   510→      return;
   511→    }
   512→    // a spent TRIAL starts its 2-turn fade away
   513→    const sk = st.skills[id];
   514→    if (sk.trial && sk.charges <= 0) this.registerGhost(id);
   515→    this.onEvent({ type: "skillFlash", id });
   516→    await this.playSkillFx("player", id, oc);
   517→    this.notify();
   518→    this.checkHealthFx();
   519→    await sleep(450);
   520→    this.busy = false;
   521→    this.notify();
   522→  }

   541→    this.notify();
   542→
   543→    this.scene.enemySetMood("idle2");
   544→    this.scene.enemySay([pick("enemyTurn")], 2.2);
   545→    await sleep(rnd(1000, 1700));
   546→
   547→    let skillsUsed = 0;
   548→    let guard = 0;
   549→    let decision: AIDecision | null = null;
   550→    // AI loop: burn skills first, then shoot
   551→    for (;;) {
   552→      if (st.hp.enemy <= 0 || st.hp.player <= 0) return;
   553→      if (this.logic.isEmpty || guard++ > 24) break;
   554→      const d = this.ai.decide();
   555→      if (d.kind === "skill" && d.skill && skillsUsed < 3) {
   556→        const oc = this.logic.useSkill("enemy", d.skill);
   557→        if (!oc.ok) continue;
   558→        skillsUsed++;
   559→        await this.playSkillFx("enemy", d.skill, oc);
   560→        this.notify();
   561→        await sleep(rnd(500, 850));
   562→        continue;
   563→      }
   564→      decision = d;
   565→      break;
   566→    }
   567→
   568→    if (decision && decision.kind === "shoot" && decision.target && !this.logic.isEmpty) {
   569→      this.scene.enemySetMood("aim");
   570→      gameAudio.riser(1.0);
   571→      await sleep(rnd(850, 1400));
   572→      const res = await this.shootSequence("enemy", decision.target);
   573→      this.scene.enemySetMood("idle1");
   574→      if (res === "dead") {
   575→        await this.endGame();
   576→        return;
   577→      }
   578→    }
   579→
   580→    if (this.logic.isEmpty) {
   581→      await this.maybeShopAndReload();
   582→    }
   583→    const next = this.passTurn();
   584→    this.notify();
   585→    if (next === "player") {
   586→      await this.playerTurnStart();
   587→    } else {
   588→      // the player is cuffed
   589→      this.toast("You're cuffed! The DEALER goes again.", "bad");
   590→      await sleep(800);
   591→      await this.enemyTurnStart();
   592→    }
   593→  }

   595→  /* ---------------- shot resolution ---------------- */
   596→
   597→  /** Fire one shot and resolve EVERYTHING that follows it:
   598→   *  the voodoo curse (fresh shell + forced shot) and the global
   599→   *  self-blank rule (one automatic free shot at the opponent). */
   600→  private async shootSequence(shooter: Side, target: Side, tensionSelf = false, forced = false): Promise<"pass" | "dead"> {
   601→    const out = this.logic.shoot(shooter, target, forced);
   602→    const res = await this.resolveShot(out, tensionSelf);
   603→    if (res === "dead") return "dead";
   604→
   605→    const opp: Side = shooter === "player" ? "enemy" : "player";
   606→
   607→    if (out.voodooReroll) {
   608→      // VOODOO CURSE: blank was rerolled — the shooter MUST fire once more,
   609→      // and the gun swings toward the opponent.
   610→      this.notify();
   611→      await sleep(300);
   612→      const out2 = this.logic.shoot(shooter, opp, true);
   613→      const res2 = await this.resolveShot(out2, false);
   614→      if (res2 === "dead") return "dead";
   615→    } else if (out.bonusShot && !this.logic.isEmpty) {
   616→      // GLOBAL RULE: shot yourself with a blank -> ONE automatic shot at the enemy.
   617→      this.notify();
   618→      if (shooter === "player") {
   619→        this.toast("BLANK! Free shot at the DEALER!", "good");
   620→        this.scene.gunAim("enemy");
   621→        gameAudio.hammerCock();
   622→        await sleep(750);
   623→      } else {
   624→        this.scene.enemySetMood("aim");
   625→        this.toast("DEALER blanks himself — free shot at you!", "bad");
   626→        await sleep(750);
   627→      }
   628→      const out2 = this.logic.shoot(shooter, opp);
   629→      const res2 = await this.resolveShot(out2, false);
   630→      if (res2 === "dead") return "dead";
   631→    }
   632→    return "pass";
   633→  }

   635→  /** Animation + effects for ONE trigger pull (both sides). */
   636→  private async resolveShot(out: ShotOutcome, tensionSelf = false): Promise<"pass" | "dead"> {
   637→    const shooterIsPlayer = out.shooter === "player";
   638→
   639→    // PHYSICAL RECOUNT: the spent shell lands on the table's front edge,
   640→    // in firing order — the round's history stays visible and checkable.
   641→    this.scene.spawnSpentShell(
   642→      out.shell,
   643→      shooterIsPlayer
   644→        ? this.scene.muzzleWorldPos()
   645→        : this.scene.enemyWorldPos().add(new THREE.Vector3(0.35, 0.42, 0.3)),
   646→    );
   647→
   648→    if (out.voodooReroll) {
   649→      gameAudio.blankClick();
   650→      this.toast("VOODOO: blank rerolled — a FRESH shell enters the chamber! Now FIRE at the enemy!", "voodoo");
   651→      await sleep(700);
   652→      gameAudio.spinCylinder();
   653→      this.scene.addTrauma(0.25);
   654→      await sleep(900);
   655→      this.notify();
   656→      return "pass";
   657→    }
   658→
   659→    // CASH: every hit you land on the DEALER pays (reflections pay extra).
   660→    if (out.damageTo === "enemy") {
   661→      const gain = out.reflected ? 150 : 100;
   662→      this.money += gain;
   663→      this.toast(`+$${gain}${out.reflected ? " (reflected!)" : ""}`, "good");
   664→    }
   665→
   666→    if (out.phoenixSaved) {
   667→      // the killing blow burned the charm instead
   668→      gameAudio.phoenix();
   669→      this.scene.flash(0.65);
   670→      this.scene.setDesat(0.2);
   671→      setTimeout(() => this.scene.setDesat(0), 700);
   672→      this.registerGhost("phoenix");
   673→      this.scene.enemySetMood("scared");
   674→      this.scene.enemySay([pick("phoenix")], 2.6);
   675→      this.onEvent({ type: "banner", title: "PHOENIX!", sub: "You refuse to die — the charm burns for you." });
   676→      await sleep(1200);
   677→    }
   678→
   679→    if (out.shell === "live") {
   680→      gameAudio.gunshot();
   681→      if (shooterIsPlayer) {
   682→        this.scene.muzzleFlash();
   683→        this.scene.gunKick();
   684→      } else {
   685→        this.scene.addTrauma(0.5);
   686→        this.scene.flash(0.4);
   687→      }
   688→      // bullet-exit delay: at low HP a self-shot lingers 0.2s longer
   689→      await sleep(tensionSelf ? 340 : 140);
   690→
   691→      if (out.reflected) {
   692→        // voodoo reflected the pain back to the shooter
   693→        gameAudio.voodooReflect();
   694→        this.scene.flash(0.55);
   695→        this.onEvent({ type: "dmgPop", side: out.damageTo!, text: "REFLECTED!", tone: "reflect" });
   696→        if (out.damageTo === "enemy") {
   697→          this.scene.bloodBurst("enemy");
   698→          this.scene.enemySetMood("hit");
   699→          this.scene.enemySay([pick("reflect")], 2.2);
   700→        } else {
   701→          this.scene.bloodBurst("player");
   702→          this.onEvent({ type: "screenSplat" });
   703→        }
   704→      } else if (out.damageTo === "player") {
   705→        gameAudio.hitPlayer();
   706→        this.scene.addTrauma(0.85);
   707→        this.scene.setAberration(2.4);
   708→        setTimeout(() => this.scene.setAberration(0), 380);
   709→        this.onEvent({ type: "screenSplat" });
   710→        this.scene.bloodBurst("player");
   711→        if (!shooterIsPlayer) {
   712→          this.scene.enemySetMood("laugh");
   713→          this.onEvent({ type: "dmgPop", side: "player", text: "-1 HP", tone: "hit" });
   714→        } else {
   715→          this.scene.enemySetMood("laugh");
   716→          this.scene.enemySay([pick("playerHitSelf")], 2.4);
   717→          this.onEvent({ type: "dmgPop", side: "player", text: "-1 HP", tone: "hit" });
   718→        }
   719→      } else if (out.damageTo === "enemy") {
   720→        gameAudio.hitEnemy();
   721→        this.scene.addTrauma(0.4);
   722→        this.scene.bloodBurst("enemy", !!out.death);
   723→        this.scene.enemySetMood("hit");
   724→        this.onEvent({ type: "dmgPop", side: "enemy", text: "-1 HP", tone: "hit" });
   725→        await sleep(650);
   726→        if (out.death !== "enemy") {
   727→          this.scene.enemySetMood("grin");
   728→          this.scene.enemySay([pick("playerHitsEnemy")], 2.2);
   729→        }
   730→      }
   731→      await sleep(520);
   732→    } else {
   733→      // blank
   734→      gameAudio.blankClick();
   735→      if (shooterIsPlayer) {
   736→        this.scene.dustPuffAtCamera();
   737→      } else {
   738→        this.scene.enemySetMood("scared");
   739→      }
   740→      this.onEvent({ type: "dmgPop", side: out.target, text: "CLICK! blank", tone: "blank" });
   741→      const who =
   742→        !shooterIsPlayer && out.target === "enemy"
   743→          ? pick("enemyBlankSelf")
   744→          : shooterIsPlayer && out.target === "player"
   745→            ? pick("playerBlankSelf")
   746→            : pick("playerShootsBlank");
   747→      if (!shooterIsPlayer) {
   748→        this.scene.enemySay([who], 2);
   749→      }
   750→      await sleep(500);
   751→    }
   752→
   753→    if (out.death) return "dead";
   754→    this.notify();
   755→    this.checkHealthFx();
   756→    return "pass";
   757→  }

   936→  /** Every passed turn dissolves the ghosts a little more. */
   937→  private tickGhosts() {
   938→    if (!this.ghosts.length) return;
   939→    for (const g of this.ghosts) g.turns -= 1;
   940→    const done = this.ghosts.filter((g) => g.turns <= 0).map((g) => g.id);
   941→    for (const id of done) {
   942→      const sk = this.logic.state.skills[id];
   943→      sk.owned = false;
   944→      sk.trial = false;
   945→      this.toast(`${SKILL_META[id].name} faded away.`, "info");
   946→    }
   947→    this.ghosts = this.ghosts.filter((g) => g.turns > 0);
   948→  }

  1016→    if (!offer || offer.sold) return;

  1022→        poor ? `Not enough cash — $${offer.price} needed.` : "Hands full — 4 skills max!",

  1062→  private shopNdc(e: MouseEvent): THREE.Vector2 | null {
  1063→    const c = this.container;
  1064→    if (!c) return null;
  1065→    const r = c.getBoundingClientRect();
  1066→    return new THREE.Vector2(((e.clientX - r.left) / r.width) * 2 - 1, -((e.clientY - r.top) / r.height) * 2 + 1);
  1067→  }

  1069→  private onShopPointerMove(e: MouseEvent) {
  1070→    if (this.logic.state.phase !== "shop") {
  1071→      if (this.hoverId !== null) this.setShopHover(null);
  1072→      if (this.container) this.container.style.cursor = "default";
  1073→      return;
  1074→    }
  1075→    const ndc = this.shopNdc(e);
  1076→    if (!ndc) return;
  1077→    const hit = this.scene.pickShop(ndc);
  1078→    const id = hit ? (hit.type === "bell" ? "bell" : hit.id) : null;
  1079→    if (id !== this.hoverId) {
  1080→      this.setShopHover(id);
  1081→      if (id) gameAudio.hover();
  1082→    }
  1083→    if (this.container) this.container.style.cursor = id ? "pointer" : "default";
  1084→  }

  1086→  private onShopClick(e: MouseEvent) {
  1087→    if (this.logic.state.phase !== "shop") return;
  1088→    const ndc = this.shopNdc(e);
  1089→    if (!ndc) return;
  1090→    const hit = this.scene.pickShop(ndc);
  1091→    if (!hit) return;
  1092→    if (hit.type === "bell") void this.ringBell();
  1093→    else this.buyTrial(hit.id);
  1094→  }

  1096→  /** Hover changed: tip the 3D item and anchor the HTML tooltip above it. */
  1097→  setShopHover(id: SkillId | "bell" | null) {
  1098→    this.hoverId = id;
  1099→    this.scene.setShopHover(id);
  1100→    if (id === null) {
  1101→      this.onEvent({ type: "shopTip", id: null, xPct: 0, yPct: 0 });
  1102→      return;
  1103→    }
  1104→    const anchor = this.scene.shopAnchor(id);
  1105→    if (anchor) this.onEvent({ type: "shopTip", id, xPct: anchor.xPct, yPct: anchor.yPct });

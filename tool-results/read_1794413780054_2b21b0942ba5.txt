    40→const FinalShader = {
    41→  uniforms: {
    42→    tDiffuse: { value: null as THREE.Texture | null },
    43→    uTime: { value: 0 },
    44→    uAberration: { value: 0 },
    45→    uVignette: { value: 0.55 },
    46→    uFlash: { value: 0 },
    47→    uDesat: { value: 0 },
    48→    uRedPulse: { value: 0 },
    49→    uGrain: { value: 0.045 },
    50→  },
    51→  vertexShader: /* glsl */ `
    52→    varying vec2 vUv;
    53→    void main() {
    54→      vUv = uv;
    55→      gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0);
    56→    }
    57→  `,
    58→  fragmentShader: /* glsl */ `
    59→    uniform sampler2D tDiffuse;
    60→    uniform float uTime;
    61→    uniform float uAberration;
    62→    uniform float uVignette;
    63→    uniform float uFlash;
    64→    uniform float uDesat;
    65→    uniform float uRedPulse;
    66→    uniform float uGrain;
    67→    varying vec2 vUv;
    68→
    69→    float hash(vec2 p) {
    70→      return fract(sin(dot(p, vec2(12.9898, 78.233))) * 43758.5453);
    71→    }
    72→
    73→    void main() {
    74→      vec2 uv = vUv;
    75→      vec2 dir = uv - 0.5;
    76→      float d = length(dir);
    77→      float ab = uAberration * (0.004 + d * 0.02);
    78→      float r = texture2D(tDiffuse, uv + dir * ab).r;
    79→      float g = texture2D(tDiffuse, uv).g;
    80→      float b = texture2D(tDiffuse, uv - dir * ab).b;
    81→      vec3 col = vec3(r, g, b);
    82→      col = mix(col, vec3(1.0, 0.97, 0.9), uFlash);
    83→      col = mix(col, vec3(0.72, 0.06, 0.04), uRedPulse * (0.3 + 0.7 * d));
    84→      float l = dot(col, vec3(0.299, 0.587, 0.114));
    85→      col = mix(col, vec3(l), uDesat);
    86→      col *= 1.0 - uVignette * smoothstep(0.3, 0.82, d);
    87→      float n = hash(uv * (1.0 + fract(uTime) * 13.7) + fract(uTime) * 7.3);
    88→      col += (n - 0.5) * uGrain;
    89→      gl_FragColor = vec4(col, 1.0);
    90→    }
    91→  `,
    92→};

   105→function doodleMat(tex: THREE.Texture, opts: { lit?: boolean; alphaTest?: number } = {}) {
   106→  if (opts.lit) {
   107→    return new THREE.MeshLambertMaterial({
   108→      map: tex,
   109→      transparent: true,
   110→      alphaTest: opts.alphaTest ?? 0.02,
   111→      side: THREE.DoubleSide,
   112→      emissive: new THREE.Color(0x8a8378),
   113→      emissiveMap: tex,
   114→      emissiveIntensity: 0.55,
   115→    });
   116→  }
   117→  return new THREE.MeshBasicMaterial({
   118→    map: tex,
   119→    transparent: true,
   120→    alphaTest: opts.alphaTest ?? 0.02,
   121→    side: THREE.DoubleSide,
   122→    depthWrite: true,
   123→  });
   124→}

   126→function ctex(cv: HTMLCanvasElement, repeat?: [number, number]) {
   127→  const t = new THREE.CanvasTexture(cv);
   128→  t.colorSpace = THREE.SRGBColorSpace;
   129→  t.anisotropy = 4;
   130→  if (repeat) {
   131→    t.wrapS = THREE.RepeatWrapping;
   132→    t.wrapT = THREE.RepeatWrapping;
   133→    t.repeat.set(repeat[0], repeat[1]);
   134→  }
   135→  return t;
   136→}

   196→    const w = container.clientWidth;

   230→    this.composer.addPass(this.bloomPass);

   233→
   234→    window.addEventListener("resize", this.onResize);

   239→  private onResize = () => {
   240→    const el = this.renderer.domElement.parentElement;
   241→    if (!el) return;
   242→    const w = el.clientWidth;
   243→    const h = el.clientHeight;
   244→    this.camera.aspect = w / h;
   245→    this.camera.updateProjectionMatrix();
   246→    this.renderer.setSize(w, h);
   247→    this.composer.setSize(w, h);
   248→  };

   581→  private spawn(
   582→    tex: THREE.Texture,
   583→    pos: THREE.Vector3,
   584→    opts: {
   585→      size?: number;
   586→      vel?: THREE.Vector3;
   587→      rotVel?: number;
   588→      life?: number;
   589→      grow?: number;
   590→      gravity?: number;
   591→      fadePow?: number;
   592→      blending?: THREE.Blending;
   593→      lit?: boolean;
   594→      parent?: THREE.Object3D;
   595→    } = {},
   596→  ) {
   597→    const mat = new THREE.MeshBasicMaterial({
   598→      map: tex,
   599→      transparent: true,
   600→      depthWrite: false,
   601→      blending: opts.blending ?? THREE.NormalBlending,
   602→      opacity: 1,
   603→      side: THREE.DoubleSide,
   604→    });
   605→    const mesh = new THREE.Mesh(new THREE.PlaneGeometry(opts.size ?? 0.3, opts.size ?? 0.3), mat);
   606→    mesh.position.copy(pos);
   607→    (opts.parent ?? this.scene).add(mesh);
   608→    // billboard ke kamera untuk partikel dunia
   609→    if (!opts.parent) mesh.lookAt(this.camera.getWorldPosition(new THREE.Vector3()));
   610→    this.particles.push({
   611→      mesh,
   612→      vel: opts.vel ?? new THREE.Vector3(),
   613→      rotVel: opts.rotVel ?? 0,
   614→      life: 0,
   615→      maxLife: opts.life ?? 1,
   616→      grow: opts.grow ?? 0,
   617→      gravity: opts.gravity ?? 0,
   618→      fadePow: opts.fadePow ?? 1,
   619→    });
   620→    return mesh;
   621→  }

   731→  /** Shell melayang (reload / eject). onDone dipanggil saat selesai. */
   732→  flyShell(kind: "live" | "blank", from: THREE.Vector3, to: THREE.Vector3, dur: number, onDone?: () => void) {
   733→    const mat = new THREE.MeshBasicMaterial({ map: this.shellTex[kind], transparent: true, side: THREE.DoubleSide });
   734→    const mesh = new THREE.Mesh(new THREE.PlaneGeometry(0.16, 0.16), mat);
   735→    mesh.position.copy(from);
   736→    this.scene.add(mesh);
   737→    const start = performance.now();
   738→    const step = () => {
   739→      const t = Math.min(1, (performance.now() - start) / (dur * 1000));
   740→      const e = 1 - Math.pow(1 - t, 2);
   741→      mesh.position.lerpVectors(from, to, e);
   742→      mesh.position.y += Math.sin(t * Math.PI) * 0.55;
   743→      mesh.rotation.z += 0.22;
   744→      mesh.lookAt(this.camera.getWorldPosition(new THREE.Vector3()));
   745→      if (t < 1) requestAnimationFrame(step);
   746→      else {
   747→        this.scene.remove(mesh);
   748→        mesh.geometry.dispose();
   749→        mat.dispose();
   750→        onDone?.();
   751→      }
   752→    };
   753→    requestAnimationFrame(step);
   754→  }

   834→  /* ---------------- render loop ---------------- */
   835→
   836→  render(now: number) {
   837→    const dt0 = Math.min(0.1, (now - (this.lastNow ?? now)) / 1000);
   838→    this.lastNow = now;
   839→    this.timeScale += (this.timeScaleTarget - this.timeScale) * Math.min(1, dt0 * 6);
   840→    const dt = dt0 * this.timeScale;
   841→    const t = now / 1000;
   842→
   843→    // kamera: parallax + napas + shake
   844→    this.lookX += (this.lookXT - this.lookX) * Math.min(1, dt0 * 5);
   845→    this.lookY += (this.lookYT - this.lookY) * Math.min(1, dt0 * 5);
   846→    this.trauma = Math.max(0, this.trauma - dt0 * 1.7);
   847→    const sh = this.trauma * this.trauma;
   848→    const n1 = Math.sin(t * 61) * Math.sin(t * 23.7);
   849→    const n2 = Math.cos(t * 47.3) * Math.sin(t * 31.1);
   850→    this.shakeOff.set(n1 * sh * 0.09, n2 * sh * 0.07);
   851→
   852→    this.camGroup.rotation.y = -this.lookX * 0.09 + n1 * sh * 0.05;
   853→    this.camGroup.rotation.x = -this.lookY * 0.06 + n2 * sh * 0.045;
   854→    this.camGroup.position.x = this.shakeOff.x;
   855→    this.camGroup.position.y = Math.sin(t * 1.7) * 0.012 + this.shakeOff.y;
   856→
   857→    // lampu flicker
   858→    this.lampFlicker += (1 - this.lampFlicker) * Math.min(1, dt0 * 8);
   859→    if (Math.random() < 0.012) this.lampFlicker = 0.55 + Math.random() * 0.3;
   860→    const flick = this.lampFlicker * this.lampBoost * (0.92 + Math.sin(t * 13.3) * 0.05);
   861→    this.spot.intensity = 85 * flick;
   862→    this.ambience.intensity = 0.55 * flick;
   863→    this.lampPivot.rotation.z = Math.sin(t * 0.9) * 0.035;
   864→    this.lampPivot.rotation.x = Math.cos(t * 0.7) * 0.02;
   865→
   866→    // enemy: idle bobbing
   867→    this.enemyWobble += dt;
   868→    this.enemyMesh.position.y = this.enemyBasePos.y + Math.sin(t * 1.9) * 0.02;
   869→    this.enemyMesh.rotation.z = Math.sin(t * 1.3) * 0.012;
   870→
   871→    // speech bubble
   872→    if (this.speechTimer > 0) {
   873→      this.speechTimer -= dt0;
   874→      const s = Math.min(1, this.speechMesh.scale.x + dt0 * 5);
   875→      this.speechMesh.scale.set(s, s, 1);
   876→      if (this.speechTimer <= 0) {
   877→        this.speechMesh.material.opacity = 0;
   878→      }
   879→    }
   880→
   881→    // gun lerp + recoil
   882→    this.gunCurrent.lerp(this.gunTarget, Math.min(1, dt0 * 7));
   883→    this.gunRecoil = Math.max(0, this.gunRecoil - dt0 * 5.5);
   884→    this.fovKick = Math.max(0, this.fovKick - dt0 * 4);
   885→    const targetFov = 56 + this.fovKick * 4.5;
   886→    if (Math.abs(this.camera.fov - targetFov) > 0.01) {
   887→      this.camera.fov = targetFov;
   888→      this.camera.updateProjectionMatrix();
   889→    }
   890→    this.gunMesh.position.copy(this.gunCurrent);
   891→    this.gunMesh.position.z += this.gunRecoil * 0.22;
   892→    this.gunMesh.rotation.x = this.gunRecoil * 0.3 - Math.sin(t * 1.7) * 0.008;
   893→    this.gunMesh.rotation.z = this.gunPose === "self" ? -0.06 : Math.sin(t * 1.4) * 0.006;
   894→
   895→    // muzzle fade
   896→    if (this.muzzleMesh.material.opacity > 0) {
   897→      this.muzzleMesh.material.opacity = Math.max(0, this.muzzleMesh.material.opacity - dt0 * 7);
   898→      this.muzzleLight.intensity = Math.max(0, this.muzzleLight.intensity - dt0 * 260);
   899→    }
   900→
   901→    // flash & pulse fade
   902→    const u = this.finalPass.uniforms;
   903→    u.uFlash.value = Math.max(0, u.uFlash.value - dt0 * 2.6);
   904→    u.uTime.value = t;
   905→
   906→    // partikel
   907→    for (let i = this.particles.length - 1; i >= 0; i--) {
   908→      const p = this.particles[i];
   909→      p.life += dt;
   910→      const k = p.life / p.maxLife;
   911→      if (k >= 1) {
   912→        (p.mesh.parent ?? this.scene).remove(p.mesh);
   913→        p.mesh.geometry.dispose();
   914→        (p.mesh.material as THREE.Material).dispose();
   915→        this.particles.splice(i, 1);
   916→        continue;
   917→      }
   918→      p.vel.y += p.gravity * dt;
   919→      p.mesh.position.addScaledVector(p.vel, dt);
   920→      p.mesh.rotation.z += p.rotVel * dt;
   921→      const sc = 1 + p.grow * p.life;
   922→      p.mesh.scale.set(sc, sc, 1);
   923→      (p.mesh.material as THREE.MeshBasicMaterial).opacity = Math.pow(1 - k, p.fadePow);
   924→    }
   925→
   926→    this.onFrame?.(dt, t);
   927→    this.composer.render();
   928→  }

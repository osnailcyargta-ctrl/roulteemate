     1→"use client";
     2→
     3→/**
     4→ * GameShell — React layer for the ULTIMATE ROULETTE HUD.
     5→ * Mounts the engine (Three.js) plus all doodle overlays: menu, HP,
     6→ * shell counter, skill bar, toasts, banner, rules, win/lose screens,
     7→ * and the low-HP blood vignette.
     8→ */
     9→
    10→import { useCallback, useEffect, useMemo, useRef, useState } from "react";
    11→import { GameEngine, MAX_SKILLS, SHOP_META, type HudEvent, type HudSnapshot } from "@/game/engine";
    12→import { SKILL_META, dataURL, texBulletPip, texHeart, texSkillIcon, texSplat, texVoodooMask, type SkillId } from "@/game/doodle";
    13→import type { ShellType } from "@/game/logic";
    14→import { gameAudio } from "@/game/audio";
    15→
    16→type Toast = { id: number; text: string; tone: string };
    17→type Pop = { id: number; side: string; text: string; tone: string };
    18→type Splat = { id: number; url: string; left: number; top: number; size: number; rot: number };
    19→type Stamp = { id: number; idSkill: SkillId };
    20→
    21→const DIFFS: { id: "easy" | "normal" | "insane"; label: string; desc: string }[] = [
    22→  { id: "easy", label: "EASY", desc: "6 vs 6 HP. The Dealer has 1 skill." },
    23→  { id: "normal", label: "NORMAL", desc: "5 vs 5 HP. The Dealer has 2 skills." },
    24→  { id: "insane", label: "INSANE", desc: "You 4 HP vs his 5. 3 skills + shop legends." },
    25→];
    26→
    27→const fmtTime = (ms: number) => {
    28→  const s = Math.max(0, Math.floor(ms / 1000));
    29→  return `${String(Math.floor(s / 60)).padStart(2, "0")}:${String(s % 60).padStart(2, "0")}`;
    30→};
    31→
    32→export default function GameShell() {
    33→  const mountRef = useRef<HTMLDivElement>(null);
    34→  const engineRef = useRef<GameEngine | null>(null);
    35→  const evId = useRef(0);
    36→  const [ready, setReady] = useState(false);
    37→  const [snap, setSnap] = useState<HudSnapshot | null>(null);
    38→  const [toasts, setToasts] = useState<Toast[]>([]);
    39→  const [pops, setPops] = useState<Pop[]>([]);
    40→  const [splats, setSplats] = useState<Splat[]>([]);
    41→  const [stamps, setStamps] = useState<Stamp[]>([]);
    42→  const [banner, setBanner] = useState<{ title: string; sub?: string } | null>(null);
    43→  const [reveal, setReveal] = useState<{ shell: "live" | "blank" } | null>(null);
    44→  const [showHelp, setShowHelp] = useState(false);
    45→  const [muted, setMuted] = useState(false);
    46→  const [diff, setDiff] = useState<"easy" | "normal" | "insane">("normal");
    47→  const [reloadPips, setReloadPips] = useState(0);
    48→  const [flashRed, setFlashRed] = useState(false);
    49→  const [revealQueue, setRevealQueue] = useState<(ShellType | null)[] | null>(null);
    50→  const [times, setTimes] = useState<{ roundMs: number; totalMs: number }>({ roundMs: 0, totalMs: 0 });
    51→
    52→  /* ---------- HUD asset dataURLs (after mount, client-only) ---------- */
    53→  const assets = useMemo(() => {
    54→    if (!ready) return null;
    55→    return {
    56→      heartFull: dataURL("heart-full", texHeart("full")),
    57→      heartEmpty: dataURL("heart-empty", texHeart("empty")),
    58→      pipLive: dataURL("pip-live", texBulletPip("live")),
    59→      pipBlank: dataURL("pip-blank", texBulletPip("blank")),
    60→      voodoo: dataURL("voodoo-mask", texVoodooMask()),
    61→      splat0: dataURL("splat-hud-0", texSplat(333)),
    62→      splat1: dataURL("splat-hud-1", texSplat(444)),
    63→    };
    64→  }, [ready]);
    65→
    66→  const assetsRef = useRef(assets);
    67→  useEffect(() => {
    68→    assetsRef.current = assets;
    69→  }, [assets]);
    70→
    71→  const handleEvent = useCallback((e: HudEvent) => {
    72→    const id = ++evId.current;
    73→    switch (e.type) {
    74→      case "toast":
    75→        setToasts((t) => [...t.slice(-4), { id, text: e.text, tone: e.tone ?? "info" }]);
    76→        setTimeout(() => setToasts((t) => t.filter((x) => x.id !== id)), 3200);
    77→        break;
    78→      case "banner":
    79→        setBanner({ title: e.title, sub: e.sub });
    80→        setTimeout(() => setBanner(null), 1700);
    81→        break;
    82→      case "skillGet":
    83→        setStamps((s) => [...s, { id, idSkill: e.id }]);
    84→        setTimeout(() => setStamps((s) => s.filter((x) => x.id !== id)), 1500);
    85→        break;
    86→      case "shellReveal":
    87→        if (e.shell) {
    88→          setReveal({ shell: e.shell });
    89→          setTimeout(() => setReveal(null), 1900);
    90→        }
    91→        break;
    92→      case "shellRevealQueue":
    93→        setRevealQueue(e.shells);
    94→        setTimeout(() => setRevealQueue(null), 2800);
    95→        break;
    96→      case "dmgPop":
    97→        setPops((p) => [...p.slice(-5), { id, side: e.side, text: e.text, tone: e.tone ?? "hit" }]);
    98→        setTimeout(() => setPops((p) => p.filter((x) => x.id !== id)), 1500);
    99→        break;
   100→      case "screenSplat": {
   101→        const url = evId.current % 2 === 0 ? "splat-hud-0" : "splat-hud-1";
   102→        const a = assetsRef.current;
   103→        if (!a) break;
   104→        setSplats((s) => [
   105→          ...s.slice(-3),
   106→          {
   107→            id,
   108→            url: a[url],
   109→            left: 15 + Math.random() * 55,
   110→            top: 8 + Math.random() * 50,
   111→            size: 130 + Math.random() * 160,
   112→            rot: Math.random() * 360,
   113→          },
   114→        ]);
   115→        setFlashRed(true);
   116→        setTimeout(() => setFlashRed(false), 180);
   117→        setTimeout(() => setSplats((s) => s.filter((x) => x.id !== id)), 2600);
   118→        break;
   119→      }
   120→      case "skillFlash":
   121→        break;
   122→      case "reloadStep":
   123→        setReloadPips(e.loaded);
   124→        setTimeout(() => setReloadPips(0), 2400);
   125→        break;
   126→    }
   127→  }, []);
   128→
   129→  /* ---------- mount engine ---------- */
   130→  useEffect(() => {
   131→    let disposed = false;
   132→    (async () => {
   133→      try {
   134→        await document.fonts.ready;
   135→      } catch {
   136→        /* noop */
   137→      }
   138→      if (disposed || !mountRef.current) return;
   139→      const engine = new GameEngine();
   140→      engineRef.current = engine;
   141→      (window as unknown as { __eng?: GameEngine }).__eng = engine;
   142→      engine.onHud = (s) => setSnap(s);
   143→      engine.onEvent = (e) => handleEvent(e);
   144→      engine.mount(mountRef.current);
   145→      setReady(true);
   146→    })();
   147→    return () => {
   148→      disposed = true;
   149→      engineRef.current?.dispose();
   150→      engineRef.current = null;
   151→    };
   152→
   153→  }, []);
   154→
   155→  const toggleMute = useCallback(() => {
   156→    setMuted((prev) => {
   157→      const m = !prev;
   158→      gameAudio.setMuted(m);
   159→      return m;
   160→    });
   161→  }, []);
   162→
   163→  /* ---------- keyboard ---------- */
   164→  useEffect(() => {
   165→    const onKey = (ev: KeyboardEvent) => {
   166→      const eng = engineRef.current;
   167→      if (!eng || !snap) return;
   168→      const k = ev.key.toLowerCase();
   169→      if (k === "q") void eng.playerShoot("player");
   170→      else if (k === "e") void eng.playerShoot("enemy");
   171→      else if (k === "h") setShowHelp((v) => !v);
   172→      else if (k === "m") toggleMute();
   173→      else if (/^[1-4]$/.test(k)) {
   174→        const owned = snap.skills.filter((s) => s.owned && s.ghostTurns <= 0);
   175→        const target = owned[parseInt(k) - 1];
   176→        if (target) void eng.playerUseSkill(target.id);
   177→      }
   178→    };
   179→    window.addEventListener("keydown", onKey);
   180→    return () => window.removeEventListener("keydown", onKey);
   181→
   182→  }, [snap, toggleMute]);
   183→
   184→  /* ---------- playtime ticker (round + total) ---------- */
   185→  const phaseKey = snap?.phase ?? "menu";
   186→  useEffect(() => {
   187→    if (phaseKey === "menu") return;
   188→    const tick = () => setTimes(engineRef.current?.getTimes() ?? { roundMs: 0, totalMs: 0 });
   189→    tick();
   190→    const iv = setInterval(tick, 1000);
   191→    return () => clearInterval(iv);
   192→  }, [phaseKey]);
   193→
   194→  const isPlayerTurn = snap?.phase === "playerTurn" && snap.turn === "player" && !snap.busy;
   195→  const phase = snap?.phase ?? "menu";
   196→  const inGame = phase !== "menu";
   197→  const lowHp = inGame && snap ? snap.hpPlayer : 0;
   198→
   199→  /* ---------- shop catalog (pure, from snapshot) ---------- */
   200→  const activeSkillCount = snap
   201→    ? snap.skills.filter((s) => s.owned && !(s.trial && s.charges <= 0)).length
   202→    : 0;
   203→  const catalog = snap
   204→    ? SHOP_META.filter((m) => snap.difficulty === "insane" || !m.legendary)
   205→        .map((m) => {
   206→          const sk = snap.skills.find((s) => s.id === m.id)!;
   207→          let disabled = false;
   208→          let reason: string | null = null;
   209→          if (sk.owned) {
   210→            disabled = true;
   211→            reason = sk.trial && sk.charges <= 0 ? "FADING AWAY" : "ALREADY HELD";
   212→          } else if (activeSkillCount >= MAX_SKILLS) {
   213→            disabled = true;
   214→            reason = `SLOTS FULL (${MAX_SKILLS})`;
   215→          } else if (snap.money < m.price) {
   216→            disabled = true;
   217→            reason = "TOO POOR";
   218→          }
   219→          return { ...m, disabled, reason };
   220→        })
   221→    : [];
   222→
   223→  /* ---------- render ---------- */
   224→  return (
   225→    <div className="fixed inset-0 overflow-hidden bg-[#0b0a08] select-none">
   226→      {/* 3D world */}
   227→      <div ref={mountRef} className="absolute inset-0" />
   228→
   229→      {/* cinematic overlays */}
   230→      <div className="pointer-events-none absolute inset-0 vignette-overlay z-10" />
   231→      <div className="pointer-events-none absolute inset-0 grain-overlay z-10" />
   232→      {flashRed && <div className="pointer-events-none absolute inset-0 z-10 bg-[rgba(199,62,46,0.35)]" />}
   233→
   234→      {/* LOW-HP BLOOD VIGNETTE: starts at 2 HP, thickest at 1 HP */}
   235→      {snap && lowHp === 2 && (
   236→        <div className="pointer-events-none absolute inset-0 z-10 blood-vignette" />
   237→      )}
   238→      {snap && lowHp === 1 && (
   239→        <>
   240→          <div className="pointer-events-none absolute inset-0 z-10 blood-vignette heavy" />
   241→          <div className="pointer-events-none absolute inset-0 z-10 overflow-hidden">
   242→            <span className="blood-drip drip-1" />
   243→            <span className="blood-drip drip-2" />
   244→            <span className="blood-drip drip-3" />
   245→            <span className="blood-drip drip-4" />
   246→          </div>
   247→          <div className="pointer-events-none absolute inset-0 z-10 pulse-danger" />
   248→        </>
   249→      )}
   250→
   251→      {snap && snap.voodooPlayer > 0 && phase !== "menu" && (
   252→        <div
   253→          className="pointer-events-none absolute inset-0 z-10"
   254→          style={{ boxShadow: "inset 0 0 120px rgba(94,156,91,0.45)" }}
   255→        />
   256→      )}
   257→
   258→      {/* blood on screen */}
   259→      <div className="pointer-events-none absolute inset-0 z-20 overflow-hidden">
   260→        {assets &&
   261→          splats.map((s) => (
   262→
   263→            <img
   264→              key={s.id}
   265→              src={s.url}
   266→              alt=""
   267→              className="absolute pop-in"
   268→              style={{
   269→                left: `${s.left}%`,
   270→                top: `${s.top}%`,
   271→                width: s.size,
   272→                transform: `rotate(${s.rot}deg)`,
   273→                transition: "opacity 1.2s ease 1s",
   274→                opacity: 0.92,
   275→              }}
   276→            />
   277→          ))}
   278→      </div>
   279→
   280→      {/* damage pops */}
   281→      <div className="pointer-events-none absolute inset-0 z-30">
   282→        {pops.map((p) => (
   283→          <div
   284→            key={p.id}
   285→            className={`absolute rise-fade font-marker text-3xl drop-shadow-[3px_3px_0_rgba(0,0,0,0.7)] ${
   286→              p.tone === "reflect" ? "text-[#8fdc8a]" : p.tone === "blank" ? "text-[#cfd8e3]" : "text-[#ff6b57]"
   287→            }`}
   288→            style={p.side === "enemy" ? { left: "58%", top: "28%" } : { left: "42%", top: "62%" }}
   289→          >
   290→            {p.text}
   291→          </div>
   292→        ))}
   293→      </div>
   294→
   295→      {/* ================= IN-GAME HUD ================= */}
   296→      {inGame && assets && snap && (
   297→        <>
   298→          {/* shell counter board — honest for 3 turns, then it fades */}
   299→          <div className="absolute left-1/2 top-3 z-30 -translate-x-1/2">
   300→            <div
   301→              className={`doodle-box px-4 py-1.5 flex items-center gap-3 ${
   302→                !snap.countsPublic ? "doodle-dark board-dark-box" : ""
   303→              }`}
   304→            >
   305→              {snap.countsPublic ? (
   306→                <>
   307→                  <div className="flex items-center gap-1">
   308→                    <img src={assets.pipLive} alt="live" className="h-7 w-7" />
   309→                    <span className="font-marker text-xl text-[#8f2418]">x{snap.live}</span>
   310→                  </div>
   311→                  <div className="h-6 w-0.5 bg-[#221a12]/30" />
   312→                  <div className="flex items-center gap-1">
   313→                    <img src={assets.pipBlank} alt="blank" className="h-7 w-7" />
   314→                    <span className="font-marker text-xl text-[#57697c]">x{snap.blank}</span>
   315→                  </div>
   316→                </>
   317→              ) : (
   318→                <div className="pop-in flex items-center gap-2">
   319→                  <span className="font-marker text-xl tracking-widest text-[#8b8577]">? ? ?</span>
   320→                  <span className="font-hand-sc text-xs text-[#d9a441] rotate-[-2deg]">memorize!</span>
   321→                </div>
   322→              )}
   323→            </div>
   324→            {snap.countsPublic && (
   325→              <div className="mt-1 flex justify-center">
   326→                <div
   327→                  className={`font-hand-sc border-2 border-[#221a12]/40 bg-[#f3ecd7]/85 px-2 py-0.5 text-xs rotate-[1.5deg] ${
   328→                    snap.countdown <= 1 ? "blink-soft text-[#8f2418]" : "text-[#4a4036]"
   329→                  }`}
   330→                >
   331→                  BOARD FADES IN {snap.countdown}
   332→                </div>
   333→              </div>
   334→            )}
   335→            {snap.known && (
   336→              <div className="mt-1 flex justify-center">
   337→                <div
   338→                  className={`pop-in font-hand-sc px-2 py-0.5 border-2 border-[#221a12] bg-[#f3ecd7] text-sm rotate-[-2deg] ${
   339→                    snap.known === "live" ? "text-[#8f2418]" : "text-[#57697c]"
   340→                  }`}
   341→                >
   342→                  CHAMBER: {snap.known.toUpperCase()}
   343→                </div>
   344→              </div>
   345→            )}
   346→          </div>
   347→
   348→          {/* top left: round + buttons */}
   349→          <div className="absolute left-2 top-2 z-30 flex flex-col gap-1.5 sm:left-3 sm:top-3 sm:gap-2">
   350→            <div className="doodle-box doodle-dark px-2 py-0.5 font-marker text-xs rotate-[-1.5deg] sm:px-3 sm:py-1 sm:text-sm">
   351→              ROUND {snap.round}
   352→            </div>
   353→            <div
   354→              className="doodle-box w-fit px-2 py-0.5 font-marker text-xs text-[#8a6d1d] rotate-[1deg] sm:text-sm"
   355→              title="Cash — land bullets ($100), voodoo reflections ($150), survive rounds ($75)"
   356→            >
   357→              $ {snap.money}
   358→            </div>
   359→            <div className="flex gap-1.5">
   360→              <div className="doodle-box doodle-dark px-2 py-0.5 font-hand-sc text-[10px] sm:text-xs" title="Playtime in this round">
   361→                RND {fmtTime(times.roundMs)}
   362→              </div>
   363→              <div className="doodle-box doodle-dark px-2 py-0.5 font-hand-sc text-[10px] sm:text-xs" title="Playtime across all rounds">
   364→                SUM {fmtTime(times.totalMs)}
   365→              </div>
   366→            </div>
   367→            <div className="flex gap-1.5 sm:gap-2">
   368→              <button
   369→                className="doodle-btn px-2 py-0.5 font-hand-sc text-xs sm:px-2.5 sm:py-1 sm:text-sm"
   370→                onClick={() => setShowHelp(true)}
   371→                aria-label="How to play"
   372→              >
   373→                ? RULES
   374→              </button>
   375→              <button className="doodle-btn px-2 py-0.5 font-hand-sc text-xs sm:px-2.5 sm:py-1 sm:text-sm" onClick={toggleMute} aria-label="Sound">
   376→                {muted ? "SOUND OFF" : "SOUND ON"}
   377→              </button>
   378→            </div>
   379→          </div>
   380→
   381→          {/* top right: THE DEALER */}
   382→          <div className="absolute right-3 top-3 z-30 flex flex-col items-end gap-1">
   383→            <div className="doodle-box px-3 py-1 rotate-[1.5deg]">
   384→              <span className="font-marker text-lg text-[#8f2418]">THE DEALER</span>
   385→              {snap.voodooEnemy > 0 && (
   386→
   387→                <img src={assets.voodoo} alt="voodoo" className="ml-2 inline h-6 w-6 wiggle align-middle" />
   388→              )}
   389→              {snap.enemyCuffed && (
   390→                <span className="ml-2 font-hand-sc text-sm text-[#9b8ec4] underline">cuffed</span>
   391→              )}
   392→            </div>
   393→            {snap.enemySkills.length > 0 && (
   394→              <div className="doodle-box flex items-center gap-1 px-2 py-0.5 rotate-[1deg]">
   395→                <span className="font-hand-sc text-[10px] text-[#4a4036]">SKILLS</span>
   396→                {snap.enemySkills.map((s) => (
   397→                  <img
   398→                    key={s.id}
   399→                    src={s.icon}
   400→                    alt={s.id}
   401→                    title={SKILL_META[s.id].name}
   402→                    className="h-6 w-6"
   403→                  />
   404→                ))}
   405→              </div>
   406→            )}
   407→            <div className="flex gap-1">
   408→              {Array.from({ length: snap.maxHpEnemy }).map((_, i) => (
   409→
   410→                <img key={i} src={i < snap.hpEnemy ? assets.heartFull : assets.heartEmpty} alt="hp" className="h-8 w-8" />
   411→              ))}
   412→            </div>
   413→          </div>
   414→
   415→          {/* player: top-left on mobile, bottom-left on desktop */}
   416→          <div className="absolute left-2 top-[88px] z-30 flex flex-col gap-1 sm:left-3 sm:top-auto sm:bottom-3">
   417→            <div className="doodle-box w-fit px-2 py-0.5 rotate-[-1.5deg] sm:px-3 sm:py-1">
   418→              <span className="font-marker text-sm sm:text-lg">YOU</span>
   419→              {snap.voodooPlayer > 0 && (
   420→                <span className="ml-2 font-hand-sc text-sm text-[#5e9c5b]">voodoo {snap.voodooPlayer}</span>
   421→              )}
   422→              {snap.playerCuffed && (
   423→                <span className="ml-2 font-hand-sc text-sm text-[#9b8ec4] underline">cuffed</span>
   424→              )}
   425→            </div>
   426→            <div className="flex gap-0.5 sm:gap-1">
   427→              {Array.from({ length: snap.maxHpPlayer }).map((_, i) => (
   428→                <img key={i} src={i < snap.hpPlayer ? assets.heartFull : assets.heartEmpty} alt="hp" className="h-7 w-7 sm:h-9 sm:w-9" />
   429→              ))}
   430→            </div>
   431→          </div>
   432→
   433→          {/* turn indicator */}
   434→          <div className="pointer-events-none absolute left-1/2 top-[74px] z-20 -translate-x-1/2 sm:top-16">
   435→            {snap.turn === "player" && (
   436→              <div className="doodle-box px-4 py-0.5 font-marker text-[#8f2418] rotate-[-2deg] float-y">
   437→                YOUR TURN
   438→              </div>
   439→            )}
   440→            {snap.turn === "enemy" && (
   441→              <div className="doodle-box doodle-dark px-4 py-0.5 font-marker rotate-[2deg] float-y">
   442→                THE DEALER&apos;S TURN...
   443→              </div>
   444→            )}
   445→          </div>
   446→
   447→          {/* must shoot */}
   448→          {snap.mustShoot && snap.turn === "player" && (
   449→            <div className="pointer-events-none absolute left-1/2 top-28 z-30 -translate-x-1/2">
   450→              <div className="doodle-box doodle-red px-4 py-1 font-marker text-xl blink-soft rotate-[-3deg]">
   451→                YOU MUST SHOOT!
   452→              </div>
   453→            </div>
   454→          )}
   455→
   456→          {/* reload pips */}
   457→          {reloadPips > 0 && (
   458→            <div className="pointer-events-none absolute left-1/2 top-[62%] z-30 flex -translate-x-1/2 gap-1.5">
   459→              {Array.from({ length: 6 }).map((_, i) => (
   460→                <div
   461→                  key={i}
   462→                  className={`h-4 w-4 rounded-full border-[3px] border-[#f3ecd7] ${
   463→                    i < reloadPips ? "bg-[#c73e2e]" : "bg-transparent"
   464→                  }`}
   465→                />
   466→              ))}
   467→            </div>
   468→          )}
   469→
   470→          {/* skill bar (dice skills + shop trials; spent trials fade away) */}

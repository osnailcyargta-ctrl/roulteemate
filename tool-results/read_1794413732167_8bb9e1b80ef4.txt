   109→   109→function shuffled<T>(arr: T[]): T[] {
   110→   110→  const a = [...arr];
   111→   111→  for (let i = a.length - 1; i > 0; i--) {
   112→   112→    const j = Math.floor(Math.random() * (i + 1));
   113→   113→    [a[i], a[j]] = [a[j], a[i]];
   114→   114→  }
   115→   115→  return a;
   116→   116→}

   291→   291→    this.logic.setPhase("dice");
   292→   292→    this.notify();

   299→   299→
   300→   300→    this.scene.diceVisible(true);

   313→   313→    // the enemy quietly takes its skills
   314→   314→    const n = enemySkillCount(diff);
   315→   315→    for (const id of shuffled(SKILL_IDS).slice(0, n)) this.logic.grantSkill("enemy", id);
   316→   316→
   317→   317→    this.scene.enemySetMood("laugh");
   318→   318→    gameAudio.laugh();
   319→   319→    this.scene.enemySay([pick("gameStart")], 2.4);
   320→   320→    await sleep(1200);

   323→   323→  /** One dice tumble animation in front of the camera. */
   324→   324→  private diceTumble(finalId: SkillId): Promise<void> {
   325→   325→    const { main, ghosts } = this.scene.diceMeshes();
   326→   326→    let lastSwitch = 0;
   327→   327→    let cycle = Math.floor(Math.random() * SKILL_IDS.length);
   328→   328→    const setFace = (id: SkillId) => this.scene.diceSetFace(id, texDieFace(id));
   329→   329→    const dur = 1400;
   330→   330→    const start = performance.now();
   331→   331→    gameAudio.diceRoll(); // physical tumbling knocks while the die spins
   332→   332→    return new Promise<void>((resolve) => {
   333→   333→      const step = () => {
   334→   334→        const t = Math.min(1, (performance.now() - start) / dur);
   335→   335→        main.rotation.z = Math.pow(1 - t, 1.4) * 10;
   336→   336→        const bounce = Math.abs(Math.sin(t * Math.PI * 3.4)) * (1 - t) * 0.3;
   337→   337→        main.position.y = -0.02 + bounce;
   338→   338→        main.scale.setScalar(0.7 + t * 0.3);
   339→   339→        if (t > 0.7) {
   340→   340→          if (main.userData.face !== finalId) {
   341→   341→            setFace(finalId);
   342→   342→            main.userData.face = finalId;
   343→   343→          }
   344→   344→        } else if (performance.now() - lastSwitch > 40 + t * t * 240) {
   345→   345→          lastSwitch = performance.now();
   346→   346→          cycle = (cycle + 1) % SKILL_IDS.length;
   347→   347→          setFace(SKILL_IDS[cycle]);
   348→   348→          main.userData.face = SKILL_IDS[cycle];
   349→   349→        }
   350→   350→        ghosts.forEach((g, i) => {
   351→   351→          g.rotation.z = main.rotation.z - 0.05 * (i + 1);
   352→   352→          g.position.y = main.position.y + 0.012 * (i + 1);
   353→   353→          g.material.opacity = (0.13 / (i + 1)) * (1 - t * 0.4);
   354→   354→          g.material.map = main.material.map;
   355→   355→        });
   356→   356→        if (t < 1) {
   357→   357→          requestAnimationFrame(step);
   358→   358→        } else {
   359→   359→          gameAudio.diceLand();
   360→   360→          this.scene.flash(0.3);
   361→   361→          // squash & settle
   362→   362→          const s0 = performance.now();
   363→   363→          const squash = () => {
   364→   364→            const k = Math.min(1, (performance.now() - s0) / 220);
   365→   365→            main.scale.set(1 + Math.sin(k * Math.PI) * 0.12, 1 - Math.sin(k * Math.PI) * 0.18, 1);
   366→   366→            main.rotation.z = (1 - k) * 0.08;
   367→   367→            if (k < 1) requestAnimationFrame(squash);
   368→   368→            else resolve();
   369→   369→          };
   370→   370→          squash();
   371→   371→        }
   372→   372→      };
   373→   373→      step();
   374→   374→    });
   375→   375→  }

   379→   379→    this.logic.state.round += 1;

   381→   381→    this.tRoundStart = performance.now();
   382→   382→    this.logic.setPhase("reloading");
   383→   383→    this.notify();
   384→   384→    this.busy = true;
   385→   385→
   386→   386→    this.scene.setLampBoost(0.4);
   387→   387→    await sleep(380);
   388→   388→    this.scene.gunAim("enemy", true);
   389→   389→    this.scene.gunOpenCylinder(true);
   390→   390→    gameAudio.reloadOpen();
   391→   391→    await sleep(620);
   392→   392→
   393→   393→    const total = 6;
   394→   394→    for (let i = 0; i < total; i++) {
   395→   395→      gameAudio.reloadLoad(i);
   396→   396→      this.onEvent({ type: "reloadStep", loaded: i + 1, total });
   397→   397→      await sleep(190);
   398→   398→    }
   399→   399→    gameAudio.spinCylinder();
   400→   400→    await sleep(560);
   401→   401→    this.scene.gunOpenCylinder(false);
   402→   402→    gameAudio.reloadClose();
   403→   403→    this.scene.setLampBoost(1);
   404→   404→    this.scene.flash(0.28);
   405→   405→    await sleep(420);
   406→   406→    this.scene.gunAim(null);
   407→   407→
   408→   408→    const live = this.logic.liveRemaining;
   409→   409→    const blank = this.logic.blankRemaining;
   410→   410→    this.onEvent({
   411→   411→      type: "banner",
   412→   412→      title: `ROUND ${this.logic.state.round}`,
   413→   413→      sub: `LIVE ${live} — BLANK ${blank}`,
   414→   414→    });
   415→   415→    gameAudio.hammerCock();
   416→   416→    await sleep(1400);
   417→   417→    this.busy = false;
   418→   418→  }

   428→   428→      this.toast("THE BOARD FADED — memorize your shells!", "bad");

   435→   435→  async playerShoot(target: Side) {
   436→   436→    const st = this.logic.state;
   437→   437→    if (this.busy || st.phase !== "playerTurn" || st.turn !== "player") return;
   438→   438→    if (st.cylinder.length === 0) return;
   439→   439→    this.busy = true;
   440→   440→    this.notify();
   441→   441→
   442→   442→    // LOW-HP TENSION: below 3 HP a self-shot gets 0.7s of pure dread first.
   443→   443→    const tension = target === "player" && st.hp.player < 3;
   444→   444→
   445→   445→    this.scene.gunAim(target === "player" ? "self" : "enemy");
   446→   446→    gameAudio.hammerCock();
   447→   447→    gameAudio.riser(tension ? 1.5 : 0.55);
   448→   448→    await sleep(800);
   449→   449→    if (tension) {
   450→   450→      // 0.7s: the trigger finger trembles... 240 + 240 + 220

   512→   512→    let skillsUsed = 0;
   513→   513→    let guard = 0;
   514→   514→    let decision: AIDecision | null = null;
   515→   515→    // AI loop: burn skills first, then shoot
   516→   516→    for (;;) {
   517→   517→      if (st.hp.enemy <= 0 || st.hp.player <= 0) return;

   540→   540→        await this.endGame();
   541→   541→        return;
   542→   542→      }
   543→   543→    }
   544→   544→
   545→   545→    if (this.logic.isEmpty) {
   546→   546→      await this.maybeShopAndReload();
   547→   547→    }
   548→   548→    const next = this.passTurn();
   549→   549→    this.notify();
   550→   550→    if (next === "player") {
   551→   551→      await this.playerTurnStart();
   552→   552→    } else {
   553→   553→      // the player is cuffed
   554→   554→      this.toast("You're cuffed! The DEALER goes again.", "bad");

   556→   556→      await this.enemyTurnStart();

   562→   562→  /** Fire one shot and resolve EVERYTHING that follows it:
   563→   563→   *  the voodoo curse (fresh shell + forced shot) and the global
   564→   564→   *  self-blank rule (one automatic free shot at the opponent). */
   565→   565→  private async shootSequence(shooter: Side, target: Side, tensionSelf = false, forced = false): Promise<"pass" | "dead"> {
   566→   566→    const out = this.logic.shoot(shooter, target, forced);
   567→   567→    const res = await this.resolveShot(out, tensionSelf);
   568→   568→    if (res === "dead") return "dead";
   569→   569→
   570→   570→    const opp: Side = shooter === "player" ? "enemy" : "player";
   571→   571→
   572→   572→    if (out.voodooReroll) {
   573→   573→      // VOODOO CURSE: blank was rerolled — the shooter MUST fire once more,
   574→   574→      // and the gun swings toward the opponent.
   575→   575→      this.notify();
   576→   576→      await sleep(300);
   577→   577→      const out2 = this.logic.shoot(shooter, opp, true);
   578→   578→      const res2 = await this.resolveShot(out2, false);
   579→   579→      if (res2 === "dead") return "dead";
   580→   580→    } else if (out.bonusShot && !this.logic.isEmpty) {
   581→   581→      // GLOBAL RULE: shot yourself with a blank -> ONE automatic shot at the enemy.
   582→   582→      this.notify();
   583→   583→      if (shooter === "player") {
   584→   584→        this.toast("BLANK! Free shot at the DEALER!", "good");
   585→   585→        this.scene.gunAim("enemy");
   586→   586→        gameAudio.hammerCock();
   587→   587→        await sleep(750);
   588→   588→      } else {
   589→   589→        this.scene.enemySetMood("aim");
   590→   590→        this.toast("DEALER blanks himself — free shot at you!", "bad");
   591→   591→        await sleep(750);
   592→   592→      }
   593→   593→      const out2 = this.logic.shoot(shooter, opp);
   594→   594→      const res2 = await this.resolveShot(out2, false);
   595→   595→      if (res2 === "dead") return "dead";
   596→   596→    }
   597→   597→    return "pass";
   598→   598→  }

   884→   884→    else this.ghosts.push({ id, turns: 2 });

   886→   886→  }

   888→   888→  /** Every passed turn dissolves the ghosts a little more. */
   889→   889→  private tickGhosts() {
   890→   890→    if (!this.ghosts.length) return;
   891→   891→    for (const g of this.ghosts) g.turns -= 1;
   892→   892→    const done = this.ghosts.filter((g) => g.turns <= 0).map((g) => g.id);
   893→   893→    for (const id of done) {
   894→   894→      const sk = this.logic.state.skills[id];
   895→   895→      sk.owned = false;
   896→   896→      sk.trial = false;
   897→   897→      this.toast(`${SKILL_META[id].name} faded away.`, "info");
   898→   898→    }
   899→   899→    this.ghosts = this.ghosts.filter((g) => g.turns > 0);
   900→   900→  }
   901→   901→
   902→   902→  /** endTurnAndPass + ghost fade bookkeeping. */
   903→   903→  private passTurn(): Side {
   904→   904→    const next = this.logic.endTurnAndPass();
   905→   905→    this.tickGhosts();
   906→   906→    return next;
   907→   907→  }

   911→   911→  /** What the shop stocks this visit. Legends only on INSANE. */
   912→   912→  shopCatalog(): ShopOffer[] {
   913→   913→    const st = this.logic.state;
   914→   914→    const activeCount = ALL_SKILL_IDS.filter((id) => {
   915→   915→      const sk = st.skills[id];
   916→   916→      return sk.owned && !(sk.trial && sk.charges <= 0);
   917→   917→    }).length;
   918→   918→    return ALL_SKILL_IDS.filter((id) => {
   919→   919→      if (!(id in SHOP_PRICES)) return false;
   920→   920→      if (LEGENDARY.includes(id) && st.difficulty !== "insane") return false;
   921→   921→      return true;
   922→   922→    }).map((id) => {
   923→   923→      const sk = st.skills[id];
   924→   924→      const price = SHOP_PRICES[id]!;
   925→   925→      let disabled = false;
   926→   926→      let reason: string | null = null;
   927→   927→      if (sk.owned) {
   928→   928→        disabled = true;
   929→   929→        reason = sk.trial && sk.charges <= 0 ? "FADING AWAY" : "ALREADY HELD";
   930→   930→      } else if (activeCount >= MAX_SKILLS) {
   931→   931→        disabled = true;
   932→   932→        reason = `SLOTS FULL (${MAX_SKILLS})`;
   933→   933→      } else if (this.money < price) {
   934→   934→        disabled = true;
   935→   935→        reason = "TOO POOR";
   936→   936→      }
   937→   937→      return { id, price, disabled, reason, legendary: LEGENDARY.includes(id) };
   938→   938→    });
   939→   939→  }

   950→   950→  /** Open the shop between rounds and wait for the player to leave. */
   951→   951→  private async openShop() {
   952→   952→    const st = this.logic.state;
   953→   953→    this.money += 75; // survival bonus
   954→   954→    this.toast("ROUND SURVIVED: +$75", "good");

   956→   956→    this.busy = true;
   957→   957→    gameAudio.shopOpen();

   961→   961→      this.shopResolve = res;
   962→   962→    });
   963→   963→    this.busy = false;
   964→   964→  }

   966→   966→  /** Between rounds: shop first (if the cylinder ran dry), then reload. */
   967→   967→  private async maybeShopAndReload() {

   969→   969→      await this.openShop();

   974→   974→  buyTrial(id: SkillId) {
   975→   975→    if (this.logic.state.phase !== "shop") return;
   976→   976→    const offer = this.shopCatalog().find((o) => o.id === id);
   977→   977→    if (!offer || offer.disabled) return;
   978→   978→    this.money -= offer.price;
   979→   979→    this.logic.grantTrial("player", id);
   980→   980→    gameAudio.buy();
   981→   981→    this.onEvent({ type: "skillGet", id });
   982→   982→    this.toast(`TRIAL ${SKILL_META[id].name} bought — one use, then it fades!`, "good");
   983→   983→    this.notify();
   984→   984→  }

   986→   986→  closeShop() {
   987→   987→    if (this.logic.state.phase !== "shop") return;
   988→   988→    gameAudio.click();
   989→   989→    const r = this.shopResolve;
   990→   990→    this.shopResolve = null;
   991→   991→    r?.();
   992→   992→  }

   132→   132→function shuffled<T>(arr: T[]): T[] {
   133→   133→  const a = [...arr];
   134→   134→  for (let i = a.length - 1; i > 0; i--) {
   135→   135→    const j = Math.floor(Math.random() * (i + 1));
   136→   136→    [a[i], a[j]] = [a[j], a[i]];
   137→   137→  }
   138→   138→  return a;
   139→   139→}

   291→   291→  /** The SHOP pool for this table. EPIC cards live here ONLY — and only on
   292→   292→   *  NORMAL and above. LEGEND needs INSANE / ENDLESS (+ the allow-legend
   293→   293→   *  switch). The mutator card pool gates everything further. */
   294→   294→  private poolFor(diff: Difficulty, mode: Mode): SkillId[] {
   295→   295→    const mut = this.mut;
   296→   296→    let pool = ALL_SKILL_IDS;
   297→   297→    if (mut) pool = pool.filter((id) => mut.pool.includes(id));
   298→   298→    const epicOk = diff !== "easy"; // EPIC: shop-only, NORMAL+
   299→   299→    const legendOk = (diff === "insane" || mode === "endless") && (mut ? mut.allowLegend : true);
   300→   300→    return pool.filter((id) => {
   301→   301→      const r = RARITY[id];
   302→   302→      if (r === "epic") return epicOk;
   303→   303→      if (r === "legend") return legendOk;
   304→   304→      return true;
   305→   305→    });
   306→   306→  }

   317→   317→  /** Draw n skills from the pool. The DEALER keeps the stacking record;
   318→   318→   *  the player's dice grants go through diceIntro (which manages slots). */
   319→   319→  private drawSkills(side: Side, pool: SkillId[], n: number) {
   320→   320→    const allowDup = this.mut?.allowDup ?? false;
   321→   321→    const st = this.logic.state;
   322→   322→    const bag = allowDup
   323→   323→      ? pool
   324→   324→      : pool.filter((id) => !st.enemySkills[id].owned || side === "player");
   325→   325→    const src = bag.length ? bag : pool;
   326→   326→    for (let k = 0; k < n; k++) {
   327→   327→      const id = src[Math.floor(Math.random() * src.length)];
   328→   328→      this.logic.grantSkill(side, id);
   329→   329→    }
   330→   330→  }

   377→   377→  /** Compile the mutator's ENGINE C program to WebAssembly and wire it up.
   378→   378→   *  The script runs NATIVELY in the browser's WASM engine — not in JS. */
   379→   379→  private async setupScripts() {
   380→   380→    const src = this.mut?.rules ?? "";
   381→   381→    this.script = null;
   382→   382→    if (!src.trim()) return;
   383→   383→    const compiled = compileScript(src);
   384→   384→    for (const e of compiled.errors.slice(0, 4)) this.toast(`[ENGINE C] ${e}`, "bad");
   385→   385→    if (!compiled.ok || !compiled.bytes) return;
   386→   386→    const host: ScriptHost = {
   387→   387→      runSkill: (id, target) => void this.forceSkillFx(id, target),
   388→   388→      heal: (side, n) => {
   389→   389→        const did = this.logic.healSide(side, n);
   390→   390→        if (did) this.onEvent({ type: "dmgPop", side, text: `+${n} HP`, tone: "reflect" });
   391→   391→      },
   392→   392→      damage: (side, n) => {
   393→   393→        const dead = this.logic.damageSide(side, n);
   394→   394→        if (side === "enemy") {
   395→   395→          this.scene.bloodBurst("enemy");
   396→   396→          if (this.logic.state.mode === "endless") this.onEvent({ type: "dmgPop", side, text: `SCORE +${n}`, tone: "hit" });
   397→   397→          else this.onEvent({ type: "dmgPop", side, text: `-${n} HP`, tone: "hit" });
   398→   398→        } else {
   399→   399→          this.onEvent({ type: "screenSplat" });
   400→   400→          this.scene.addTrauma(0.4);
   401→   401→          if (!dead) this.onEvent({ type: "dmgPop", side, text: `-${n} HP`, tone: "hit" });
   402→   402→        }
   403→   403→      },
   404→   404→      say: (text) => this.toast(text, "info"),
   405→   405→      stateRef: () => ({
   406→   406→        hp: this.logic.state.hp,
   407→   407→        score: this.logic.state.score,
   408→   408→        round: this.logic.state.round,
   409→   409→      }),
   410→   410→    };
   411→   411→    try {
   412→   412→      this.script = await ScriptRuntime.instantiate(compiled.bytes, host, (msg) => this.toast(msg, "bad"));
   413→   413→    } catch (e) {
   414→   414→      this.toast(`[ENGINE C] ${(e as Error).message}`, "bad");
   415→   415→      return;
   416→   416→    }
   417→   417→    this.script.fire("setup");
   418→   418→    if (this.logic.state.hp.player <= 0) {
   419→   419→      // a setup rule killed the player before the first round — mercy: undo it
   420→   420→      this.logic.state.hp.player = 1;
   421→   421→      this.toast("[custom rules] the setup killed you already... you start at 1 HP instead.", "bad");
   422→   422→    }
   423→   423→  }

   425→   425→  /** Fire a custom-rule event (best-effort; scripts may never crash the game). */
   426→   426→  private fireScript(event: ScriptEventName, ctx: { skill?: SkillId } = {}) {
   427→   427→    try {
   428→   428→      this.script?.fire(event, ctx);

   431→   431→    }

   434→   434→  /** A script FORCED a skill's effect — apply it with a matching mini-FX. */
   435→   435→  private async forceSkillFx(id: SkillId | "all", target: Side) {

   585→   585→  /** One dice tumble animation in front of the camera. */
   586→   586→  private diceTumble(finalId: SkillId): Promise<void> {
   587→   587→    const { main, ghosts } = this.scene.diceMeshes();
   588→   588→    let lastSwitch = 0;
   589→   589→    let cycle = Math.floor(Math.random() * SKILL_IDS.length);
   590→   590→    const setFace = (id: SkillId) => this.scene.diceSetFace(id, texDieFace(id));
   591→   591→    const dur = 1400;
   592→   592→    const start = performance.now();
   593→   593→    gameAudio.diceRoll(); // physical tumbling knocks while the die spins
   594→   594→    return new Promise<void>((resolve) => {
   595→   595→      const step = () => {
   596→   596→        const t = Math.min(1, (performance.now() - start) / dur);
   597→   597→        main.rotation.z = Math.pow(1 - t, 1.4) * 10;
   598→   598→        const bounce = Math.abs(Math.sin(t * Math.PI * 3.4)) * (1 - t) * 0.3;
   599→   599→        main.position.y = -0.02 + bounce;
   600→   600→        main.scale.setScalar(0.7 + t * 0.3);
   601→   601→        if (t > 0.7) {
   602→   602→          if (main.userData.face !== finalId) {
   603→   603→            setFace(finalId);
   604→   604→            main.userData.face = finalId;
   605→   605→          }
   606→   606→        } else if (performance.now() - lastSwitch > 40 + t * t * 240) {
   607→   607→          lastSwitch = performance.now();
   608→   608→          cycle = (cycle + 1) % SKILL_IDS.length;
   609→   609→          setFace(SKILL_IDS[cycle]);
   610→   610→          main.userData.face = SKILL_IDS[cycle];
   611→   611→        }
   612→   612→        ghosts.forEach((g, i) => {
   613→   613→          g.rotation.z = main.rotation.z - 0.05 * (i + 1);
   614→   614→          g.position.y = main.position.y + 0.012 * (i + 1);
   615→   615→          g.material.opacity = (0.13 / (i + 1)) * (1 - t * 0.4);
   616→   616→          g.material.map = main.material.map;
   617→   617→        });
   618→   618→        if (t < 1) {
   619→   619→          requestAnimationFrame(step);
   620→   620→        } else {
   621→   621→          gameAudio.diceLand();
   622→   622→          this.scene.flash(0.3);
   623→   623→          // squash & settle
   624→   624→          const s0 = performance.now();
   625→   625→          const squash = () => {
   626→   626→            const k = Math.min(1, (performance.now() - s0) / 220);
   627→   627→            main.scale.set(1 + Math.sin(k * Math.PI) * 0.12, 1 - Math.sin(k * Math.PI) * 0.18, 1);
   628→   628→            main.rotation.z = (1 - k) * 0.08;
   629→   629→            if (k < 1) requestAnimationFrame(squash);
   630→   630→            else resolve();
   631→   631→          };
   632→   632→          squash();
   633→   633→        }
   634→   634→      };
   635→   635→      step();
   636→   636→    });
   637→   637→  }

   736→   736→  async playerUseSkill(uid: number) {

   739→   739→    const slot = this.logic.playerSlot(uid);

   744→   744→
   745→   745→    let oc: SkillOutcome;
   746→   746→    if (slot.id === "gamble") {

   790→   790→      await this.endGame();
   791→   791→      return;
   792→   792→    }
   793→   793→
   794→   794→    // EJECT on the LAST shell empties the cylinder — nothing left to fire,
   795→   795→    // so the round ends RIGHT HERE: shop break, reload, pass the turn.

   797→   797→      this.toast(t("chamberEmpty"), "info");

   805→   805→        await this.playerTurnStart();
   806→   806→      } else {
   807→   807→        await this.enemyTurnStart();
   808→   808→      }
   809→   809→      return;

   814→   814→    this.notify();
   815→   815→  }

   830→   830→      this.toast(t("boardFaded"), "bad");
   831→   831→      gameAudio.ghostFade();
   832→   832→    }
   833→   833→    if (this.logic.lastSlip) {
   834→   834→      // the Dealer lost count — and says so out loud

   842→   842→    await sleep(rnd(1000, 1700));
   843→   843→
   844→   844→    let skillsUsed = 0;
   845→   845→    let guard = 0;
   846→   846→    let decision: AIDecision | null = null;
   847→   847→    // AI loop: burn skills first, then shoot
   848→   848→    for (;;) {
   849→   849→      if (st.hp.enemy <= 0 || st.hp.player <= 0) return;
   850→   850→      if (this.logic.isEmpty || guard++ > 24) break;
   851→   851→      const d = this.ai.decide();
   852→   852→      if (d.kind === "skill" && d.skill && skillsUsed < 3) {
   853→   853→        const oc = this.logic.useSkill("enemy", d.skill);
   854→   854→        if (!oc.ok) continue;

   857→   857→        this.fireScript("skill", { skill: d.skill });
   858→   858→        this.notify();
   859→   859→        await sleep(rnd(500, 850));
   860→   860→        continue;
   861→   861→      }
   862→   862→      decision = d;
   863→   863→      break;
   864→   864→    }
   865→   865→
   866→   866→    if (decision && decision.kind === "shoot" && decision.target && !this.logic.isEmpty) {
   867→   867→      this.scene.enemySetMood("aim");

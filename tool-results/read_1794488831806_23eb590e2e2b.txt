     1→"use client";
     2→
     3→/**
     4→ * GameShell — React layer for ROULETTE-MATE.
     5→ * Mounts the engine (Three.js) plus all doodle overlays: the animated
     6→ * main menu (title top-left, 3 buttons bottom-left), settings, HP,
     7→ * shell counter, skill bar, toasts, banner, rules, win/lose screens,
     8→ * and the low-HP blood vignette.
     9→ */
    10→
    11→import { useCallback, useEffect, useMemo, useRef, useState } from "react";
    12→import { GameEngine, type HudEvent, type HudSnapshot } from "@/game/engine";
    13→import {
    14→  SKILL_META,
    15→  dataURL,
    16→  texBulletPip,
    17→  texDiffIcon,
    18→  texHeart,
    19→  texSkillIcon,
    20→  texSplat,
    21→  texVoodooMask,
    22→  type SkillId,
    23→} from "@/game/doodle";
    24→import type { ShellType } from "@/game/logic";
    25→import { gameAudio } from "@/game/audio";
    26→
    27→type Toast = { id: number; text: string; tone: string; leaving?: boolean };
    28→type Pop = { id: number; side: string; text: string; tone: string };
    29→type Splat = { id: number; url: string; left: number; top: number; size: number; rot: number };
    30→type Stamp = { id: number; idSkill: SkillId };
    31→type MenuPanel = "play" | "settings" | null;
    32→
    33→type Quality = "low" | "medium" | "high";
    34→
    35→/** All toggles the SETTINGS screen owns (persisted in localStorage). */
    36→type Settings = {
    37→  muted: boolean;
    38→  quality: Quality;
    39→  /** the live/blank counter board at the top */
    40→  showShellStat: boolean;
    41→  /** TIME expands into these two */
    42→  showRoundTime: boolean;
    43→  showTotalTime: boolean;
    44→  /** frame-per-second meter right above the YOU indicator */
    45→  showFps: boolean;
    46→};
    47→
    48→const SETTINGS_KEY = "roulette-mate-settings";
    49→const DEFAULT_SETTINGS: Settings = {
    50→  muted: false,
    51→  quality: "high",
    52→  showShellStat: true,
    53→  showRoundTime: true,
    54→  showTotalTime: true,
    55→  showFps: false,
    56→};
    57→
    58→function loadSettings(): Settings {
    59→  try {
    60→    const raw = localStorage.getItem(SETTINGS_KEY);
    61→    if (raw) return { ...DEFAULT_SETTINGS, ...(JSON.parse(raw) as Partial<Settings>) };
    62→  } catch {
    63→    /* corrupted storage — fall back to defaults */
    64→  }
    65→  return DEFAULT_SETTINGS;
    66→}
    67→
    68→const DIFFS: { id: "easy" | "normal" | "insane"; label: string; desc: string }[] = [
    69→  { id: "easy", label: "EASY", desc: "Your 6 HP vs his 5. The Dealer has 1 skill." },
    70→  { id: "normal", label: "NORMAL", desc: "5 vs 5 HP. The Dealer has 2 skills." },
    71→  { id: "insane", label: "INSANE", desc: "Your 4 HP vs his 5. 3 skills + shop legends." },
    72→];
    73→
    74→const fmtTime = (ms: number) => {
    75→  const s = Math.max(0, Math.floor(ms / 1000));
    76→  return `${String(Math.floor(s / 60)).padStart(2, "0")}:${String(s % 60).padStart(2, "0")}`;
    77→};
    78→
    79→export default function GameShell() {
    80→  const mountRef = useRef<HTMLDivElement>(null);
    81→  const engineRef = useRef<GameEngine | null>(null);
    82→  const evId = useRef(0);
    83→  const [ready, setReady] = useState(false);
    84→  const [snap, setSnap] = useState<HudSnapshot | null>(null);
    85→  const [toasts, setToasts] = useState<Toast[]>([]);
    86→  const [pops, setPops] = useState<Pop[]>([]);
    87→  const [splats, setSplats] = useState<Splat[]>([]);
    88→  const [stamps, setStamps] = useState<Stamp[]>([]);
    89→  const [banner, setBanner] = useState<{ title: string; sub?: string } | null>(null);
    90→  const [reveal, setReveal] = useState<{ shell: "live" | "blank" } | null>(null);
    91→  const [showHelp, setShowHelp] = useState(false);
    92→  const [diff, setDiff] = useState<"easy" | "normal" | "insane">("normal");
    93→  const [reloadPips, setReloadPips] = useState(0);
    94→  const [flashRed, setFlashRed] = useState(false);
    95→  const [revealQueue, setRevealQueue] = useState<(ShellType | null)[] | null>(null);
    96→  const [times, setTimes] = useState<{ roundMs: number; totalMs: number }>({ roundMs: 0, totalMs: 0 });
    97→  const [shopTip, setShopTip] = useState<{ id: SkillId | "bell"; xPct: number; yPct: number } | null>(null);
    98→
    99→  /* ---------- MAIN MENU screens ---------- */
   100→  const [menuPanel, setMenuPanel] = useState<MenuPanel>(null);
   101→  /* ---------- SETTINGS (persisted) ---------- */
   102→  const [settings, setSettings] = useState<Settings>(DEFAULT_SETTINGS);
   103→  const [timeOpen, setTimeOpen] = useState(false);
   104→  /* ---------- FPS meter ---------- */
   105→  const [fps, setFps] = useState(0);
   106→
   107→  /* load persisted settings once */
   108→  useEffect(() => {
   109→    setSettings(loadSettings());
   110→  }, []);
   111→
   112→  const updateSettings = useCallback((patch: Partial<Settings>) => {
   113→    setSettings((prev) => {
   114→      const next = { ...prev, ...patch };
   115→      try {
   116→        localStorage.setItem(SETTINGS_KEY, JSON.stringify(next));
   117→      } catch {
   118→        /* private mode — settings just won't persist */
   119→      }
   120→      return next;
   121→    });
   122→  }, []);
   123→
   124→  /* apply sound + graphics settings */
   125→  useEffect(() => {
   126→    gameAudio.setMuted(settings.muted);
   127→  }, [settings.muted]);
   128→
   129→  useEffect(() => {
   130→    if (ready) engineRef.current?.setSceneQuality(settings.quality);
   131→  }, [settings.quality, ready]);
   132→
   133→  /* FPS meter (only measured while visible) */
   134→  useEffect(() => {
   135→    if (!settings.showFps) return;
   136→    let frames = 0;
   137→    let last = performance.now();
   138→    let raf = 0;
   139→    const tick = () => {
   140→      frames++;
   141→      const now = performance.now();
   142→      if (now - last >= 500) {
   143→        setFps(Math.round((frames * 1000) / (now - last)));
   144→        frames = 0;
   145→        last = now;
   146→      }
   147→      raf = requestAnimationFrame(tick);
   148→    };
   149→    raf = requestAnimationFrame(tick);
   150→    return () => cancelAnimationFrame(raf);
   151→  }, [settings.showFps]);
   152→
   153→  /* ---------- HUD asset dataURLs (after mount, client-only) ---------- */
   154→  const assets = useMemo(() => {
   155→    if (!ready) return null;
   156→    return {
   157→      heartFull: dataURL("heart-full", texHeart("full")),
   158→      heartEmpty: dataURL("heart-empty", texHeart("empty")),
   159→      pipLive: dataURL("pip-live", texBulletPip("live")),
   160→      pipBlank: dataURL("pip-blank", texBulletPip("blank")),
   161→      voodoo: dataURL("voodoo-mask", texVoodooMask()),
   162→      splat0: dataURL("splat-hud-0", texSplat(333)),
   163→      splat1: dataURL("splat-hud-1", texSplat(444)),
   164→      diffEasy: dataURL("diff-easy", texDiffIcon("easy")),
   165→      diffNormal: dataURL("diff-normal", texDiffIcon("normal")),
   166→      diffInsane: dataURL("diff-insane", texDiffIcon("insane")),
   167→    };
   168→  }, [ready]);
   169→
   170→  const assetsRef = useRef(assets);
   171→  useEffect(() => {
   172→    assetsRef.current = assets;
   173→  }, [assets]);
   174→
   175→  const handleEvent = useCallback((e: HudEvent) => {
   176→    const id = ++evId.current;
   177→    switch (e.type) {
   178→      case "toast": {
   179→        setToasts((t) => [...t.slice(-4), { id, text: e.text, tone: e.tone ?? "info" }]);
   180→        // graceful exit: slide-fade out before removal
   181→        setTimeout(() => setToasts((t) => t.map((x) => (x.id === id ? { ...x, leaving: true } : x))), 2700);
   182→        setTimeout(() => setToasts((t) => t.filter((x) => x.id !== id)), 3200);
   183→        break;
   184→      }
   185→      case "banner":
   186→        setBanner({ title: e.title, sub: e.sub });
   187→        setTimeout(() => setBanner(null), e.long ? 2600 : 1700);
   188→        break;
   189→      case "skillGet":
   190→        setStamps((s) => [...s, { id, idSkill: e.id }]);
   191→        setTimeout(() => setStamps((s) => s.filter((x) => x.id !== id)), 1500);
   192→        break;
   193→      case "shellReveal":
   194→        if (e.shell) {
   195→          setReveal({ shell: e.shell });
   196→          setTimeout(() => setReveal(null), 1900);
   197→        }
   198→        break;
   199→      case "shellRevealQueue":
   200→        setRevealQueue(e.shells);
   201→        setTimeout(() => setRevealQueue(null), 2800);
   202→        break;
   203→      case "dmgPop":
   204→        setPops((p) => [...p.slice(-5), { id, side: e.side, text: e.text, tone: e.tone ?? "hit" }]);
   205→        setTimeout(() => setPops((p) => p.filter((x) => x.id !== id)), 1500);
   206→        break;
   207→      case "screenSplat": {
   208→        const url = evId.current % 2 === 0 ? "splat-hud-0" : "splat-hud-1";
   209→        const a = assetsRef.current;
   210→        if (!a) break;
   211→        setSplats((s) => [
   212→          ...s.slice(-3),
   213→          {
   214→            id,
   215→            url: a[url],
   216→            left: 15 + Math.random() * 55,
   217→            top: 8 + Math.random() * 50,
   218→            size: 130 + Math.random() * 160,
   219→            rot: Math.random() * 360,
   220→          },
   221→        ]);
   222→        setFlashRed(true);
   223→        setTimeout(() => setFlashRed(false), 180);
   224→        setTimeout(() => setSplats((s) => s.filter((x) => x.id !== id)), 2600);
   225→        break;
   226→      }
   227→      case "skillFlash":
   228→        break;
   229→      case "reloadStep":
   230→        setReloadPips(e.loaded);
   231→        setTimeout(() => setReloadPips(0), 2400);
   232→        break;
   233→      case "shopTip":
   234→        setShopTip(e.id ? { id: e.id, xPct: e.xPct, yPct: e.yPct } : null);
   235→        break;
   236→    }
   237→  }, []);
   238→
   239→  /* ---------- mount engine ---------- */
   240→  useEffect(() => {
   241→    let disposed = false;
   242→    (async () => {
   243→      try {
   244→        await document.fonts.ready;
   245→      } catch {
   246→        /* noop */
   247→      }
   248→      if (disposed || !mountRef.current) return;
   249→      const engine = new GameEngine();
   250→      engineRef.current = engine;
   251→      (window as unknown as { __eng?: GameEngine }).__eng = engine;
   252→      engine.onHud = (s) => setSnap(s);
   253→      engine.onEvent = (e) => handleEvent(e);
   254→      engine.mount(mountRef.current);
   255→      setReady(true);
   256→    })();
   257→    return () => {
   258→      disposed = true;
   259→      engineRef.current?.dispose();
   260→      engineRef.current = null;
   261→    };
   262→  }, []);
   263→
   264→  const toggleMute = useCallback(() => {
   265→    updateSettings({ muted: !settings.muted });
   266→  }, [settings.muted, updateSettings]);
   267→
   268→  /* ---------- keyboard ---------- */
   269→  useEffect(() => {
   270→    const onKey = (ev: KeyboardEvent) => {
   271→      const eng = engineRef.current;
   272→      const k = ev.key.toLowerCase();
   273→      if (k === "escape") {
   274→        setMenuPanel(null);
   275→        setShowHelp(false);
   276→        return;
   277→      }
   278→      if (!eng || !snap) return;
   279→      if (k === "q") void eng.playerShoot("player");
   280→      else if (k === "e") void eng.playerShoot("enemy");
   281→      else if (k === "h") setShowHelp((v) => !v);
   282→      else if (k === "m") toggleMute();
   283→      else if (/^[1-4]$/.test(k)) {
   284→        const owned = snap.skills.filter((s) => s.owned && s.ghostTurns <= 0);
   285→        const target = owned[parseInt(k) - 1];
   286→        if (target) void eng.playerUseSkill(target.id);
   287→      }
   288→    };
   289→    window.addEventListener("keydown", onKey);
   290→    return () => window.removeEventListener("keydown", onKey);
   291→  }, [snap, toggleMute]);
   292→
   293→  /* ---------- playtime ticker (round + total) ---------- */
   294→  const phaseKey = snap?.phase ?? "menu";
   295→  useEffect(() => {
   296→    if (phaseKey === "menu") return;
   297→    const tick = () => setTimes(engineRef.current?.getTimes() ?? { roundMs: 0, totalMs: 0 });
   298→    tick();
   299→    const iv = setInterval(tick, 1000);
   300→    return () => clearInterval(iv);
   301→  }, [phaseKey]);
   302→
   303→  const isPlayerTurn = snap?.phase === "playerTurn" && snap.turn === "player" && !snap.busy;
   304→  const phase = snap?.phase ?? "menu";
   305→  const inGame = phase !== "menu";
   306→  const lowHp = inGame && snap ? snap.hpPlayer : 0;
   307→
   308→  /* ---------- render ---------- */
   309→  return (
   310→    <div className="fixed inset-0 overflow-hidden bg-[#0b0a08] select-none">
   311→      {/* 3D world */}
   312→      <div ref={mountRef} className="absolute inset-0" />
   313→
   314→      {/* cinematic overlays */}
   315→      <div className="pointer-events-none absolute inset-0 vignette-overlay z-10" />
   316→      <div className="pointer-events-none absolute inset-0 grain-overlay z-10" />
   317→      {flashRed && <div className="pointer-events-none absolute inset-0 z-10 bg-[rgba(199,62,46,0.35)]" />}
   318→
   319→      {/* LOW-HP BLOOD VIGNETTE: starts at 2 HP, thickest at 1 HP */}
   320→      {snap && lowHp === 2 && (
   321→        <div className="pointer-events-none absolute inset-0 z-10 blood-vignette" />
   322→      )}
   323→      {snap && lowHp === 1 && (
   324→        <>
   325→          <div className="pointer-events-none absolute inset-0 z-10 blood-vignette heavy" />
   326→          <div className="pointer-events-none absolute inset-0 z-10 overflow-hidden">
   327→            <span className="blood-drip drip-1" />
   328→            <span className="blood-drip drip-2" />
   329→            <span className="blood-drip drip-3" />
   330→            <span className="blood-drip drip-4" />
   331→          </div>
   332→          <div className="pointer-events-none absolute inset-0 z-10 pulse-danger" />
   333→        </>
   334→      )}
   335→
   336→      {snap && snap.voodooPlayer > 0 && phase !== "menu" && (
   337→        <div
   338→          className="pointer-events-none absolute inset-0 z-10"
   339→          style={{ boxShadow: "inset 0 0 120px rgba(94,156,91,0.45)" }}
   340→        />
   341→      )}
   342→
   343→      {/* blood on screen */}
   344→      <div className="pointer-events-none absolute inset-0 z-20 overflow-hidden">
   345→        {assets &&
   346→          splats.map((s) => (
   347→            <img
   348→              key={s.id}
   349→              src={s.url}
   350→              alt=""
   351→              className="absolute pop-in"
   352→              style={{
   353→                left: `${s.left}%`,
   354→                top: `${s.top}%`,
   355→                width: s.size,
   356→                transform: `rotate(${s.rot}deg)`,
   357→                transition: "opacity 1.2s ease 1s",
   358→                opacity: 0.92,
   359→              }}
   360→            />
   361→          ))}
   362→      </div>
   363→
   364→      {/* damage pops */}
   365→      <div className="pointer-events-none absolute inset-0 z-30">
   366→        {pops.map((p) => (
   367→          <div
   368→            key={p.id}
   369→            className={`absolute rise-fade font-marker text-3xl drop-shadow-[3px_3px_0_rgba(0,0,0,0.7)] ${
   370→              p.tone === "reflect" ? "text-[#8fdc8a]" : p.tone === "blank" ? "text-[#cfd8e3]" : "text-[#ff6b57]"
   371→            }`}
   372→            style={p.side === "enemy" ? { left: "58%", top: "28%" } : { left: "42%", top: "62%" }}
   373→          >
   374→            {p.text}
   375→          </div>
   376→        ))}
   377→      </div>
   378→
   379→      {/* ================= IN-GAME HUD ================= */}
   380→      {inGame && assets && snap && (
   381→        <>
   382→          {/* shell counter board — the honest numbers fade turn by turn,
   383→              their opacity really stepping down until it hits zero.
   384→              Can be switched off entirely in SETTINGS > UI & QOL. */}
   385→          {snap.round > 0 && settings.showShellStat && (
   386→          <div className="absolute left-1/2 top-3 z-30 -translate-x-1/2">
   387→            <div className="relative">
   388→              <div
   389→                className="doodle-box flex items-center gap-3 px-4 py-1.5 transition-opacity duration-[1100ms] ease-in-out"
   390→                style={{ opacity: snap.countsPublic ? Math.min(1, snap.countdown / 3) : 0 }}
   391→              >
   392→                <div className="flex items-center gap-1">
   393→                  <img src={assets.pipLive} alt="live" className="h-7 w-7" />
   394→                  <span className="font-marker text-xl text-[#8f2418]">x{snap.live}</span>
   395→                </div>
   396→                <div className="h-6 w-0.5 bg-[#221a12]/30" />
   397→                <div className="flex items-center gap-1">
   398→                  <img src={assets.pipBlank} alt="blank" className="h-7 w-7" />
   399→                  <span className="font-marker text-xl text-[#57697c]">x{snap.blank}</span>
   400→                </div>
   401→              </div>
   402→              {!snap.countsPublic && (
   403→                <div className="absolute inset-0 flex items-center justify-center">
   404→                  <div className="doodle-box doodle-dark board-dark-box pop-in-delayed flex items-center gap-2 whitespace-nowrap px-4 py-1.5">
   405→                    <span className="font-marker text-xl tracking-widest text-[#8b8577]">? ? ?</span>
   406→                    <span className="rotate-[-2deg] font-hand-sc text-xs text-[#d9a441]">memorize!</span>
   407→                  </div>
   408→                </div>
   409→              )}
   410→            </div>
   411→            {snap.known && (
   412→              <div className="mt-1 flex justify-center">
   413→                <div
   414→                  className={`pop-in font-hand-sc px-2 py-0.5 border-2 border-[#221a12] bg-[#f3ecd7] text-sm rotate-[-2deg] ${
   415→                    snap.known === "live" ? "text-[#8f2418]" : "text-[#57697c]"
   416→                  }`}
   417→                >
   418→                  CHAMBER: {snap.known.toUpperCase()}
   419→                </div>
   420→              </div>
   421→            )}
   422→          </div>
   423→          )}
   424→          {snap.round > 0 && !settings.showShellStat && snap.known && (
   425→            <div className="absolute left-1/2 top-3 z-30 -translate-x-1/2">
   426→              <div
   427→                className={`pop-in font-hand-sc px-2 py-0.5 border-2 border-[#221a12] bg-[#f3ecd7] text-sm rotate-[-2deg] ${
   428→                  snap.known === "live" ? "text-[#8f2418]" : "text-[#57697c]"
   429→                }`}
   430→              >
   431→                CHAMBER: {snap.known.toUpperCase()}
   432→              </div>
   433→            </div>
   434→          )}
   435→
   436→          {/* top left: round + cash + optional timers (toggles in SETTINGS) */}
   437→          <div className="absolute left-2 top-2 z-30 flex flex-col gap-1.5 sm:left-3 sm:top-3 sm:gap-2">
   438→            <div className="doodle-box doodle-dark px-2 py-0.5 font-marker text-xs rotate-[-1.5deg] sm:px-3 sm:py-1 sm:text-sm">
   439→              ROUND {Math.max(1, snap.round)}
   440→            </div>
   441→            <div
   442→              className="doodle-box w-fit px-2 py-0.5 font-marker text-xs text-[#8a6d1d] rotate-[1deg] sm:text-sm"
   443→              title="Cash — land bullets ($100), voodoo reflections ($150), survive rounds ($75). Between rounds 3 random items appear on the table."
   444→            >
   445→              $ {snap.money}
   446→            </div>
   447→            {(settings.showRoundTime || settings.showTotalTime) && (
   448→              <div className="flex gap-1.5">
   449→                {settings.showRoundTime && (
   450→                  <div className="doodle-box doodle-dark px-2 py-0.5 font-hand-sc text-[10px] sm:text-xs" title="Playtime in this round">
   451→                    RND {fmtTime(times.roundMs)}
   452→                  </div>
   453→                )}
   454→                {settings.showTotalTime && (
   455→                  <div className="doodle-box doodle-dark px-2 py-0.5 font-hand-sc text-[10px] sm:text-xs" title="Playtime across all rounds">
   456→                    SUM {fmtTime(times.totalMs)}
   457→                  </div>
   458→                )}
   459→              </div>
   460→            )}
   461→          </div>
   462→
   463→          {/* top right: THE DEALER */}
   464→          <div className="absolute right-3 top-3 z-30 flex flex-col items-end gap-1">
   465→            <div className="doodle-box px-3 py-1 rotate-[1.5deg]">
   466→              <span className="font-marker text-lg text-[#8f2418]">THE DEALER</span>
   467→              {snap.voodooEnemy > 0 && (
   468→                <img src={assets.voodoo} alt="voodoo" className="ml-2 inline h-6 w-6 wiggle align-middle" />
   469→              )}
   470→              {snap.enemyCuffed && (
   471→                <span className="ml-2 font-hand-sc text-sm text-[#9b8ec4] underline">cuffed</span>
   472→              )}
   473→            </div>
   474→            {snap.enemySkills.length > 0 && (
   475→              <div className="doodle-box flex items-center gap-1 px-2 py-0.5 rotate-[1deg]">
   476→                <span className="font-hand-sc text-[10px] text-[#4a4036]">SKILLS</span>
   477→                {snap.enemySkills.map((s) => (
   478→                  <img
   479→                    key={s.id}
   480→                    src={s.icon}
   481→                    alt={s.id}
   482→                    title={SKILL_META[s.id].name}
   483→                    className="h-6 w-6"
   484→                  />
   485→                ))}
   486→              </div>
   487→            )}
   488→            <div className="flex gap-1">
   489→              {Array.from({ length: snap.maxHpEnemy }).map((_, i) => (
   490→                <img key={i} src={i < snap.hpEnemy ? assets.heartFull : assets.heartEmpty} alt="hp" className="h-8 w-8" />
   491→              ))}
   492→            </div>
   493→          </div>
   494→

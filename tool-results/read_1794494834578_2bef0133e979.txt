   132→function shuffled<T>(arr: T[]): T[] {
   133→  const a = [...arr];
   134→  for (let i = a.length - 1; i > 0; i--) {
   135→    const j = Math.floor(Math.random() * (i + 1));
   136→    [a[i], a[j]] = [a[j], a[i]];
   137→  }
   138→  return a;
   139→}

   245→  /** Live buyability of one offer (slots & cash can change any moment). */
   246→  private offerView(o: ShopOffer): ShopOfferView {
   247→    const st = this.logic.state;
   248→    let disabled = false;
   249→    let reason: string | null = null;
   250→    if (o.sold) {
   251→      reason = "SOLD";
   252→    } else if (this.activeSkillCount() >= MAX_SKILLS) {
   253→      disabled = true;
   254→      reason = `SLOTS FULL (${MAX_SKILLS})`;
   255→    } else if (this.money < o.price) {
   256→      disabled = true;
   257→      reason = "TOO POOR";
   258→    }
   259→    void st;
   260→    return { ...o, disabled, reason };
   261→  }

   294→  /** Which cards are legally available on THIS table?
   295→   *  EPIC needs NORMAL or above; LEGEND needs INSANE / ENDLESS AND the
   296→   *  allow-legend switch. The mutator card pool gates everything further. */
   297→  private poolFor(diff: Difficulty, mode: Mode): SkillId[] {
   298→    const mut = this.mut;
   299→    let pool = ALL_SKILL_IDS;
   300→    if (mut) pool = pool.filter((id) => mut.pool.includes(id));
   301→    const epicOk = diff !== "easy";
   302→    const legendOk = (diff === "insane" || mode === "endless") && (mut ? mut.allowLegend : true);
   303→    return pool.filter((id) => {
   304→      const r = RARITY[id];
   305→      if (r === "epic") return epicOk;
   306→      if (r === "legend") return legendOk;
   307→      return true;
   308→    });
   309→  }

   311→  /** The dice pool: legends are NEVER rolled (shop only). */
   312→  private dicePool(): SkillId[] {
   313→    return this.poolFor(
   314→      this.logic.state.difficulty,
   315→      this.logic.state.mode,
   316→    ).filter((id) => RARITY[id] !== "legend");
   317→  }

   319→  /** Draw n skills from the pool; duplicates allowed stack charges instead. */
   320→  private drawSkills(side: Side, pool: SkillId[], n: number) {
   321→    const allowDup = this.mut?.allowDup ?? false;
   322→    const bag = allowDup ? pool : pool.filter((id) => !this.logic.state[side === "player" ? "skills" : "enemySkills"][id].owned);
   323→    const src = bag.length ? bag : pool;
   324→    for (let k = 0; k < n; k++) {
   325→      const id = src[Math.floor(Math.random() * src.length)];
   326→      this.logic.grantSkill(side, id);
   327→    }
   328→  }

   330→  async startGame(diff: Difficulty | "endless", mut?: MutConfig) {
   331→    gameAudio.ensure();
   332→    gameAudio.startAmbient();
   333→    this.lowHpTaunted = false;
   334→    this.money = START_MONEY;
   335→    this.ghosts = [];
   336→    this.shopResolve = null;
   337→    this.tStart = performance.now();
   338→    this.tRoundStart = this.tStart;
   339→    this.tEnd = 0;
   340→    // ENDLESS counts as INSANE: same sharp brain, legends allowed in the shop.
   341→    const mode: Mode = diff === "endless" ? "endless" : "standard";
   342→    const baseDiff: Difficulty = diff === "endless" ? "insane" : diff;
   343→    this.mode = mode;
   344→    this.mut = mut ?? null;
   345→    if (mut) {
   346→      const filled: MutConfig = { ...MUT_DEFAULTS, ...mut };
   347→      this.mut = filled;
   348→      this.logic = new RouletteLogic(baseDiff, mode, {
   349→        hpPlayer: filled.hpPlayer,
   350→        hpEnemy: filled.hpEnemy,
   351→      });
   352→    } else {
   353→      this.logic = new RouletteLogic(baseDiff, mode);
   354→    }
   355→    this.ai = new EnemyAI(this.logic, SMART[this.mut?.ai ?? baseDiff]);
   356→    this.setupScripts();
   357→    this.scene.clearSpentShells();
   358→    this.scene.dealerCuffed(false);
   359→    this.scene.dollyHome();
   360→    this.scene.voodooEndFX("player");
   361→    this.scene.voodooEndFX("enemy");
   362→    this.notify();
   363→    if (mode === "endless") {
   364→      this.toast(t("endlessIntro"), "info");
   365→    } else {
   366→      this.toast(t("startCash", { m: START_MONEY }), "info");
   367→    }
   368→
   369→    await sleep(300);
   370→    await this.diceIntro();
   371→    await this.reloadSequence();
   372→    await this.playerTurnStart();
   373→  }

   375→  /** Compile the mutator's custom-effect scripts and wire their hooks. */
   376→  private setupScripts() {
   377→    const rules = this.mut?.rules ?? [];
   378→    this.script = null;
   379→    if (!rules.length) return;
   380→    let program: Program | null = null;
   381→    const errors: string[] = [];
   382→    for (const src of rules) {
   383→      const res = parseScript(src);
   384→      errors.push(...res.errors);
   385→      if (res.errors.length) continue;
   386→      if (!program) program = { setup: [], handlers: [] };
   387→      program.setup.push(...res.program.setup);
   388→      program.handlers.push(...res.program.handlers);
   389→    }
   390→    for (const e of errors.slice(0, 4)) this.toast(`[rule] ${e}`, "bad");
   391→    if (!program) return;
   392→    const rt = new ScriptRuntime(program, {
   393→      runSkill: (id, target) => this.forceSkillFx(id, target),
   394→      heal: (side, n) => {
   395→        const did = this.logic.healSide(side, n);
   396→        if (did) this.onEvent({ type: "dmgPop", side, text: `+${n} HP`, tone: "reflect" });
   397→      },
   398→      damage: (side, n) => {
   399→        const dead = this.logic.damageSide(side, n);
   400→        if (side === "enemy") {
   401→          this.scene.bloodBurst("enemy");
   402→          if (this.logic.state.mode === "endless") this.onEvent({ type: "dmgPop", side, text: `SCORE +${n}`, tone: "hit" });
   403→          else this.onEvent({ type: "dmgPop", side, text: `-${n} HP`, tone: "hit" });
   404→        } else {
   405→          this.onEvent({ type: "screenSplat" });
   406→          this.scene.addTrauma(0.4);
   407→          if (!dead) this.onEvent({ type: "dmgPop", side, text: `-${n} HP`, tone: "hit" });
   408→        }
   409→      },
   410→      say: (text) => this.toast(text, "info"),
   411→    });
   412→    rt.stateRef = () => ({ hp: this.logic.state.hp, score: this.logic.state.score, round: this.logic.state.round });
   413→    this.script = rt;
   414→    rt.setup();
   415→    if (this.logic.state.hp.player <= 0) {
   416→      // a setup rule killed the player before the first round — mercy: undo it
   417→      this.logic.state.hp.player = 1;
   418→      this.toast("[custom rules] the setup killed you already... you start at 1 HP instead.", "bad");
   419→    }
   420→  }

   431→  /** A script FORCED a skill's effect — apply it with a matching mini-FX. */
   432→  private async forceSkillFx(id: SkillId | "all", target: Side) {
   433→    const real: SkillId =
   434→      id === "all"
   435→        ? (ALL_SKILL_IDS[Math.floor(Math.random() * ALL_SKILL_IDS.length)] as SkillId)
   436→        : id;
   437→    this.logic.forceSkill(real, target);
   438→    this.onEvent({ type: "skillFlash", id: real });
   439→    const label = SKILL_META[real].name;
   440→    switch (real) {
   441→      case "cuffs":
   442→        gameAudio.cuffs();
   443→        if (target === "enemy") {
   444→          this.scene.dealerCuffed(true);
   445→          this.toast(`CUSTOM RULE: ${label} — the DEALER is cuffed!`, "good");
   446→        } else {
   447→          this.toast(`CUSTOM RULE: ${label} — YOUR wrists are locked!`, "bad");
   448→        }
   449→        break;
   450→      case "coffee":
   451→        gameAudio.heal();
   452→        void this.scene.coffeeDrinkFX(target);
   453→        this.toast(`CUSTOM RULE: ${label} — ${target === "player" ? "you" : "the DEALER"} take a sip.`, target === "player" ? "good" : "bad");
   454→        break;
   455→      case "reroll":
   456→        gameAudio.spinCylinder();
   457→        this.scene.addTrauma(0.2);
   458→        this.toast(`CUSTOM RULE: ${label} — the chamber spins!`, "info");
   459→        break;
   460→      case "voodoo":
   461→        gameAudio.voodooCast();
   462→        this.scene.voodooCastFX(target);
   463→        this.toast(`CUSTOM RULE: ${label} curse on ${target === "player" ? "YOU" : "the DEALER"}!`, "voodoo");
   464→        break;
   465→      case "peek":
   466→        gameAudio.scanDing();
   467→        this.toast(`CUSTOM RULE: ${label} — ${target === "player" ? "you" : "the DEALER"} inspect the chamber.`, "info");
   468→        break;
   469→      case "eject":
   470→        gameAudio.ejectShell();
   471→        this.toast(`CUSTOM RULE: ${label} — a shell is tossed!`, "info");
   472→        break;
   473→      case "gamble":
   474→        this.onEvent({ type: "coinFlip", win: Math.random() < 0.5 });
   475→        this.toast(`CUSTOM RULE: ${label} — a coin flips!`, "info");
   476→        break;
   477→      default:
   478→        this.toast(`CUSTOM RULE: ${label} fires!`, "info");
   479→    }
   480→    this.notify();
   481→  }

   520→  /** Dice intro: the dice hands out the player's skills (2 by default,
   521→   *  mutators may change the count and the pool), then the enemy quietly

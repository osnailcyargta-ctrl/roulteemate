     1→     1→"use client";

     3→     3→/**

     5→     5→ * Mounts the engine (Three.js) plus all doodle overlays: menu, HP,

     9→     9→
    10→    10→import { useCallback, useEffect, useMemo, useRef, useState } from "react";
    11→    11→import { GameEngine, type HudEvent, type HudSnapshot } from "@/game/engine";
    12→    12→import { SKILL_META, dataURL, texBulletPip, texHeart, texSkillIcon, texSplat, texVoodooMask, type SkillId } from "@/game/doodle";
    13→    13→import type { ShellType } from "@/game/logic";
    14→    14→import { gameAudio } from "@/game/audio";
    15→    15→
    16→    16→type Toast = { id: number; text: string; tone: string };
    17→    17→type Pop = { id: number; side: string; text: string; tone: string };
    18→    18→type Splat = { id: number; url: string; left: number; top: number; size: number; rot: number };
    19→    19→type Stamp = { id: number; idSkill: SkillId };
    20→    20→
    21→    21→const DIFFS: { id: "easy" | "normal" | "insane"; label: string; desc: string }[] = [
    22→    22→  { id: "easy", label: "EASY", desc: "Your 6 HP vs his 5. The Dealer has 1 skill." },
    23→    23→  { id: "normal", label: "NORMAL", desc: "5 vs 5 HP. The Dealer has 2 skills." },
    24→    24→  { id: "insane", label: "INSANE", desc: "Your 4 HP vs his 5. 3 skills + shop legends." },
    25→    25→];
    26→    26→
    27→    27→const fmtTime = (ms: number) => {
    28→    28→  const s = Math.max(0, Math.floor(ms / 1000));
    29→    29→  return `${String(Math.floor(s / 60)).padStart(2, "0")}:${String(s % 60).padStart(2, "0")}`;
    30→    30→};
    31→    31→
    32→    32→export default function GameShell() {
    33→    33→  const mountRef = useRef<HTMLDivElement>(null);
    34→    34→  const engineRef = useRef<GameEngine | null>(null);
    35→    35→  const evId = useRef(0);
    36→    36→  const [ready, setReady] = useState(false);
    37→    37→  const [snap, setSnap] = useState<HudSnapshot | null>(null);
    38→    38→  const [toasts, setToasts] = useState<Toast[]>([]);
    39→    39→  const [pops, setPops] = useState<Pop[]>([]);
    40→    40→  const [splats, setSplats] = useState<Splat[]>([]);
    41→    41→  const [stamps, setStamps] = useState<Stamp[]>([]);
    42→    42→  const [banner, setBanner] = useState<{ title: string; sub?: string } | null>(null);
    43→    43→  const [reveal, setReveal] = useState<{ shell: "live" | "blank" } | null>(null);
    44→    44→  const [showHelp, setShowHelp] = useState(false);
    45→    45→  const [muted, setMuted] = useState(false);
    46→    46→  const [diff, setDiff] = useState<"easy" | "normal" | "insane">("normal");
    47→    47→  const [reloadPips, setReloadPips] = useState(0);
    48→    48→  const [flashRed, setFlashRed] = useState(false);
    49→    49→  const [revealQueue, setRevealQueue] = useState<(ShellType | null)[] | null>(null);
    50→    50→  const [times, setTimes] = useState<{ roundMs: number; totalMs: number }>({ roundMs: 0, totalMs: 0 });
    51→    51→  const [shopTip, setShopTip] = useState<{ id: SkillId | "bell"; xPct: number; yPct: number } | null>(null);
    52→    52→
    53→    53→  /* ---------- HUD asset dataURLs (after mount, client-only) ---------- */
    54→    54→  const assets = useMemo(() => {
    55→    55→    if (!ready) return null;
    56→    56→    return {
    57→    57→      heartFull: dataURL("heart-full", texHeart("full")),
    58→    58→      heartEmpty: dataURL("heart-empty", texHeart("empty")),
    59→    59→      pipLive: dataURL("pip-live", texBulletPip("live")),
    60→    60→      pipBlank: dataURL("pip-blank", texBulletPip("blank")),
    61→    61→      voodoo: dataURL("voodoo-mask", texVoodooMask()),
    62→    62→      splat0: dataURL("splat-hud-0", texSplat(333)),
    63→    63→      splat1: dataURL("splat-hud-1", texSplat(444)),
    64→    64→    };
    65→    65→  }, [ready]);
    66→    66→
    67→    67→  const assetsRef = useRef(assets);
    68→    68→  useEffect(() => {
    69→    69→    assetsRef.current = assets;
    70→    70→  }, [assets]);
    71→    71→
    72→    72→  const handleEvent = useCallback((e: HudEvent) => {
    73→    73→    const id = ++evId.current;
    74→    74→    switch (e.type) {
    75→    75→      case "toast":
    76→    76→        setToasts((t) => [...t.slice(-4), { id, text: e.text, tone: e.tone ?? "info" }]);
    77→    77→        setTimeout(() => setToasts((t) => t.filter((x) => x.id !== id)), 3200);
    78→    78→        break;
    79→    79→      case "banner":
    80→    80→        setBanner({ title: e.title, sub: e.sub });
    81→    81→        setTimeout(() => setBanner(null), 1700);
    82→    82→        break;
    83→    83→      case "skillGet":
    84→    84→        setStamps((s) => [...s, { id, idSkill: e.id }]);
    85→    85→        setTimeout(() => setStamps((s) => s.filter((x) => x.id !== id)), 1500);
    86→    86→        break;
    87→    87→      case "shellReveal":
    88→    88→        if (e.shell) {
    89→    89→          setReveal({ shell: e.shell });
    90→    90→          setTimeout(() => setReveal(null), 1900);
    91→    91→        }
    92→    92→        break;
    93→    93→      case "shellRevealQueue":
    94→    94→        setRevealQueue(e.shells);
    95→    95→        setTimeout(() => setRevealQueue(null), 2800);
    96→    96→        break;
    97→    97→      case "dmgPop":
    98→    98→        setPops((p) => [...p.slice(-5), { id, side: e.side, text: e.text, tone: e.tone ?? "hit" }]);
    99→    99→        setTimeout(() => setPops((p) => p.filter((x) => x.id !== id)), 1500);
   100→   100→        break;
   101→   101→      case "screenSplat": {
   102→   102→        const url = evId.current % 2 === 0 ? "splat-hud-0" : "splat-hud-1";
   103→   103→        const a = assetsRef.current;
   104→   104→        if (!a) break;
   105→   105→        setSplats((s) => [
   106→   106→          ...s.slice(-3),
   107→   107→          {
   108→   108→            id,
   109→   109→            url: a[url],
   110→   110→            left: 15 + Math.random() * 55,
   111→   111→            top: 8 + Math.random() * 50,
   112→   112→            size: 130 + Math.random() * 160,
   113→   113→            rot: Math.random() * 360,
   114→   114→          },
   115→   115→        ]);
   116→   116→        setFlashRed(true);
   117→   117→        setTimeout(() => setFlashRed(false), 180);
   118→   118→        setTimeout(() => setSplats((s) => s.filter((x) => x.id !== id)), 2600);
   119→   119→        break;
   120→   120→      }
   121→   121→      case "skillFlash":
   122→   122→        break;
   123→   123→      case "reloadStep":
   124→   124→        setReloadPips(e.loaded);
   125→   125→        setTimeout(() => setReloadPips(0), 2400);
   126→   126→        break;
   127→   127→      case "shopTip":
   128→   128→        setShopTip(e.id ? { id: e.id, xPct: e.xPct, yPct: e.yPct } : null);
   129→   129→        break;
   130→   130→    }
   131→   131→  }, []);
   132→   132→
   133→   133→  /* ---------- mount engine ---------- */
   134→   134→  useEffect(() => {
   135→   135→    let disposed = false;
   136→   136→    (async () => {
   137→   137→      try {
   138→   138→        await document.fonts.ready;
   139→   139→      } catch {
   140→   140→        /* noop */
   141→   141→      }
   142→   142→      if (disposed || !mountRef.current) return;
   143→   143→      const engine = new GameEngine();
   144→   144→      engineRef.current = engine;
   145→   145→      (window as unknown as { __eng?: GameEngine }).__eng = engine;
   146→   146→      engine.onHud = (s) => setSnap(s);
   147→   147→      engine.onEvent = (e) => handleEvent(e);
   148→   148→      engine.mount(mountRef.current);
   149→   149→      setReady(true);
   150→   150→    })();
   151→   151→    return () => {
   152→   152→      disposed = true;
   153→   153→      engineRef.current?.dispose();
   154→   154→      engineRef.current = null;
   155→   155→    };
   156→   156→
   157→   157→  }, []);
   158→   158→
   159→   159→  const toggleMute = useCallback(() => {
   160→   160→    setMuted((prev) => {
   161→   161→      const m = !prev;
   162→   162→      gameAudio.setMuted(m);
   163→   163→      return m;
   164→   164→    });
   165→   165→  }, []);
   166→   166→
   167→   167→  /* ---------- keyboard ---------- */
   168→   168→  useEffect(() => {
   169→   169→    const onKey = (ev: KeyboardEvent) => {
   170→   170→      const eng = engineRef.current;
   171→   171→      if (!eng || !snap) return;
   172→   172→      const k = ev.key.toLowerCase();
   173→   173→      if (k === "q") void eng.playerShoot("player");
   174→   174→      else if (k === "e") void eng.playerShoot("enemy");
   175→   175→      else if (k === "h") setShowHelp((v) => !v);
   176→   176→      else if (k === "m") toggleMute();
   177→   177→      else if (/^[1-4]$/.test(k)) {
   178→   178→        const owned = snap.skills.filter((s) => s.owned && s.ghostTurns <= 0);
   179→   179→        const target = owned[parseInt(k) - 1];
   180→   180→        if (target) void eng.playerUseSkill(target.id);
   181→   181→      }
   182→   182→    };
   183→   183→    window.addEventListener("keydown", onKey);
   184→   184→    return () => window.removeEventListener("keydown", onKey);
   185→   185→
   186→   186→  }, [snap, toggleMute]);
   187→   187→
   188→   188→  /* ---------- playtime ticker (round + total) ---------- */
   189→   189→  const phaseKey = snap?.phase ?? "menu";
   190→   190→  useEffect(() => {
   191→   191→    if (phaseKey === "menu") return;
   192→   192→    const tick = () => setTimes(engineRef.current?.getTimes() ?? { roundMs: 0, totalMs: 0 });
   193→   193→    tick();
   194→   194→    const iv = setInterval(tick, 1000);
   195→   195→    return () => clearInterval(iv);
   196→   196→  }, [phaseKey]);
   197→   197→
   198→   198→  const isPlayerTurn = snap?.phase === "playerTurn" && snap.turn === "player" && !snap.busy;
   199→   199→  const phase = snap?.phase ?? "menu";
   200→   200→  const inGame = phase !== "menu";
   201→   201→  const lowHp = inGame && snap ? snap.hpPlayer : 0;
   202→   202→
   203→   203→  /* ---------- render ---------- */
   204→   204→  return (
   205→   205→    <div className="fixed inset-0 overflow-hidden bg-[#0b0a08] select-none">
   206→   206→      {/* 3D world */}
   207→   207→      <div ref={mountRef} className="absolute inset-0" />
   208→   208→
   209→   209→      {/* cinematic overlays */}
   210→   210→      <div className="pointer-events-none absolute inset-0 vignette-overlay z-10" />
   211→   211→      <div className="pointer-events-none absolute inset-0 grain-overlay z-10" />
   212→   212→      {flashRed && <div className="pointer-events-none absolute inset-0 z-10 bg-[rgba(199,62,46,0.35)]" />}
   213→   213→
   214→   214→      {/* LOW-HP BLOOD VIGNETTE: starts at 2 HP, thickest at 1 HP */}
   215→   215→      {snap && lowHp === 2 && (
   216→   216→        <div className="pointer-events-none absolute inset-0 z-10 blood-vignette" />
   217→   217→      )}
   218→   218→      {snap && lowHp === 1 && (
   219→   219→        <>
   220→   220→          <div className="pointer-events-none absolute inset-0 z-10 blood-vignette heavy" />
   221→   221→          <div className="pointer-events-none absolute inset-0 z-10 overflow-hidden">
   222→   222→            <span className="blood-drip drip-1" />
   223→   223→            <span className="blood-drip drip-2" />
   224→   224→            <span className="blood-drip drip-3" />
   225→   225→            <span className="blood-drip drip-4" />
   226→   226→          </div>
   227→   227→          <div className="pointer-events-none absolute inset-0 z-10 pulse-danger" />
   228→   228→        </>
   229→   229→      )}
   230→   230→
   231→   231→      {snap && snap.voodooPlayer > 0 && phase !== "menu" && (
   232→   232→        <div
   233→   233→          className="pointer-events-none absolute inset-0 z-10"
   234→   234→          style={{ boxShadow: "inset 0 0 120px rgba(94,156,91,0.45)" }}
   235→   235→        />
   236→   236→      )}
   237→   237→
   238→   238→      {/* blood on screen */}
   239→   239→      <div className="pointer-events-none absolute inset-0 z-20 overflow-hidden">
   240→   240→        {assets &&
   241→   241→          splats.map((s) => (
   242→   242→
   243→   243→            <img
   244→   244→              key={s.id}
   245→   245→              src={s.url}
   246→   246→              alt=""
   247→   247→              className="absolute pop-in"
   248→   248→              style={{
   249→   249→                left: `${s.left}%`,
   250→   250→                top: `${s.top}%`,
   251→   251→                width: s.size,
   252→   252→                transform: `rotate(${s.rot}deg)`,
   253→   253→                transition: "opacity 1.2s ease 1s",
   254→   254→                opacity: 0.92,
   255→   255→              }}
   256→   256→            />
   257→   257→          ))}
   258→   258→      </div>
   259→   259→
   260→   260→      {/* damage pops */}
   261→   261→      <div className="pointer-events-none absolute inset-0 z-30">
   262→   262→        {pops.map((p) => (
   263→   263→          <div
   264→   264→            key={p.id}
   265→   265→            className={`absolute rise-fade font-marker text-3xl drop-shadow-[3px_3px_0_rgba(0,0,0,0.7)] ${
   266→   266→              p.tone === "reflect" ? "text-[#8fdc8a]" : p.tone === "blank" ? "text-[#cfd8e3]" : "text-[#ff6b57]"
   267→   267→            }`}
   268→   268→            style={p.side === "enemy" ? { left: "58%", top: "28%" } : { left: "42%", top: "62%" }}
   269→   269→          >
   270→   270→            {p.text}
   271→   271→          </div>
   272→   272→        ))}
   273→   273→      </div>
   274→   274→
   275→   275→      {/* ================= IN-GAME HUD ================= */}
   276→   276→      {inGame && assets && snap && (
   277→   277→        <>
   278→   278→          {/* shell counter board — the honest numbers fade turn by turn,
   279→   279→              their opacity really stepping down until it hits zero */}
   280→   280→          {snap.round > 0 && (
   281→   281→          <div className="absolute left-1/2 top-3 z-30 -translate-x-1/2">
   282→   282→            <div className="relative">
   283→   283→              <div
   284→   284→                className="doodle-box flex items-center gap-3 px-4 py-1.5 transition-opacity duration-[1100ms] ease-in-out"
   285→   285→                style={{ opacity: snap.countsPublic ? Math.min(1, snap.countdown / 3) : 0 }}
   286→   286→              >
   287→   287→                <div className="flex items-center gap-1">
   288→   288→                  <img src={assets.pipLive} alt="live" className="h-7 w-7" />
   289→   289→                  <span className="font-marker text-xl text-[#8f2418]">x{snap.live}</span>
   290→   290→                </div>
   291→   291→                <div className="h-6 w-0.5 bg-[#221a12]/30" />
   292→   292→                <div className="flex items-center gap-1">
   293→   293→                  <img src={assets.pipBlank} alt="blank" className="h-7 w-7" />
   294→   294→                  <span className="font-marker text-xl text-[#57697c]">x{snap.blank}</span>
   295→   295→                </div>
   296→   296→              </div>
   297→   297→              {!snap.countsPublic && (
   298→   298→                <div className="absolute inset-0 flex items-center justify-center">
   299→   299→                  <div className="doodle-box doodle-dark board-dark-box pop-in-delayed flex items-center gap-2 whitespace-nowrap px-4 py-1.5">
   300→   300→                    <span className="font-marker text-xl tracking-widest text-[#8b8577]">? ? ?</span>
   301→   301→                    <span className="rotate-[-2deg] font-hand-sc text-xs text-[#d9a441]">memorize!</span>
   302→   302→                  </div>
   303→   303→                </div>
   304→   304→              )}
   305→   305→            </div>
   306→   306→            {snap.known && (
   307→   307→              <div className="mt-1 flex justify-center">
   308→   308→                <div
   309→   309→                  className={`pop-in font-hand-sc px-2 py-0.5 border-2 border-[#221a12] bg-[#f3ecd7] text-sm rotate-[-2deg] ${
   310→   310→                    snap.known === "live" ? "text-[#8f2418]" : "text-[#57697c]"
   311→   311→                  }`}
   312→   312→                >
   313→   313→                  CHAMBER: {snap.known.toUpperCase()}
   314→   314→                </div>
   315→   315→              </div>
   316→   316→            )}
   317→   317→          </div>

   319→   319→
   320→   320→          {/* top left: round + buttons */}
   321→   321→          <div className="absolute left-2 top-2 z-30 flex flex-col gap-1.5 sm:left-3 sm:top-3 sm:gap-2">
   322→   322→            <div className="doodle-box doodle-dark px-2 py-0.5 font-marker text-xs rotate-[-1.5deg] sm:px-3 sm:py-1 sm:text-sm">
   323→   323→              ROUND {Math.max(1, snap.round)}
   324→   324→            </div>
   325→   325→            <div
   326→   326→              className="doodle-box w-fit px-2 py-0.5 font-marker text-xs text-[#8a6d1d] rotate-[1deg] sm:text-sm"

   331→   331→            <div className="flex gap-1.5">

   337→   337→              </div>
   338→   338→            </div>

   340→   340→              <button

   342→   342→                onClick={() => setShowHelp(true)}
   343→   343→                aria-label="How to play"
   344→   344→              >
   345→   345→                ? RULES
   346→   346→              </button>
   347→   347→              <button className="doodle-btn px-2 py-0.5 font-hand-sc text-xs sm:px-2.5 sm:py-1 sm:text-sm" onClick={toggleMute} aria-label="Sound">

   361→   361→              {snap.enemyCuffed && (

   370→   370→                    key={s.id}
   371→   371→                    src={s.icon}
   372→   372→                    alt={s.id}
   373→   373→                    title={SKILL_META[s.id].name}

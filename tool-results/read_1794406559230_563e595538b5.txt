     1→"use client";
     2→
     3→/**
     4→ * GameShell — lapisan React untuk HUD ULTIMATE ROULETTE.
     5→ * Mount engine (Three.js) + seluruh overlay doodle: menu, HP, shell counter,
     6→ * skill bar, toast, banner, rules, dan layar menang/kalah.
     7→ */
     8→
     9→import { useCallback, useEffect, useMemo, useRef, useState } from "react";
    10→import { GameEngine, type HudEvent, type HudSnapshot } from "@/game/engine";
    11→import { SKILL_META, dataURL, texBulletPip, texHeart, texSkillIcon, texSplat, texVoodooMask, type SkillId } from "@/game/doodle";
    12→import { gameAudio } from "@/game/audio";
    13→
    14→type Toast = { id: number; text: string; tone: string };
    15→type Pop = { id: number; side: string; text: string; tone: string };
    16→type Splat = { id: number; url: string; left: number; top: number; size: number; rot: number };
    17→type Stamp = { id: number; idSkill: SkillId };
    18→
    19→const DIFFS: { id: "gampang" | "normal" | "gila"; label: string; desc: string }[] = [
    20→  { id: "gampang", label: "GAMPANG", desc: "HP kamu 5. PAKDE agak lemot." },
    21→  { id: "normal", label: "NORMAL", desc: "HP 4 vs 4. Adil dan menakutkan." },
    22→  { id: "gila", label: "GILA", desc: "PAKDE tajam. Persiapan mati." },
    23→];
    24→
    25→export default function GameShell() {
    26→  const mountRef = useRef<HTMLDivElement>(null);
    27→  const engineRef = useRef<GameEngine | null>(null);
    28→  const evId = useRef(0);
    29→  const [ready, setReady] = useState(false);
    30→  const [snap, setSnap] = useState<HudSnapshot | null>(null);
    31→  const [toasts, setToasts] = useState<Toast[]>([]);
    32→  const [pops, setPops] = useState<Pop[]>([]);
    33→  const [splats, setSplats] = useState<Splat[]>([]);
    34→  const [stamps, setStamps] = useState<Stamp[]>([]);
    35→  const [banner, setBanner] = useState<{ title: string; sub?: string } | null>(null);
    36→  const [reveal, setReveal] = useState<{ shell: "live" | "blank" } | null>(null);
    37→  const [showHelp, setShowHelp] = useState(false);
    38→  const [muted, setMuted] = useState(false);
    39→  const [diff, setDiff] = useState<"gampang" | "normal" | "gila">("normal");
    40→  const [reloadPips, setReloadPips] = useState(0);
    41→  const [flashRed, setFlashRed] = useState(false);
    42→
    43→  /* ---------- dataURL aset HUD (setelah mount, client-only) ---------- */
    44→  const assets = useMemo(() => {
    45→    if (!ready) return null;
    46→    return {
    47→      heartFull: dataURL("heart-full", texHeart("full")),
    48→      heartEmpty: dataURL("heart-empty", texHeart("empty")),
    49→      pipLive: dataURL("pip-live", texBulletPip("live")),
    50→      pipBlank: dataURL("pip-blank", texBulletPip("blank")),
    51→      voodoo: dataURL("voodoo-mask", texVoodooMask()),
    52→      splat0: dataURL("splat-hud-0", texSplat(333)),
    53→      splat1: dataURL("splat-hud-1", texSplat(444)),
    54→    };
    55→  }, [ready]);
    56→
    57→  const assetsRef = useRef(assets);
    58→  useEffect(() => {
    59→    assetsRef.current = assets;
    60→  }, [assets]);
    61→
    62→  const handleEvent = useCallback((e: HudEvent) => {
    63→    const id = ++evId.current;
    64→    switch (e.type) {
    65→      case "toast":
    66→        setToasts((t) => [...t.slice(-4), { id, text: e.text, tone: e.tone ?? "info" }]);
    67→        setTimeout(() => setToasts((t) => t.filter((x) => x.id !== id)), 3200);
    68→        break;
    69→      case "banner":
    70→        setBanner({ title: e.title, sub: e.sub });
    71→        setTimeout(() => setBanner(null), 1700);
    72→        break;
    73→      case "skillGet":
    74→        setStamps((s) => [...s, { id, idSkill: e.id }]);
    75→        setTimeout(() => setStamps((s) => s.filter((x) => x.id !== id)), 1500);
    76→        break;
    77→      case "shellReveal":
    78→        if (e.shell) {
    79→          setReveal({ shell: e.shell });
    80→          setTimeout(() => setReveal(null), 1900);
    81→        }
    82→        break;
    83→      case "dmgPop":
    84→        setPops((p) => [...p.slice(-5), { id, side: e.side, text: e.text, tone: e.tone ?? "hit" }]);
    85→        setTimeout(() => setPops((p) => p.filter((x) => x.id !== id)), 1500);
    86→        break;
    87→      case "screenSplat": {
    88→        const url = evId.current % 2 === 0 ? "splat-hud-0" : "splat-hud-1";
    89→        const a = assetsRef.current;
    90→        if (!a) break;
    91→        setSplats((s) => [
    92→          ...s.slice(-3),
    93→          {
    94→            id,
    95→            url: a[url],
    96→            left: 15 + Math.random() * 55,
    97→            top: 8 + Math.random() * 50,
    98→            size: 130 + Math.random() * 160,
    99→            rot: Math.random() * 360,
   100→          },
   101→        ]);
   102→        setFlashRed(true);
   103→        setTimeout(() => setFlashRed(false), 180);
   104→        setTimeout(() => setSplats((s) => s.filter((x) => x.id !== id)), 2600);
   105→        break;
   106→      }
   107→      case "skillFlash":
   108→        break;
   109→      case "reloadStep":
   110→        setReloadPips(e.loaded);
   111→        setTimeout(() => setReloadPips(0), 2400);
   112→        break;
   113→    }
   114→  }, []);
   115→
   116→  /* ---------- mount engine ---------- */
   117→  useEffect(() => {
   118→    let disposed = false;
   119→    (async () => {
   120→      try {
   121→        await document.fonts.ready;
   122→      } catch {
   123→        /* noop */
   124→      }
   125→      if (disposed || !mountRef.current) return;
   126→      const engine = new GameEngine();
   127→      engineRef.current = engine;
   128→      (window as unknown as { __eng?: GameEngine }).__eng = engine;
   129→      engine.onHud = (s) => setSnap(s);
   130→      engine.onEvent = (e) => handleEvent(e);
   131→      engine.mount(mountRef.current);
   132→      setReady(true);
   133→    })();
   134→    return () => {
   135→      disposed = true;
   136→      engineRef.current?.dispose();
   137→      engineRef.current = null;
   138→    };
   139→     
   140→  }, []);
   141→
   142→  const toggleMute = useCallback(() => {
   143→    setMuted((prev) => {
   144→      const m = !prev;
   145→      gameAudio.setMuted(m);
   146→      return m;
   147→    });
   148→  }, []);
   149→
   150→  /* ---------- keyboard ---------- */
   151→  useEffect(() => {
   152→    const onKey = (ev: KeyboardEvent) => {
   153→      const eng = engineRef.current;
   154→      if (!eng || !snap) return;
   155→      const k = ev.key.toLowerCase();
   156→      if (k === "q") void eng.playerShoot("player");
   157→      else if (k === "e") void eng.playerShoot("enemy");
   158→      else if (k === "h") setShowHelp((v) => !v);
   159→      else if (k === "m") toggleMute();
   160→      else if (/^[1-6]$/.test(k)) {
   161→        const owned = snap.skills.filter((s) => s.owned);
   162→        const target = owned[parseInt(k) - 1];
   163→        if (target) void eng.playerUseSkill(target.id);
   164→      }
   165→    };
   166→    window.addEventListener("keydown", onKey);
   167→    return () => window.removeEventListener("keydown", onKey);
   168→     
   169→  }, [snap]);
   170→
   171→  const eng = engineRef.current;
   172→  const isPlayerTurn = snap?.phase === "playerTurn" && snap.turn === "player" && !snap.busy;
   173→  const phase = snap?.phase ?? "menu";
   174→  const inGame = phase !== "menu";
   175→
   176→  /* ---------- render ---------- */
   177→  return (
   178→    <div className="fixed inset-0 overflow-hidden bg-[#0b0a08] select-none">
   179→      {/* dunia 3D */}
   180→      <div ref={mountRef} className="absolute inset-0" />
   181→
   182→      {/* overlay filmis */}
   183→      <div className="pointer-events-none absolute inset-0 vignette-overlay z-10" />
   184→      <div className="pointer-events-none absolute inset-0 grain-overlay z-10" />
   185→      {flashRed && <div className="pointer-events-none absolute inset-0 z-10 bg-[rgba(199,62,46,0.35)]" />}
   186→      {snap && snap.hpPlayer === 1 && phase !== "menu" && (
   187→        <div className="pointer-events-none absolute inset-0 z-10 pulse-danger" />
   188→      )}
   189→      {snap && snap.voodooPlayer > 0 && (
   190→        <div
   191→          className="pointer-events-none absolute inset-0 z-10"
   192→          style={{ boxShadow: "inset 0 0 120px rgba(94,156,91,0.45)" }}
   193→        />
   194→      )}
   195→
   196→      {/* darah di layar */}
   197→      <div className="pointer-events-none absolute inset-0 z-20 overflow-hidden">
   198→        {assets &&
   199→          splats.map((s) => (
   200→             
   201→            <img
   202→              key={s.id}
   203→              src={s.url}
   204→              alt=""
   205→              className="absolute pop-in"
   206→              style={{
   207→                left: `${s.left}%`,
   208→                top: `${s.top}%`,
   209→                width: s.size,
   210→                transform: `rotate(${s.rot}deg)`,
   211→                transition: "opacity 1.2s ease 1s",
   212→                opacity: 0.92,
   213→              }}
   214→            />
   215→          ))}
   216→      </div>
   217→
   218→      {/* damage pops */}
   219→      <div className="pointer-events-none absolute inset-0 z-30">
   220→        {pops.map((p) => (
   221→          <div
   222→            key={p.id}
   223→            className={`absolute rise-fade font-marker text-3xl drop-shadow-[3px_3px_0_rgba(0,0,0,0.7)] ${
   224→              p.tone === "reflect" ? "text-[#8fdc8a]" : p.tone === "blank" ? "text-[#cfd8e3]" : "text-[#ff6b57]"
   225→            }`}
   226→            style={p.side === "enemy" ? { left: "58%", top: "28%" } : { left: "42%", top: "62%" }}
   227→          >
   228→            {p.text}
   229→          </div>
   230→        ))}
   231→      </div>
   232→
   233→      {/* ================= HUD IN-GAME ================= */}
   234→      {inGame && assets && snap && (
   235→        <>
   236→          {/* papan shell counter */}
   237→          <div className="absolute left-1/2 top-3 z-30 -translate-x-1/2">
   238→            <div className="doodle-box px-4 py-1.5 flex items-center gap-3">
   239→              <div className="flex items-center gap-1">
   240→                { }
   241→                <img src={assets.pipLive} alt="live" className="h-7 w-7" />
   242→                <span className="font-marker text-xl text-[#8f2418]">x{snap.live}</span>
   243→              </div>
   244→              <div className="h-6 w-0.5 bg-[#221a12]/30" />
   245→              <div className="flex items-center gap-1">
   246→                { }
   247→                <img src={assets.pipBlank} alt="blank" className="h-7 w-7" />
   248→                <span className="font-marker text-xl text-[#57697c]">x{snap.blank}</span>
   249→              </div>
   250→            </div>
   251→            {snap.known && (
   252→              <div className="mt-1 flex justify-center">
   253→                <div
   254→                  className={`pop-in font-hand-sc px-2 py-0.5 border-2 border-[#221a12] bg-[#f3ecd7] text-sm rotate-[-2deg] ${
   255→                    snap.known === "live" ? "text-[#8f2418]" : "text-[#57697c]"
   256→                  }`}
   257→                >
   258→                  CHAMBER: {snap.known.toUpperCase()}
   259→                </div>
   260→              </div>
   261→            )}
   262→          </div>
   263→
   264→          {/* kiri atas: ronde + tombol */}
   265→          <div className="absolute left-2 top-2 z-30 flex flex-col gap-1.5 sm:left-3 sm:top-3 sm:gap-2">
   266→            <div className="doodle-box doodle-dark px-2 py-0.5 font-marker text-xs rotate-[-1.5deg] sm:px-3 sm:py-1 sm:text-sm">
   267→              RONDE {snap.round}
   268→            </div>
   269→            <div className="flex gap-1.5 sm:gap-2">
   270→              <button
   271→                className="doodle-btn px-2 py-0.5 font-hand-sc text-xs sm:px-2.5 sm:py-1 sm:text-sm"
   272→                onClick={() => setShowHelp(true)}
   273→                aria-label="Cara main"
   274→              >
   275→                ? CARA
   276→              </button>
   277→              <button className="doodle-btn px-2 py-0.5 font-hand-sc text-xs sm:px-2.5 sm:py-1 sm:text-sm" onClick={toggleMute} aria-label="Suara">
   278→                {muted ? "SUARA OFF" : "SUARA ON"}
   279→              </button>
   280→            </div>
   281→          </div>
   282→
   283→          {/* kanan atas: PAKDE */}
   284→          <div className="absolute right-3 top-3 z-30 flex flex-col items-end gap-1">
   285→            <div className="doodle-box px-3 py-1 rotate-[1.5deg]">
   286→              <span className="font-marker text-lg text-[#8f2418]">PAKDE</span>
   287→              {snap.voodooEnemy > 0 && (
   288→                 
   289→                <img src={assets.voodoo} alt="voodoo" className="ml-2 inline h-6 w-6 wiggle align-middle" />
   290→              )}
   291→              {snap.enemyCuffed && (
   292→                <span className="ml-2 font-hand-sc text-sm text-[#9b8ec4] underline">diborgol</span>
   293→              )}
   294→            </div>
   295→            <div className="flex gap-1">
   296→              {Array.from({ length: snap.maxHp }).map((_, i) => (
   297→                 
   298→                <img key={i} src={i < snap.hpEnemy ? assets.heartFull : assets.heartEmpty} alt="hp" className="h-8 w-8" />
   299→              ))}
   300→            </div>
   301→          </div>
   302→
   303→          {/* pemain: kiri atas di mobile, kiri bawah di desktop */}
   304→          <div className="absolute left-2 top-[88px] z-30 flex flex-col gap-1 sm:left-3 sm:top-auto sm:bottom-3">
   305→            <div className="doodle-box w-fit px-2 py-0.5 rotate-[-1.5deg] sm:px-3 sm:py-1">
   306→              <span className="font-marker text-sm sm:text-lg">KAMU</span>
   307→              {snap.voodooPlayer > 0 && (
   308→                <span className="ml-2 font-hand-sc text-sm text-[#5e9c5b]">voodoo {snap.voodooPlayer}</span>
   309→              )}
   310→              {snap.playerCuffed && (
   311→                <span className="ml-2 font-hand-sc text-sm text-[#9b8ec4] underline">diborgol</span>
   312→              )}
   313→            </div>
   314→            <div className="flex gap-0.5 sm:gap-1">
   315→              {Array.from({ length: snap.maxHp }).map((_, i) => (
   316→                <img key={i} src={i < snap.hpPlayer ? assets.heartFull : assets.heartEmpty} alt="hp" className="h-7 w-7 sm:h-9 sm:w-9" />
   317→              ))}
   318→            </div>
   319→          </div>
   320→
   321→          {/* indikator giliran */}
   322→          <div className="pointer-events-none absolute left-1/2 top-[74px] z-20 -translate-x-1/2 sm:top-16">
   323→            {snap.turn === "player" && (
   324→              <div className="doodle-box px-4 py-0.5 font-marker text-[#8f2418] rotate-[-2deg] float-y">
   325→                GILIRANMU
   326→              </div>
   327→            )}
   328→            {snap.turn === "enemy" && (
   329→              <div className="doodle-box doodle-dark px-4 py-0.5 font-marker rotate-[2deg] float-y">
   330→                GILIRAN PAKDE...
   331→              </div>
   332→            )}
   333→          </div>
   334→
   335→          {/* must shoot */}
   336→          {snap.mustShoot && snap.turn === "player" && (
   337→            <div className="pointer-events-none absolute left-1/2 top-28 z-30 -translate-x-1/2">
   338→              <div className="doodle-box doodle-red px-4 py-1 font-marker text-xl blink-soft rotate-[-3deg]">
   339→                WAJIB TEMBAK!
   340→              </div>
   341→            </div>
   342→          )}
   343→
   344→          {/* reload pips */}
   345→          {reloadPips > 0 && (
   346→            <div className="pointer-events-none absolute left-1/2 top-[62%] z-30 flex -translate-x-1/2 gap-1.5">
   347→              {Array.from({ length: 6 }).map((_, i) => (
   348→                <div
   349→                  key={i}
   350→                  className={`h-4 w-4 rounded-full border-[3px] border-[#f3ecd7] ${
   351→                    i < reloadPips ? "bg-[#c73e2e]" : "bg-transparent"
   352→                  }`}
   353→                />
   354→              ))}
   355→            </div>
   356→          )}
   357→
   358→          {/* skill bar */}
   359→          <div className="absolute bottom-[104px] left-1/2 z-30 flex -translate-x-1/2 items-end gap-1.5 sm:bottom-3 sm:gap-2">
   360→            {snap.skills
   361→              .filter((s) => s.owned)
   362→              .map((s, idx) => (
   363→                <button
   364→                  key={s.id}
   365→                  className={`doodle-btn relative flex h-16 w-16 flex-col items-center justify-center gap-0.5 sm:h-[74px] sm:w-[74px] ${
   366→                    s.usable ? "" : "opacity-60"
   367→                  }`}
   368→                  disabled={!s.usable}
   369→                  onClick={() => void eng?.playerUseSkill(s.id)}
   370→                  title={SKILL_META[s.id].desc}
   371→                >
   372→                  <img src={s.icon} alt={s.id} className="h-8 w-8 sm:h-9 sm:w-9" />
   373→                  <span className="font-hand-sc text-[11px] leading-none">{SKILL_META[s.id].name}</span>
   374→                  <span className="absolute -top-2 -left-1.5 font-marker text-[11px] text-[#8f2418]">
   375→                    {idx + 1}
   376→                  </span>
   377→                  {s.charges >= 0 && (
   378→                    <span className="absolute -bottom-1 right-1 flex gap-0.5">
   379→                      {Array.from({ length: Math.max(s.charges, 0) }).map((_, c) => (
   380→                        <span key={c} className="h-2 w-2 rounded-full bg-[#5e9c5b] border border-[#221a12]" />
   381→                      ))}
   382→                    </span>
   383→                  )}
   384→                  {s.cooldown > 0 && (
   385→                    <span className="absolute inset-0 flex items-center justify-center rounded-[inherit] bg-black/55 font-marker text-2xl text-[#f3ecd7]">
   386→                      {s.cooldown}
   387→                    </span>
   388→                  )}
   389→                </button>
   390→              ))}
   391→          </div>
   392→
   393→          {/* tombol tembak */}
   394→          <div className="absolute bottom-3 right-2 z-30 flex flex-row gap-2 sm:right-3 sm:flex-col">
   395→            <button
   396→              className={`doodle-btn doodle-red px-3 py-2.5 font-marker text-sm sm:px-5 sm:text-lg ${isPlayerTurn ? "" : "opacity-50"}`}
   397→              disabled={!isPlayerTurn}
   398→              onClick={() => void eng?.playerShoot("enemy")}
   399→            >
   400→              TEMBAK PAKDE [E]
   401→            </button>
   402→            <button
   403→              className={`doodle-btn px-3 py-2.5 font-marker text-sm sm:px-5 sm:text-lg ${isPlayerTurn ? "" : "opacity-50"}`}
   404→              disabled={!isPlayerTurn}
   405→              onClick={() => void eng?.playerShoot("player")}
   406→            >
   407→              TEMBAK DIRI [Q]
   408→            </button>
   409→          </div>
   410→
   411→          {/* toast kiri */}
   412→          <div className="pointer-events-none absolute left-3 top-28 z-30 flex w-64 flex-col gap-1.5">
   413→            {toasts.map((t) => (
   414→              <div
   415→                key={t.id}
   416→                className={`doodle-box pop-in px-3 py-1.5 font-hand text-[15px] leading-tight ${
   417→                  t.tone === "good"
   418→                    ? "border-[#3f6b3d]"
   419→                    : t.tone === "bad"
   420→                      ? "border-[#8f2418] text-[#8f2418]"
   421→                      : t.tone === "voodoo"
   422→                        ? "border-[#5e9c5b] text-[#3f6b3d] bg-[#e9f0e2]"
   423→                        : ""
   424→                }`}
   425→              >
   426→                {t.text}
   427→              </div>
   428→            ))}
   429→          </div>
   430→        </>
   431→      )}
   432→
   433→      {/* stamp skill didapat */}
   434→      <div className="pointer-events-none absolute inset-0 z-40 flex items-center justify-center">
   435→        {assets &&
   436→          stamps.map((s) => (
   437→            <div key={s.id} className="stamp-in absolute flex flex-col items-center">
   438→              <div className="doodle-box flex items-center gap-3 px-8 py-4 rotate-[-3deg]">
   439→                { }
   440→                <img src={snap?.skills.find((x) => x.id === s.idSkill)?.icon} alt={s.idSkill} className="h-16 w-16" />
   441→                <div>
   442→                  <div className="font-marker text-3xl text-[#8f2418]">{SKILL_META[s.idSkill].name}!</div>
   443→                  <div className="font-hand text-lg">{SKILL_META[s.idSkill].tag}</div>
   444→                </div>
   445→              </div>
   446→            </div>
   447→          ))}
   448→      </div>
   449→
   450→      {/* reveal shell (intuisi) */}
   451→      <div className="pointer-events-none absolute inset-0 z-40 flex items-center justify-center">
   452→        {assets && reveal && (

   455→            <img
   456→              src={reveal.shell === "live" ? assets.pipLive : assets.pipBlank}
   457→              alt={reveal.shell}
   458→              className="h-40 w-40 float-y"
   459→              style={{ imageRendering: "auto", transform: "scale(1.6)" }}
   460→            />
   461→            <div
   462→              className={`doodle-box px-5 py-1 font-marker text-2xl rotate-[-2deg] ${
   463→                reveal.shell === "live" ? "text-[#8f2418]" : "text-[#57697c]"
   464→              }`}
   465→            >
   466→              {reveal.shell!.toUpperCase()}!
   467→            </div>

   471→
   472→      {/* banner tengah */}

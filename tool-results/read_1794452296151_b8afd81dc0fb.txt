     1→"use client";
     2→
     3→/**
     4→ * GameShell — React layer for the ULTIMATE ROULETTE HUD.
     5→ * Mounts the engine (Three.js) plus all doodle overlays: menu, HP,
     6→ * shell counter, skill bar, toasts, banner, rules, win/lose screens,
     7→ * and the low-HP blood vignette.
     8→ */
     9→
    10→import { useCallback, useEffect, useMemo, useRef, useState } from "react";
    11→import { GameEngine, type HudEvent, type HudSnapshot } from "@/game/engine";
    12→import { SKILL_META, dataURL, texBulletPip, texHeart, texSkillIcon, texSplat, texVoodooMask, type SkillId } from "@/game/doodle";
    13→import type { ShellType } from "@/game/logic";
    14→import { gameAudio } from "@/game/audio";
    15→
    16→type Toast = { id: number; text: string; tone: string };
    17→type Pop = { id: number; side: string; text: string; tone: string };
    18→type Splat = { id: number; url: string; left: number; top: number; size: number; rot: number };
    19→type Stamp = { id: number; idSkill: SkillId };
    20→
    21→const DIFFS: { id: "easy" | "normal" | "insane"; label: string; desc: string }[] = [
    22→  { id: "easy", label: "EASY", desc: "6 vs 6 HP. The Dealer has 1 skill." },
    23→  { id: "normal", label: "NORMAL", desc: "Your 5 HP vs his 6. The Dealer has 2 skills." },
    24→  { id: "insane", label: "INSANE", desc: "Your 4 HP vs his 7. 3 skills + shop legends." },
    25→];
    26→
    27→const fmtTime = (ms: number) => {
    28→  const s = Math.max(0, Math.floor(ms / 1000));
    29→  return `${String(Math.floor(s / 60)).padStart(2, "0")}:${String(s % 60).padStart(2, "0")}`;
    30→};
    31→
    32→export default function GameShell() {
    33→  const mountRef = useRef<HTMLDivElement>(null);
    34→  const engineRef = useRef<GameEngine | null>(null);
    35→  const evId = useRef(0);
    36→  const [ready, setReady] = useState(false);
    37→  const [snap, setSnap] = useState<HudSnapshot | null>(null);
    38→  const [toasts, setToasts] = useState<Toast[]>([]);
    39→  const [pops, setPops] = useState<Pop[]>([]);
    40→  const [splats, setSplats] = useState<Splat[]>([]);
    41→  const [stamps, setStamps] = useState<Stamp[]>([]);
    42→  const [banner, setBanner] = useState<{ title: string; sub?: string } | null>(null);
    43→  const [reveal, setReveal] = useState<{ shell: "live" | "blank" } | null>(null);
    44→  const [showHelp, setShowHelp] = useState(false);
    45→  const [muted, setMuted] = useState(false);
    46→  const [diff, setDiff] = useState<"easy" | "normal" | "insane">("normal");
    47→  const [reloadPips, setReloadPips] = useState(0);
    48→  const [flashRed, setFlashRed] = useState(false);
    49→  const [revealQueue, setRevealQueue] = useState<(ShellType | null)[] | null>(null);
    50→  const [times, setTimes] = useState<{ roundMs: number; totalMs: number }>({ roundMs: 0, totalMs: 0 });
    51→  const [shopTip, setShopTip] = useState<{ id: SkillId | "bell"; xPct: number; yPct: number } | null>(null);
    52→
    53→  /* ---------- HUD asset dataURLs (after mount, client-only) ---------- */
    54→  const assets = useMemo(() => {
    55→    if (!ready) return null;
    56→    return {
    57→      heartFull: dataURL("heart-full", texHeart("full")),
    58→      heartEmpty: dataURL("heart-empty", texHeart("empty")),
    59→      pipLive: dataURL("pip-live", texBulletPip("live")),
    60→      pipBlank: dataURL("pip-blank", texBulletPip("blank")),
    61→      voodoo: dataURL("voodoo-mask", texVoodooMask()),
    62→      splat0: dataURL("splat-hud-0", texSplat(333)),
    63→      splat1: dataURL("splat-hud-1", texSplat(444)),
    64→    };
    65→  }, [ready]);
    66→
    67→  const assetsRef = useRef(assets);
    68→  useEffect(() => {
    69→    assetsRef.current = assets;
    70→  }, [assets]);
    71→
    72→  const handleEvent = useCallback((e: HudEvent) => {
    73→    const id = ++evId.current;
    74→    switch (e.type) {
    75→      case "toast":
    76→        setToasts((t) => [...t.slice(-4), { id, text: e.text, tone: e.tone ?? "info" }]);
    77→        setTimeout(() => setToasts((t) => t.filter((x) => x.id !== id)), 3200);
    78→        break;
    79→      case "banner":
    80→        setBanner({ title: e.title, sub: e.sub });
    81→        setTimeout(() => setBanner(null), 1700);
    82→        break;
    83→      case "skillGet":
    84→        setStamps((s) => [...s, { id, idSkill: e.id }]);
    85→        setTimeout(() => setStamps((s) => s.filter((x) => x.id !== id)), 1500);
    86→        break;
    87→      case "shellReveal":
    88→        if (e.shell) {
    89→          setReveal({ shell: e.shell });
    90→          setTimeout(() => setReveal(null), 1900);
    91→        }
    92→        break;
    93→      case "shellRevealQueue":
    94→        setRevealQueue(e.shells);
    95→        setTimeout(() => setRevealQueue(null), 2800);
    96→        break;
    97→      case "dmgPop":
    98→        setPops((p) => [...p.slice(-5), { id, side: e.side, text: e.text, tone: e.tone ?? "hit" }]);
    99→        setTimeout(() => setPops((p) => p.filter((x) => x.id !== id)), 1500);
   100→        break;
   101→      case "screenSplat": {
   102→        const url = evId.current % 2 === 0 ? "splat-hud-0" : "splat-hud-1";
   103→        const a = assetsRef.current;
   104→        if (!a) break;
   105→        setSplats((s) => [
   106→          ...s.slice(-3),
   107→          {
   108→            id,
   109→            url: a[url],
   110→            left: 15 + Math.random() * 55,
   111→            top: 8 + Math.random() * 50,
   112→            size: 130 + Math.random() * 160,
   113→            rot: Math.random() * 360,
   114→          },
   115→        ]);
   116→        setFlashRed(true);
   117→        setTimeout(() => setFlashRed(false), 180);
   118→        setTimeout(() => setSplats((s) => s.filter((x) => x.id !== id)), 2600);
   119→        break;
   120→      }
   121→      case "skillFlash":
   122→        break;
   123→      case "reloadStep":
   124→        setReloadPips(e.loaded);
   125→        setTimeout(() => setReloadPips(0), 2400);
   126→        break;
   127→      case "shopTip":
   128→        setShopTip(e.id ? { id: e.id, xPct: e.xPct, yPct: e.yPct } : null);
   129→        break;
   130→    }
   131→  }, []);
   132→
   133→  /* ---------- mount engine ---------- */
   134→  useEffect(() => {
   135→    let disposed = false;
   136→    (async () => {
   137→      try {
   138→        await document.fonts.ready;
   139→      } catch {
   140→        /* noop */
   141→      }
   142→      if (disposed || !mountRef.current) return;
   143→      const engine = new GameEngine();
   144→      engineRef.current = engine;
   145→      (window as unknown as { __eng?: GameEngine }).__eng = engine;
   146→      engine.onHud = (s) => setSnap(s);
   147→      engine.onEvent = (e) => handleEvent(e);
   148→      engine.mount(mountRef.current);
   149→      setReady(true);
   150→    })();
   151→    return () => {
   152→      disposed = true;
   153→      engineRef.current?.dispose();
   154→      engineRef.current = null;
   155→    };
   156→
   157→  }, []);
   158→
   159→  const toggleMute = useCallback(() => {
   160→    setMuted((prev) => {
   161→      const m = !prev;
   162→      gameAudio.setMuted(m);
   163→      return m;
   164→    });
   165→  }, []);
   166→
   167→  /* ---------- keyboard ---------- */
   168→  useEffect(() => {
   169→    const onKey = (ev: KeyboardEvent) => {
   170→      const eng = engineRef.current;
   171→      if (!eng || !snap) return;
   172→      const k = ev.key.toLowerCase();
   173→      if (k === "q") void eng.playerShoot("player");
   174→      else if (k === "e") void eng.playerShoot("enemy");
   175→      else if (k === "h") setShowHelp((v) => !v);
   176→      else if (k === "m") toggleMute();
   177→      else if (/^[1-4]$/.test(k)) {
   178→        const owned = snap.skills.filter((s) => s.owned && s.ghostTurns <= 0);
   179→        const target = owned[parseInt(k) - 1];
   180→        if (target) void eng.playerUseSkill(target.id);
   181→      }
   182→    };
   183→    window.addEventListener("keydown", onKey);
   184→    return () => window.removeEventListener("keydown", onKey);
   185→
   186→  }, [snap, toggleMute]);
   187→
   188→  /* ---------- playtime ticker (round + total) ---------- */
   189→  const phaseKey = snap?.phase ?? "menu";
   190→  useEffect(() => {
   191→    if (phaseKey === "menu") return;
   192→    const tick = () => setTimes(engineRef.current?.getTimes() ?? { roundMs: 0, totalMs: 0 });
   193→    tick();
   194→    const iv = setInterval(tick, 1000);
   195→    return () => clearInterval(iv);
   196→  }, [phaseKey]);
   197→
   198→  const isPlayerTurn = snap?.phase === "playerTurn" && snap.turn === "player" && !snap.busy;
   199→  const phase = snap?.phase ?? "menu";
   200→  const inGame = phase !== "menu";
   201→  const lowHp = inGame && snap ? snap.hpPlayer : 0;
   202→
   203→  /* ---------- render ---------- */
   204→  return (
   205→    <div className="fixed inset-0 overflow-hidden bg-[#0b0a08] select-none">
   206→      {/* 3D world */}
   207→      <div ref={mountRef} className="absolute inset-0" />
   208→
   209→      {/* cinematic overlays */}
   210→      <div className="pointer-events-none absolute inset-0 vignette-overlay z-10" />
   211→      <div className="pointer-events-none absolute inset-0 grain-overlay z-10" />
   212→      {flashRed && <div className="pointer-events-none absolute inset-0 z-10 bg-[rgba(199,62,46,0.35)]" />}
   213→
   214→      {/* LOW-HP BLOOD VIGNETTE: starts at 2 HP, thickest at 1 HP */}
   215→      {snap && lowHp === 2 && (
   216→        <div className="pointer-events-none absolute inset-0 z-10 blood-vignette" />
   217→      )}
   218→      {snap && lowHp === 1 && (
   219→        <>
   220→          <div className="pointer-events-none absolute inset-0 z-10 blood-vignette heavy" />
   221→          <div className="pointer-events-none absolute inset-0 z-10 overflow-hidden">
   222→            <span className="blood-drip drip-1" />
   223→            <span className="blood-drip drip-2" />
   224→            <span className="blood-drip drip-3" />
   225→            <span className="blood-drip drip-4" />
   226→          </div>
   227→          <div className="pointer-events-none absolute inset-0 z-10 pulse-danger" />
   228→        </>
   229→      )}
   230→
   231→      {snap && snap.voodooPlayer > 0 && phase !== "menu" && (
   232→        <div
   233→          className="pointer-events-none absolute inset-0 z-10"
   234→          style={{ boxShadow: "inset 0 0 120px rgba(94,156,91,0.45)" }}
   235→        />
   236→      )}
   237→
   238→      {/* blood on screen */}
   239→      <div className="pointer-events-none absolute inset-0 z-20 overflow-hidden">
   240→        {assets &&
   241→          splats.map((s) => (
   242→
   243→            <img
   244→              key={s.id}
   245→              src={s.url}
   246→              alt=""
   247→              className="absolute pop-in"
   248→              style={{
   249→                left: `${s.left}%`,
   250→                top: `${s.top}%`,
   251→                width: s.size,
   252→                transform: `rotate(${s.rot}deg)`,
   253→                transition: "opacity 1.2s ease 1s",
   254→                opacity: 0.92,
   255→              }}
   256→            />
   257→          ))}
   258→      </div>
   259→
   260→      {/* damage pops */}
   261→      <div className="pointer-events-none absolute inset-0 z-30">
   262→        {pops.map((p) => (
   263→          <div
   264→            key={p.id}
   265→            className={`absolute rise-fade font-marker text-3xl drop-shadow-[3px_3px_0_rgba(0,0,0,0.7)] ${
   266→              p.tone === "reflect" ? "text-[#8fdc8a]" : p.tone === "blank" ? "text-[#cfd8e3]" : "text-[#ff6b57]"
   267→            }`}
   268→            style={p.side === "enemy" ? { left: "58%", top: "28%" } : { left: "42%", top: "62%" }}
   269→          >
   270→            {p.text}
   271→          </div>
   272→        ))}
   273→      </div>
   274→
   275→      {/* ================= IN-GAME HUD ================= */}
   276→      {inGame && assets && snap && (
   277→        <>
   278→          {/* shell counter board — the honest numbers fade turn by turn,
   279→              their opacity really stepping down until it hits zero */}
   280→          {snap.round > 0 && (
   281→          <div className="absolute left-1/2 top-3 z-30 -translate-x-1/2">
   282→            <div className="relative">
   283→              <div
   284→                className="doodle-box flex items-center gap-3 px-4 py-1.5 transition-opacity duration-[1100ms] ease-in-out"
   285→                style={{ opacity: snap.countsPublic ? Math.min(1, snap.countdown / 3) : 0 }}
   286→              >
   287→                <div className="flex items-center gap-1">
   288→                  <img src={assets.pipLive} alt="live" className="h-7 w-7" />
   289→                  <span className="font-marker text-xl text-[#8f2418]">x{snap.live}</span>
   290→                </div>
   291→                <div className="h-6 w-0.5 bg-[#221a12]/30" />
   292→                <div className="flex items-center gap-1">
   293→                  <img src={assets.pipBlank} alt="blank" className="h-7 w-7" />
   294→                  <span className="font-marker text-xl text-[#57697c]">x{snap.blank}</span>
   295→                </div>
   296→              </div>
   297→              {!snap.countsPublic && (
   298→                <div className="absolute inset-0 flex items-center justify-center">
   299→                  <div className="doodle-box doodle-dark board-dark-box pop-in-delayed flex items-center gap-2 whitespace-nowrap px-4 py-1.5">
   300→                    <span className="font-marker text-xl tracking-widest text-[#8b8577]">? ? ?</span>
   301→                    <span className="rotate-[-2deg] font-hand-sc text-xs text-[#d9a441]">memorize!</span>
   302→                  </div>
   303→                </div>
   304→              )}
   305→            </div>
   306→            {snap.known && (
   307→              <div className="mt-1 flex justify-center">
   308→                <div
   309→                  className={`pop-in font-hand-sc px-2 py-0.5 border-2 border-[#221a12] bg-[#f3ecd7] text-sm rotate-[-2deg] ${
   310→                    snap.known === "live" ? "text-[#8f2418]" : "text-[#57697c]"
   311→                  }`}
   312→                >
   313→                  CHAMBER: {snap.known.toUpperCase()}
   314→                </div>
   315→              </div>
   316→            )}
   317→          </div>
   318→          )}
   319→
   320→          {/* top left: round + buttons */}
   321→          <div className="absolute left-2 top-2 z-30 flex flex-col gap-1.5 sm:left-3 sm:top-3 sm:gap-2">
   322→            <div className="doodle-box doodle-dark px-2 py-0.5 font-marker text-xs rotate-[-1.5deg] sm:px-3 sm:py-1 sm:text-sm">
   323→              ROUND {Math.max(1, snap.round)}
   324→            </div>
   325→            <div
   326→              className="doodle-box w-fit px-2 py-0.5 font-marker text-xs text-[#8a6d1d] rotate-[1deg] sm:text-sm"
   327→              title="Cash — land bullets ($100), voodoo reflections ($150), survive rounds ($75). Between rounds 3 random items appear on the table."
   328→            >
   329→              $ {snap.money}
   330→            </div>
   331→            <div className="flex gap-1.5">
   332→              <div className="doodle-box doodle-dark px-2 py-0.5 font-hand-sc text-[10px] sm:text-xs" title="Playtime in this round">
   333→                RND {fmtTime(times.roundMs)}
   334→              </div>
   335→              <div className="doodle-box doodle-dark px-2 py-0.5 font-hand-sc text-[10px] sm:text-xs" title="Playtime across all rounds">
   336→                SUM {fmtTime(times.totalMs)}
   337→              </div>
   338→            </div>
   339→            <div className="flex gap-1.5 sm:gap-2">
   340→              <button
   341→                className="doodle-btn px-2 py-0.5 font-hand-sc text-xs sm:px-2.5 sm:py-1 sm:text-sm"
   342→                onClick={() => setShowHelp(true)}
   343→                aria-label="How to play"
   344→              >
   345→                ? RULES
   346→              </button>
   347→              <button className="doodle-btn px-2 py-0.5 font-hand-sc text-xs sm:px-2.5 sm:py-1 sm:text-sm" onClick={toggleMute} aria-label="Sound">
   348→                {muted ? "SOUND OFF" : "SOUND ON"}
   349→              </button>
   350→            </div>
   351→          </div>
   352→
   353→          {/* top right: THE DEALER */}
   354→          <div className="absolute right-3 top-3 z-30 flex flex-col items-end gap-1">
   355→            <div className="doodle-box px-3 py-1 rotate-[1.5deg]">
   356→              <span className="font-marker text-lg text-[#8f2418]">THE DEALER</span>
   357→              {snap.voodooEnemy > 0 && (
   358→
   359→                <img src={assets.voodoo} alt="voodoo" className="ml-2 inline h-6 w-6 wiggle align-middle" />
   360→              )}
   361→              {snap.enemyCuffed && (
   362→                <span className="ml-2 font-hand-sc text-sm text-[#9b8ec4] underline">cuffed</span>
   363→              )}
   364→            </div>
   365→            {snap.enemySkills.length > 0 && (
   366→              <div className="doodle-box flex items-center gap-1 px-2 py-0.5 rotate-[1deg]">
   367→                <span className="font-hand-sc text-[10px] text-[#4a4036]">SKILLS</span>
   368→                {snap.enemySkills.map((s) => (
   369→                  <img
   370→                    key={s.id}
   371→                    src={s.icon}
   372→                    alt={s.id}
   373→                    title={SKILL_META[s.id].name}
   374→                    className="h-6 w-6"
   375→                  />
   376→                ))}
   377→              </div>
   378→            )}
   379→            <div className="flex gap-1">
   380→              {Array.from({ length: snap.maxHpEnemy }).map((_, i) => (
   381→
   382→                <img key={i} src={i < snap.hpEnemy ? assets.heartFull : assets.heartEmpty} alt="hp" className="h-8 w-8" />
   383→              ))}
   384→            </div>
   385→          </div>
   386→
   387→          {/* player: top-left on mobile, bottom-left on desktop */}
   388→          <div className="absolute left-2 top-[88px] z-30 flex flex-col gap-1 sm:left-3 sm:top-auto sm:bottom-3">
   389→            <div className="doodle-box w-fit px-2 py-0.5 rotate-[-1.5deg] sm:px-3 sm:py-1">
   390→              <span className="font-marker text-sm sm:text-lg">YOU</span>
   391→              {snap.voodooPlayer > 0 && (
   392→                <span className="ml-2 font-hand-sc text-sm text-[#5e9c5b]">voodoo {snap.voodooPlayer}</span>
   393→              )}
   394→              {snap.playerCuffed && (
   395→                <span className="ml-2 font-hand-sc text-sm text-[#9b8ec4] underline">cuffed</span>
   396→              )}
   397→            </div>
   398→            <div className="flex gap-0.5 sm:gap-1">
   399→              {Array.from({ length: snap.maxHpPlayer }).map((_, i) => (
   400→                <img key={i} src={i < snap.hpPlayer ? assets.heartFull : assets.heartEmpty} alt="hp" className="h-7 w-7 sm:h-9 sm:w-9" />
   401→              ))}
   402→            </div>
   403→          </div>
   404→
   405→          {/* turn indicator */}
   406→          <div className="pointer-events-none absolute left-1/2 top-[74px] z-20 -translate-x-1/2 sm:top-16">
   407→            {snap.turn === "player" && (
   408→              <div className="doodle-box px-4 py-0.5 font-marker text-[#8f2418] rotate-[-2deg] float-y">
   409→                YOUR TURN
   410→              </div>
   411→            )}
   412→            {snap.turn === "enemy" && (
   413→              <div className="doodle-box doodle-dark px-4 py-0.5 font-marker rotate-[2deg] float-y">
   414→                THE DEALER&apos;S TURN...
   415→              </div>
   416→            )}
   417→          </div>
   418→
   419→          {/* must shoot */}
   420→          {snap.mustShoot && snap.turn === "player" && (
   421→            <div className="pointer-events-none absolute left-1/2 top-28 z-30 -translate-x-1/2">
   422→              <div className="doodle-box doodle-red px-4 py-1 font-marker text-xl blink-soft rotate-[-3deg]">
   423→                YOU MUST SHOOT!
   424→              </div>
   425→            </div>
   426→          )}
   427→
   428→          {/* reload pips */}
   429→          {reloadPips > 0 && (
   430→            <div className="pointer-events-none absolute left-1/2 top-[62%] z-30 flex -translate-x-1/2 gap-1.5">
   431→              {Array.from({ length: 6 }).map((_, i) => (
   432→                <div
   433→                  key={i}
   434→                  className={`h-4 w-4 rounded-full border-[3px] border-[#f3ecd7] ${
   435→                    i < reloadPips ? "bg-[#c73e2e]" : "bg-transparent"
   436→                  }`}
   437→                />
   438→              ))}
   439→            </div>
   440→          )}
   441→
   442→          {/* skill bar (dice skills + shop trials; spent trials dissolve away) */}
   443→          <div className="absolute bottom-[104px] left-1/2 z-30 flex -translate-x-1/2 items-end gap-1.5 sm:bottom-3 sm:gap-2">
   444→            {snap.skills
   445→              .filter((s) => s.owned)
   446→              .map((s, idx) => {
   447→                const ghost = s.ghostTurns > 0;
   448→                return (
   449→                  <button
   450→                    key={s.id}
   451→                    className={`doodle-btn relative flex h-16 w-16 flex-col items-center justify-center gap-0.5 sm:h-[74px] sm:w-[74px] ${
   452→                      ghost ? "ghost-skill" : s.usable ? "" : "opacity-60"
   453→                    }`}
   454→                    style={
   455→                      ghost
   456→                        ? {
   457→                            opacity: 0.1 + s.ghostTurns * 0.24,
   458→                            filter: `grayscale(${0.55 + (2 - s.ghostTurns) * 0.22}) blur(${(2 - s.ghostTurns) * 0.8}px)`,
   459→                            transition: "opacity 1.1s ease, filter 1.1s ease",
   460→                          }
   461→                        : undefined
   462→                    }

   112→function shuffled<T>(arr: T[]): T[] {
   113→  const a = [...arr];
   114→  for (let i = a.length - 1; i > 0; i--) {
   115→    const j = Math.floor(Math.random() * (i + 1));
   116→    [a[i], a[j]] = [a[j], a[i]];
   117→  }
   118→  return a;
   119→}

   218→  private offerView(o: ShopOffer): ShopOfferView {
   219→    const st = this.logic.state;
   220→    let disabled = false;
   221→    let reason: string | null = null;
   222→    if (o.sold) {
   223→      reason = "SOLD";
   224→    } else if (this.activeSkillCount() >= MAX_SKILLS) {
   225→      disabled = true;
   226→      reason = `SLOTS FULL (${MAX_SKILLS})`;
   227→    } else if (this.money < o.price) {
   228→      disabled = true;
   229→      reason = "TOO POOR";
   230→    }
   231→    void st;
   232→    return { ...o, disabled, reason };

   327→  /** Dice intro: 2 rolls for the player (their only skills for the match),
   328→   *  then the enemy silently receives 1/2/3 skills depending on difficulty. */
   329→  private async diceIntro() {
   330→    const diff = this.logic.state.difficulty;
   331→    this.logic.setPhase("dice");
   332→    this.notify();
   333→    await sleep(400);
   334→    this.scene.enemySetMood("grin");
   335→    this.scene.enemySay(["Dice first.", "Your fate is decided here."], 2.8);
   336→    gameAudio.gasp();
   337→
   338→    const granted = shuffled(SKILL_IDS).slice(0, 2);
   339→
   340→    this.scene.diceVisible(true);
   341→    for (const id of granted) {
   342→      gameAudio.diceShake();
   343→      await sleep(360);
   344→      await this.diceTumble(id);
   345→      this.logic.grantSkill("player", id);
   346→      gameAudio.stamp();
   347→      this.onEvent({ type: "skillGet", id });
   348→      this.notify();
   349→      await sleep(820);
   350→    }
   351→    this.scene.diceVisible(false);
   352→
   353→    // the enemy quietly takes its skills
   354→    const n = enemySkillCount(diff);
   355→    for (const id of shuffled(SKILL_IDS).slice(0, n)) this.logic.grantSkill("enemy", id);
   356→
   357→    this.scene.enemySetMood("laugh");
   358→    gameAudio.laugh();
   359→    this.scene.enemySay([pick("gameStart")], 2.4);
   360→    await sleep(1200);
   361→  }

   363→  /** One dice tumble animation in front of the camera. */
   364→  private diceTumble(finalId: SkillId): Promise<void> {
   365→    const { main, ghosts } = this.scene.diceMeshes();
   366→    let lastSwitch = 0;
   367→    let cycle = Math.floor(Math.random() * SKILL_IDS.length);
   368→    const setFace = (id: SkillId) => this.scene.diceSetFace(id, texDieFace(id));
   369→    const dur = 1400;
   370→    const start = performance.now();
   371→    gameAudio.diceRoll(); // physical tumbling knocks while the die spins
   372→    return new Promise<void>((resolve) => {
   373→      const step = () => {
   374→        const t = Math.min(1, (performance.now() - start) / dur);
   375→        main.rotation.z = Math.pow(1 - t, 1.4) * 10;
   376→        const bounce = Math.abs(Math.sin(t * Math.PI * 3.4)) * (1 - t) * 0.3;
   377→        main.position.y = -0.02 + bounce;
   378→        main.scale.setScalar(0.7 + t * 0.3);
   379→        if (t > 0.7) {
   380→          if (main.userData.face !== finalId) {
   381→            setFace(finalId);
   382→            main.userData.face = finalId;
   383→          }
   384→        } else if (performance.now() - lastSwitch > 40 + t * t * 240) {
   385→          lastSwitch = performance.now();
   386→          cycle = (cycle + 1) % SKILL_IDS.length;
   387→          setFace(SKILL_IDS[cycle]);
   388→          main.userData.face = SKILL_IDS[cycle];
   389→        }
   390→        ghosts.forEach((g, i) => {
   391→          g.rotation.z = main.rotation.z - 0.05 * (i + 1);
   392→          g.position.y = main.position.y + 0.012 * (i + 1);
   393→          g.material.opacity = (0.13 / (i + 1)) * (1 - t * 0.4);
   394→          g.material.map = main.material.map;
   395→        });
   396→        if (t < 1) {
   397→          requestAnimationFrame(step);
   398→        } else {
   399→          gameAudio.diceLand();
   400→          this.scene.flash(0.3);
   401→          // squash & settle
   402→          const s0 = performance.now();
   403→          const squash = () => {
   404→            const k = Math.min(1, (performance.now() - s0) / 220);
   405→            main.scale.set(1 + Math.sin(k * Math.PI) * 0.12, 1 - Math.sin(k * Math.PI) * 0.18, 1);
   406→            main.rotation.z = (1 - k) * 0.08;
   407→            if (k < 1) requestAnimationFrame(squash);
   408→            else resolve();
   409→          };
   410→          squash();
   411→        }
   412→      };
   413→      step();
   414→    });
   415→  }

   417→  /** Between-round reload: gun opens, shells drop in, cylinder spins. */
   418→  private async reloadSequence() {
   419→    this.logic.state.round += 1;
   420→    this.logic.loadCylinder();
   421→    this.tRoundStart = performance.now();
   422→    this.logic.setPhase("reloading");
   423→    this.notify();
   424→    this.busy = true;
   425→
   426→    // the Dealer sweeps the last round's casings off the table
   427→    this.scene.clearSpentShells();
   428→    this.scene.setLampBoost(0.4);
   429→    await sleep(380);
   430→    this.scene.gunAim("enemy", true);
   431→    this.scene.gunOpenCylinder(true);
   432→    gameAudio.reloadOpen();
   433→    await sleep(620);
   434→
   435→    const total = 6;
   436→    for (let i = 0; i < total; i++) {
   437→      gameAudio.reloadLoad(i);
   438→      this.onEvent({ type: "reloadStep", loaded: i + 1, total });
   439→      await sleep(190);
   440→    }
   441→    gameAudio.spinCylinder();
   442→    await sleep(560);
   443→    this.scene.gunOpenCylinder(false);
   444→    gameAudio.reloadClose();
   445→    this.scene.setLampBoost(1);
   446→    this.scene.flash(0.28);
   447→    await sleep(420);
   448→    this.scene.gunAim(null);
   449→
   450→    const live = this.logic.liveRemaining;
   451→    const blank = this.logic.blankRemaining;
   452→    this.onEvent({
   453→      type: "banner",
   454→      title: `ROUND ${this.logic.state.round}`,
   455→      sub: `LIVE ${live} — BLANK ${blank} — every fired shell lands on the table edge`,
   456→      long: true,
   457→    });
   458→    gameAudio.hammerCock();
   459→    await sleep(1400);
   460→    this.busy = false;
   461→  }

   472→      gameAudio.ghostFade();

   479→  async playerShoot(target: Side) {
   480→    const st = this.logic.state;
   481→    if (this.busy || st.phase !== "playerTurn" || st.turn !== "player") return;
   482→    if (st.cylinder.length === 0) return;
   483→    this.busy = true;
   484→    this.notify();
   485→
   486→    // LOW-HP TENSION: below 3 HP a self-shot gets 0.7s of pure dread first.
   487→    const tension = target === "player" && st.hp.player < 3;
   488→
   489→    this.scene.gunAim(target === "player" ? "self" : "enemy");
   490→    gameAudio.hammerCock();
   491→    gameAudio.riser(tension ? 1.5 : 0.55);
   492→    await sleep(800);
   493→    if (tension) {
   494→      // 0.7s: the trigger finger trembles... 240 + 240 + 220
   495→      this.scene.addTrauma(0.14);
   496→      await sleep(240);
   497→      this.scene.addTrauma(0.18);
   498→      await sleep(240);
   499→      this.scene.addTrauma(0.22);
   500→      await sleep(220);
   501→    }
   502→
   503→    const res = await this.shootSequence("player", target, tension);
   504→    this.scene.gunAim(null);
   505→    await this.afterShot("player", res);
   506→  }

   508→  async playerUseSkill(id: SkillId) {
   509→    const st = this.logic.state;
   510→    if (this.busy || st.phase !== "playerTurn" || st.turn !== "player") return;
   511→    if (!this.logic.canUseSkill("player", id)) return;
   512→    this.busy = true;
   513→    this.notify();
   514→
   515→    const oc = this.logic.useSkill("player", id);
   516→    if (!oc.ok) {
   517→      this.busy = false;
   518→      this.notify();
   519→      return;
   520→    }
   521→    // a spent TRIAL starts its 2-turn fade away
   522→    const sk = st.skills[id];
   523→    if (sk.trial && sk.charges <= 0) this.registerGhost(id);
   524→    this.onEvent({ type: "skillFlash", id });
   525→    await this.playSkillFx("player", id, oc);
   526→    this.notify();
   527→    this.checkHealthFx();
   528→
   529→    // EJECT on the LAST shell empties the cylinder — nothing left to fire,
   530→    // so the round ends RIGHT HERE: shop break, reload, pass the turn.
   531→    if (this.logic.isEmpty) {
   532→      this.toast("CHAMBER EMPTY — the round is over!", "info");
   533→      await sleep(900);
   534→      await this.maybeShopAndReload();
   535→      const next = this.passTurn();
   536→      this.notify();
   537→      if (next === "player") {

   554→  private async enemyTurnStart() {
   555→    const st = this.logic.state;
   556→    if (st.hp.enemy <= 0 || st.hp.player <= 0) return;
   557→    st.phase = "enemyTurn";
   558→    const wasPublic = st.countdown > 0;
   559→    this.logic.startTurnTick();
   560→    if (wasPublic && st.countdown <= 0 && st.cylinder.length > 0) {
   561→      this.toast("THE BOARD FADED — memorize your shells!", "bad");
   562→      gameAudio.ghostFade();
   563→    }
   564→    if (this.logic.lastSlip) {
   565→      // the Dealer lost count — and says so out loud
   566→      this.scene.enemySay([pick("aiForget")], 2.2);
   567→    }
   568→    this.busy = true;
   569→    this.notify();
   570→
   571→    this.scene.enemySetMood("idle2");
   572→    this.scene.enemySay([pick("enemyTurn")], 2.2);
   573→    await sleep(rnd(1000, 1700));
   574→
   575→    let skillsUsed = 0;
   576→    let guard = 0;
   577→    let decision: AIDecision | null = null;
   578→    // AI loop: burn skills first, then shoot
   579→    for (;;) {
   580→      if (st.hp.enemy <= 0 || st.hp.player <= 0) return;
   581→      if (this.logic.isEmpty || guard++ > 24) break;
   582→      const d = this.ai.decide();
   583→      if (d.kind === "skill" && d.skill && skillsUsed < 3) {
   584→        const oc = this.logic.useSkill("enemy", d.skill);
   585→        if (!oc.ok) continue;
   586→        skillsUsed++;
   587→        await this.playSkillFx("enemy", d.skill, oc);
   588→        this.notify();
   589→        await sleep(rnd(500, 850));
   590→        continue;
   591→      }
   592→      decision = d;
   593→      break;
   594→    }
   595→
   596→    if (decision && decision.kind === "shoot" && decision.target && !this.logic.isEmpty) {
   597→      this.scene.enemySetMood("aim");
   598→      gameAudio.riser(1.0);
   599→      await sleep(rnd(850, 1400));
   600→      const res = await this.shootSequence("enemy", decision.target);
   601→      this.scene.enemySetMood("idle1");
   602→      if (res === "dead") {
   603→        await this.endGame();
   604→        return;
   605→      }
   606→    }
   607→
   608→    if (this.logic.isEmpty) {
   609→      await this.maybeShopAndReload();
   610→    }
   611→    const next = this.passTurn();
   612→    this.notify();
   613→    if (next === "player") {
   614→      await this.playerTurnStart();
   615→    } else {
   616→      // the player is cuffed
   617→      this.toast("You're cuffed! The DEALER goes again.", "bad");
   618→      await sleep(800);
   619→      await this.enemyTurnStart();
   620→    }
   621→  }

   623→  /* ---------------- shot resolution ---------------- */
   624→
   625→  /** Fire one shot and resolve EVERYTHING that follows it:
   626→   *  the voodoo curse (fresh shell + forced shot) and the global
   627→   *  self-blank rule (one automatic free shot at the opponent). */
   628→  private async shootSequence(shooter: Side, target: Side, tensionSelf = false, forced = false): Promise<"pass" | "dead"> {
   629→    const out = this.logic.shoot(shooter, target, forced);
   630→    const res = await this.resolveShot(out, tensionSelf);
   631→    if (res === "dead") return "dead";
   632→
   633→    const opp: Side = shooter === "player" ? "enemy" : "player";
   634→
   635→    if (out.voodooReroll) {
   636→      // VOODOO CURSE: blank was rerolled — the shooter MUST fire once more,
   637→      // and the gun swings toward the opponent.
   638→      this.notify();
   639→      await sleep(300);
   640→      const out2 = this.logic.shoot(shooter, opp, true);
   641→      const res2 = await this.resolveShot(out2, false);
   642→      if (res2 === "dead") return "dead";
   643→    } else if (out.bonusShot && !this.logic.isEmpty) {
   644→      // GLOBAL RULE: shot yourself with a blank -> ONE automatic shot at the enemy.
   645→      this.notify();
   646→      if (shooter === "player") {
   647→        this.toast("BLANK! Free shot at the DEALER!", "good");
   648→        this.scene.gunAim("enemy");
   649→        gameAudio.hammerCock();
   650→        await sleep(750);
   651→      } else {
   652→        this.scene.enemySetMood("aim");
   653→        this.toast("DEALER blanks himself — free shot at you!", "bad");
   654→        await sleep(750);
   655→      }
   656→      const out2 = this.logic.shoot(shooter, opp);
   657→      const res2 = await this.resolveShot(out2, false);
   658→      if (res2 === "dead") return "dead";
   659→    }
   660→    return "pass";
   661→  }

   999→  private registerGhost(id: SkillId) {
  1000→    const g = this.ghosts.find((x) => x.id === id);

  1006→  /** Every passed turn dissolves the ghosts a little more. */
  1007→  private tickGhosts() {
  1008→    if (!this.ghosts.length) return;
  1009→    for (const g of this.ghosts) g.turns -= 1;
  1010→    const done = this.ghosts.filter((g) => g.turns <= 0).map((g) => g.id);
  1011→    for (const id of done) {
  1012→      const sk = this.logic.state.skills[id];
  1013→      sk.owned = false;
  1014→      sk.trial = false;
  1015→      this.toast(`${SKILL_META[id].name} faded away.`, "info");
  1016→    }
  1017→    this.ghosts = this.ghosts.filter((g) => g.turns > 0);
  1018→  }

  1045→    const pool = (Object.keys(SHOP_PRICES) as SkillId[]).filter((id) => {
  1046→      if (LEGENDARY.includes(id) && st.difficulty !== "insane") return false;
  1047→      return !st.skills[id].owned;
  1048→    });

  1050→      .slice(0, 3)

  1089→    if (!offer || offer.sold) return;

  1092→      gameAudio.dryFire();

  1095→        poor ? `Not enough cash — $${offer.price} needed.` : "Hands full — 4 skills max!",
  1096→        "bad",
  1097→      );
  1098→      this.scene.enemySay([pick(poor ? "shopPoor" : "shopFull")], 2.4);
  1099→      return;
  1100→    }
  1101→    this.money -= offer.price;
  1102→    offer.sold = true;
  1103→    this.logic.grantTrial("player", id);
  1104→    gameAudio.buy();

  1135→  private shopNdc(e: MouseEvent): THREE.Vector2 | null {
  1136→    const c = this.container;
  1137→    if (!c) return null;
  1138→    const r = c.getBoundingClientRect();
  1139→    return new THREE.Vector2(((e.clientX - r.left) / r.width) * 2 - 1, -((e.clientY - r.top) / r.height) * 2 + 1);
  1140→  }

  1142→  private onShopPointerMove(e: MouseEvent) {
  1143→    if (this.logic.state.phase !== "shop") {
  1144→      if (this.hoverId !== null) this.setShopHover(null);
  1145→      if (this.container) this.container.style.cursor = "default";
  1146→      return;
  1147→    }
  1148→    const ndc = this.shopNdc(e);
  1149→    if (!ndc) return;
  1150→    const hit = this.scene.pickShop(ndc);
  1151→    const id = hit ? (hit.type === "bell" ? "bell" : hit.id) : null;
  1152→    if (id !== this.hoverId) {
  1153→      this.setShopHover(id);
  1154→      if (id) gameAudio.hover();
  1155→    }
  1156→    if (this.container) this.container.style.cursor = id ? "pointer" : "default";
  1157→  }

  1159→  private onShopClick(e: MouseEvent) {
  1160→    if (this.logic.state.phase !== "shop") return;
  1161→    const ndc = this.shopNdc(e);
  1162→    if (!ndc) return;
  1163→    const hit = this.scene.pickShop(ndc);
  1164→    if (!hit) return;
  1165→    if (hit.type === "bell") void this.ringBell();
  1166→    else this.buyTrial(hit.id);
  1167→  }

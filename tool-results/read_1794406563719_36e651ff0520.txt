    95→    95→            url: a[url],
    96→    96→            left: 15 + Math.random() * 55,
    97→    97→            top: 8 + Math.random() * 50,
    98→    98→            size: 130 + Math.random() * 160,
    99→    99→            rot: Math.random() * 360,

   106→   106→      }

   108→   108→        break;

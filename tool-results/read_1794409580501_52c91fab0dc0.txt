     1→"use client";
     2→
     3→/**
     4→ * GameShell — React layer for the ULTIMATE ROULETTE HUD.
     5→ * Mounts the engine (Three.js) plus all doodle overlays: menu, HP,
     6→ * shell counter, skill bar, toasts, banner, rules, win/lose screens,
     7→ * and the low-HP blood vignette.
     8→ */
     9→
    10→import { useCallback, useEffect, useMemo, useRef, useState } from "react";
    11→import { GameEngine, type HudEvent, type HudSnapshot } from "@/game/engine";
    12→import { SKILL_META, dataURL, texBulletPip, texHeart, texSkillIcon, texSplat, texVoodooMask, type SkillId } from "@/game/doodle";
    13→import { gameAudio } from "@/game/audio";
    14→
    15→type Toast = { id: number; text: string; tone: string };
    16→type Pop = { id: number; side: string; text: string; tone: string };
    17→type Splat = { id: number; url: string; left: number; top: number; size: number; rot: number };
    18→type Stamp = { id: number; idSkill: SkillId };
    19→
    20→const DIFFS: { id: "easy" | "normal" | "insane"; label: string; desc: string }[] = [
    21→  { id: "easy", label: "EASY", desc: "You have 5 HP. The Dealer has 1 skill." },
    22→  { id: "normal", label: "NORMAL", desc: "4 vs 4 HP. The Dealer has 2 skills." },
    23→  { id: "insane", label: "INSANE", desc: "The Dealer has 3 skills. Prepare to die." },
    24→];
    25→
    26→export default function GameShell() {
    27→  const mountRef = useRef<HTMLDivElement>(null);
    28→  const engineRef = useRef<GameEngine | null>(null);
    29→  const evId = useRef(0);
    30→  const [ready, setReady] = useState(false);
    31→  const [snap, setSnap] = useState<HudSnapshot | null>(null);
    32→  const [toasts, setToasts] = useState<Toast[]>([]);
    33→  const [pops, setPops] = useState<Pop[]>([]);
    34→  const [splats, setSplats] = useState<Splat[]>([]);
    35→  const [stamps, setStamps] = useState<Stamp[]>([]);
    36→  const [banner, setBanner] = useState<{ title: string; sub?: string } | null>(null);
    37→  const [reveal, setReveal] = useState<{ shell: "live" | "blank" } | null>(null);
    38→  const [showHelp, setShowHelp] = useState(false);
    39→  const [muted, setMuted] = useState(false);
    40→  const [diff, setDiff] = useState<"easy" | "normal" | "insane">("normal");
    41→  const [reloadPips, setReloadPips] = useState(0);
    42→  const [flashRed, setFlashRed] = useState(false);
    43→
    44→  /* ---------- HUD asset dataURLs (after mount, client-only) ---------- */
    45→  const assets = useMemo(() => {
    46→    if (!ready) return null;
    47→    return {
    48→      heartFull: dataURL("heart-full", texHeart("full")),
    49→      heartEmpty: dataURL("heart-empty", texHeart("empty")),
    50→      pipLive: dataURL("pip-live", texBulletPip("live")),
    51→      pipBlank: dataURL("pip-blank", texBulletPip("blank")),
    52→      voodoo: dataURL("voodoo-mask", texVoodooMask()),
    53→      splat0: dataURL("splat-hud-0", texSplat(333)),
    54→      splat1: dataURL("splat-hud-1", texSplat(444)),
    55→    };
    56→  }, [ready]);
    57→
    58→  const assetsRef = useRef(assets);
    59→  useEffect(() => {
    60→    assetsRef.current = assets;
    61→  }, [assets]);
    62→
    63→  const handleEvent = useCallback((e: HudEvent) => {
    64→    const id = ++evId.current;
    65→    switch (e.type) {
    66→      case "toast":
    67→        setToasts((t) => [...t.slice(-4), { id, text: e.text, tone: e.tone ?? "info" }]);
    68→        setTimeout(() => setToasts((t) => t.filter((x) => x.id !== id)), 3200);
    69→        break;
    70→      case "banner":
    71→        setBanner({ title: e.title, sub: e.sub });
    72→        setTimeout(() => setBanner(null), 1700);
    73→        break;
    74→      case "skillGet":
    75→        setStamps((s) => [...s, { id, idSkill: e.id }]);
    76→        setTimeout(() => setStamps((s) => s.filter((x) => x.id !== id)), 1500);
    77→        break;
    78→      case "shellReveal":
    79→        if (e.shell) {
    80→          setReveal({ shell: e.shell });
    81→          setTimeout(() => setReveal(null), 1900);
    82→        }
    83→        break;
    84→      case "dmgPop":
    85→        setPops((p) => [...p.slice(-5), { id, side: e.side, text: e.text, tone: e.tone ?? "hit" }]);
    86→        setTimeout(() => setPops((p) => p.filter((x) => x.id !== id)), 1500);
    87→        break;
    88→      case "screenSplat": {
    89→        const url = evId.current % 2 === 0 ? "splat-hud-0" : "splat-hud-1";
    90→        const a = assetsRef.current;
    91→        if (!a) break;
    92→        setSplats((s) => [
    93→          ...s.slice(-3),
    94→          {
    95→            id,
    96→            url: a[url],
    97→            left: 15 + Math.random() * 55,
    98→            top: 8 + Math.random() * 50,
    99→            size: 130 + Math.random() * 160,
   100→            rot: Math.random() * 360,
   101→          },
   102→        ]);
   103→        setFlashRed(true);
   104→        setTimeout(() => setFlashRed(false), 180);
   105→        setTimeout(() => setSplats((s) => s.filter((x) => x.id !== id)), 2600);
   106→        break;
   107→      }
   108→      case "skillFlash":
   109→        break;
   110→      case "reloadStep":
   111→        setReloadPips(e.loaded);
   112→        setTimeout(() => setReloadPips(0), 2400);
   113→        break;
   114→    }
   115→  }, []);
   116→
   117→  /* ---------- mount engine ---------- */
   118→  useEffect(() => {
   119→    let disposed = false;
   120→    (async () => {
   121→      try {
   122→        await document.fonts.ready;
   123→      } catch {
   124→        /* noop */
   125→      }
   126→      if (disposed || !mountRef.current) return;
   127→      const engine = new GameEngine();
   128→      engineRef.current = engine;
   129→      (window as unknown as { __eng?: GameEngine }).__eng = engine;
   130→      engine.onHud = (s) => setSnap(s);
   131→      engine.onEvent = (e) => handleEvent(e);
   132→      engine.mount(mountRef.current);
   133→      setReady(true);
   134→    })();
   135→    return () => {
   136→      disposed = true;
   137→      engineRef.current?.dispose();
   138→      engineRef.current = null;
   139→    };
   140→
   141→  }, []);
   142→
   143→  const toggleMute = useCallback(() => {
   144→    setMuted((prev) => {
   145→      const m = !prev;
   146→      gameAudio.setMuted(m);
   147→      return m;
   148→    });
   149→  }, []);
   150→
   151→  /* ---------- keyboard ---------- */
   152→  useEffect(() => {
   153→    const onKey = (ev: KeyboardEvent) => {
   154→      const eng = engineRef.current;
   155→      if (!eng || !snap) return;
   156→      const k = ev.key.toLowerCase();
   157→      if (k === "q") void eng.playerShoot("player");
   158→      else if (k === "e") void eng.playerShoot("enemy");
   159→      else if (k === "h") setShowHelp((v) => !v);
   160→      else if (k === "m") toggleMute();
   161→      else if (/^[1-6]$/.test(k)) {
   162→        const owned = snap.skills.filter((s) => s.owned);
   163→        const target = owned[parseInt(k) - 1];
   164→        if (target) void eng.playerUseSkill(target.id);
   165→      }
   166→    };
   167→    window.addEventListener("keydown", onKey);
   168→    return () => window.removeEventListener("keydown", onKey);
   169→
   170→  }, [snap, toggleMute]);
   171→
   172→  const eng = engineRef.current;
   173→  const isPlayerTurn = snap?.phase === "playerTurn" && snap.turn === "player" && !snap.busy;
   174→  const phase = snap?.phase ?? "menu";
   175→  const inGame = phase !== "menu";
   176→  const lowHp = inGame && snap ? snap.hpPlayer : 0;
   177→
   178→  /* ---------- render ---------- */
   179→  return (
   180→    <div className="fixed inset-0 overflow-hidden bg-[#0b0a08] select-none">
   181→      {/* 3D world */}
   182→      <div ref={mountRef} className="absolute inset-0" />
   183→
   184→      {/* cinematic overlays */}
   185→      <div className="pointer-events-none absolute inset-0 vignette-overlay z-10" />
   186→      <div className="pointer-events-none absolute inset-0 grain-overlay z-10" />
   187→      {flashRed && <div className="pointer-events-none absolute inset-0 z-10 bg-[rgba(199,62,46,0.35)]" />}
   188→
   189→      {/* LOW-HP BLOOD VIGNETTE: starts at 2 HP, thickest at 1 HP */}
   190→      {snap && lowHp === 2 && (
   191→        <div className="pointer-events-none absolute inset-0 z-10 blood-vignette" />
   192→      )}
   193→      {snap && lowHp === 1 && (
   194→        <>
   195→          <div className="pointer-events-none absolute inset-0 z-10 blood-vignette heavy" />
   196→          <div className="pointer-events-none absolute inset-0 z-10 overflow-hidden">
   197→            <span className="blood-drip drip-1" />
   198→            <span className="blood-drip drip-2" />
   199→            <span className="blood-drip drip-3" />
   200→            <span className="blood-drip drip-4" />
   201→          </div>
   202→          <div className="pointer-events-none absolute inset-0 z-10 pulse-danger" />
   203→        </>
   204→      )}
   205→
   206→      {snap && snap.voodooPlayer > 0 && phase !== "menu" && (
   207→        <div
   208→          className="pointer-events-none absolute inset-0 z-10"
   209→          style={{ boxShadow: "inset 0 0 120px rgba(94,156,91,0.45)" }}
   210→        />
   211→      )}
   212→
   213→      {/* blood on screen */}
   214→      <div className="pointer-events-none absolute inset-0 z-20 overflow-hidden">
   215→        {assets &&
   216→          splats.map((s) => (
   217→
   218→            <img
   219→              key={s.id}
   220→              src={s.url}
   221→              alt=""
   222→              className="absolute pop-in"
   223→              style={{
   224→                left: `${s.left}%`,
   225→                top: `${s.top}%`,
   226→                width: s.size,
   227→                transform: `rotate(${s.rot}deg)`,
   228→                transition: "opacity 1.2s ease 1s",
   229→                opacity: 0.92,
   230→              }}
   231→            />
   232→          ))}
   233→      </div>
   234→
   235→      {/* damage pops */}
   236→      <div className="pointer-events-none absolute inset-0 z-30">
   237→        {pops.map((p) => (
   238→          <div
   239→            key={p.id}
   240→            className={`absolute rise-fade font-marker text-3xl drop-shadow-[3px_3px_0_rgba(0,0,0,0.7)] ${
   241→              p.tone === "reflect" ? "text-[#8fdc8a]" : p.tone === "blank" ? "text-[#cfd8e3]" : "text-[#ff6b57]"
   242→            }`}
   243→            style={p.side === "enemy" ? { left: "58%", top: "28%" } : { left: "42%", top: "62%" }}
   244→          >
   245→            {p.text}
   246→          </div>
   247→        ))}
   248→      </div>
   249→
   250→      {/* ================= IN-GAME HUD ================= */}
   251→      {inGame && assets && snap && (
   252→        <>
   253→          {/* shell counter board */}
   254→          <div className="absolute left-1/2 top-3 z-30 -translate-x-1/2">
   255→            <div className="doodle-box px-4 py-1.5 flex items-center gap-3">
   256→              <div className="flex items-center gap-1">
   257→
   258→                <img src={assets.pipLive} alt="live" className="h-7 w-7" />
   259→                <span className="font-marker text-xl text-[#8f2418]">x{snap.live}</span>
   260→              </div>
   261→              <div className="h-6 w-0.5 bg-[#221a12]/30" />
   262→              <div className="flex items-center gap-1">
   263→
   264→                <img src={assets.pipBlank} alt="blank" className="h-7 w-7" />
   265→                <span className="font-marker text-xl text-[#57697c]">x{snap.blank}</span>
   266→              </div>
   267→            </div>
   268→            {snap.known && (
   269→              <div className="mt-1 flex justify-center">
   270→                <div
   271→                  className={`pop-in font-hand-sc px-2 py-0.5 border-2 border-[#221a12] bg-[#f3ecd7] text-sm rotate-[-2deg] ${
   272→                    snap.known === "live" ? "text-[#8f2418]" : "text-[#57697c]"
   273→                  }`}
   274→                >
   275→                  CHAMBER: {snap.known.toUpperCase()}
   276→                </div>
   277→              </div>
   278→            )}
   279→          </div>
   280→
   281→          {/* top left: round + buttons */}
   282→          <div className="absolute left-2 top-2 z-30 flex flex-col gap-1.5 sm:left-3 sm:top-3 sm:gap-2">
   283→            <div className="doodle-box doodle-dark px-2 py-0.5 font-marker text-xs rotate-[-1.5deg] sm:px-3 sm:py-1 sm:text-sm">
   284→              ROUND {snap.round}
   285→            </div>
   286→            <div className="flex gap-1.5 sm:gap-2">
   287→              <button
   288→                className="doodle-btn px-2 py-0.5 font-hand-sc text-xs sm:px-2.5 sm:py-1 sm:text-sm"
   289→                onClick={() => setShowHelp(true)}
   290→                aria-label="How to play"
   291→              >
   292→                ? RULES
   293→              </button>
   294→              <button className="doodle-btn px-2 py-0.5 font-hand-sc text-xs sm:px-2.5 sm:py-1 sm:text-sm" onClick={toggleMute} aria-label="Sound">
   295→                {muted ? "SOUND OFF" : "SOUND ON"}
   296→              </button>
   297→            </div>
   298→          </div>
   299→
   300→          {/* top right: THE DEALER */}
   301→          <div className="absolute right-3 top-3 z-30 flex flex-col items-end gap-1">
   302→            <div className="doodle-box px-3 py-1 rotate-[1.5deg]">
   303→              <span className="font-marker text-lg text-[#8f2418]">THE DEALER</span>
   304→              {snap.voodooEnemy > 0 && (
   305→
   306→                <img src={assets.voodoo} alt="voodoo" className="ml-2 inline h-6 w-6 wiggle align-middle" />
   307→              )}
   308→              {snap.enemyCuffed && (
   309→                <span className="ml-2 font-hand-sc text-sm text-[#9b8ec4] underline">cuffed</span>
   310→              )}
   311→            </div>
   312→            {snap.enemySkills.length > 0 && (
   313→              <div className="doodle-box flex items-center gap-1 px-2 py-0.5 rotate-[1deg]">
   314→                <span className="font-hand-sc text-[10px] text-[#4a4036]">SKILLS</span>
   315→                {snap.enemySkills.map((s) => (
   316→                  <img
   317→                    key={s.id}
   318→                    src={s.icon}
   319→                    alt={s.id}
   320→                    title={SKILL_META[s.id].name}
   321→                    className="h-6 w-6"
   322→                  />
   323→                ))}
   324→              </div>
   325→            )}
   326→            <div className="flex gap-1">
   327→              {Array.from({ length: snap.maxHp }).map((_, i) => (
   328→
   329→                <img key={i} src={i < snap.hpEnemy ? assets.heartFull : assets.heartEmpty} alt="hp" className="h-8 w-8" />
   330→              ))}
   331→            </div>
   332→          </div>
   333→
   334→          {/* player: top-left on mobile, bottom-left on desktop */}
   335→          <div className="absolute left-2 top-[88px] z-30 flex flex-col gap-1 sm:left-3 sm:top-auto sm:bottom-3">
   336→            <div className="doodle-box w-fit px-2 py-0.5 rotate-[-1.5deg] sm:px-3 sm:py-1">
   337→              <span className="font-marker text-sm sm:text-lg">YOU</span>
   338→              {snap.voodooPlayer > 0 && (
   339→                <span className="ml-2 font-hand-sc text-sm text-[#5e9c5b]">voodoo {snap.voodooPlayer}</span>
   340→              )}
   341→              {snap.playerCuffed && (
   342→                <span className="ml-2 font-hand-sc text-sm text-[#9b8ec4] underline">cuffed</span>
   343→              )}
   344→            </div>
   345→            <div className="flex gap-0.5 sm:gap-1">
   346→              {Array.from({ length: snap.maxHp }).map((_, i) => (
   347→                <img key={i} src={i < snap.hpPlayer ? assets.heartFull : assets.heartEmpty} alt="hp" className="h-7 w-7 sm:h-9 sm:w-9" />
   348→              ))}
   349→            </div>
   350→          </div>
   351→
   352→          {/* turn indicator */}
   353→          <div className="pointer-events-none absolute left-1/2 top-[74px] z-20 -translate-x-1/2 sm:top-16">
   354→            {snap.turn === "player" && (
   355→              <div className="doodle-box px-4 py-0.5 font-marker text-[#8f2418] rotate-[-2deg] float-y">
   356→                YOUR TURN
   357→              </div>
   358→            )}
   359→            {snap.turn === "enemy" && (
   360→              <div className="doodle-box doodle-dark px-4 py-0.5 font-marker rotate-[2deg] float-y">
   361→                THE DEALER&apos;S TURN...
   362→              </div>
   363→            )}
   364→          </div>
   365→
   366→          {/* must shoot */}
   367→          {snap.mustShoot && snap.turn === "player" && (
   368→            <div className="pointer-events-none absolute left-1/2 top-28 z-30 -translate-x-1/2">
   369→              <div className="doodle-box doodle-red px-4 py-1 font-marker text-xl blink-soft rotate-[-3deg]">
   370→                YOU MUST SHOOT!
   371→              </div>
   372→            </div>
   373→          )}
   374→
   375→          {/* reload pips */}
   376→          {reloadPips > 0 && (
   377→            <div className="pointer-events-none absolute left-1/2 top-[62%] z-30 flex -translate-x-1/2 gap-1.5">
   378→              {Array.from({ length: 6 }).map((_, i) => (
   379→                <div
   380→                  key={i}
   381→                  className={`h-4 w-4 rounded-full border-[3px] border-[#f3ecd7] ${
   382→                    i < reloadPips ? "bg-[#c73e2e]" : "bg-transparent"
   383→                  }`}
   384→                />
   385→              ))}
   386→            </div>
   387→          )}
   388→
   389→          {/* skill bar */}
   390→          <div className="absolute bottom-[104px] left-1/2 z-30 flex -translate-x-1/2 items-end gap-1.5 sm:bottom-3 sm:gap-2">
   391→            {snap.skills
   392→              .filter((s) => s.owned)
   393→              .map((s, idx) => (
   394→                <button
   395→                  key={s.id}
   396→                  className={`doodle-btn relative flex h-16 w-16 flex-col items-center justify-center gap-0.5 sm:h-[74px] sm:w-[74px] ${
   397→                    s.usable ? "" : "opacity-60"
   398→                  }`}
   399→                  disabled={!s.usable}
   400→                  onClick={() => void eng?.playerUseSkill(s.id)}
   401→                  title={SKILL_META[s.id].desc}
   402→                >
   403→                  <img src={s.icon} alt={s.id} className="h-8 w-8 sm:h-9 sm:w-9" />
   404→                  <span className="font-hand-sc text-[11px] leading-none">{SKILL_META[s.id].name}</span>
   405→                  <span className="absolute -top-2 -left-1.5 font-marker text-[11px] text-[#8f2418]">
   406→                    {idx + 1}
   407→                  </span>
   408→                  {s.charges >= 0 && (
   409→                    <span className="absolute -bottom-1 right-1 flex gap-0.5">
   410→                      {Array.from({ length: Math.max(s.charges, 0) }).map((_, c) => (
   411→                        <span key={c} className="h-2 w-2 rounded-full bg-[#5e9c5b] border border-[#221a12]" />
   412→                      ))}
   413→                    </span>
   414→                  )}
   415→                  {s.cooldown > 0 && (
   416→                    <span className="absolute inset-0 flex items-center justify-center rounded-[inherit] bg-black/55 font-marker text-2xl text-[#f3ecd7]">
   417→                      {s.cooldown}
   418→                    </span>
   419→                  )}
   420→                </button>
   421→              ))}
   422→          </div>
   423→
   424→          {/* shoot buttons */}
   425→          <div className="absolute bottom-3 right-2 z-30 flex flex-row gap-2 sm:right-3 sm:flex-col">
   426→            <button
   427→              className={`doodle-btn doodle-red px-3 py-2.5 font-marker text-sm sm:px-5 sm:text-lg ${isPlayerTurn ? "" : "opacity-50"}`}
   428→              disabled={!isPlayerTurn}
   429→              onClick={() => void eng?.playerShoot("enemy")}
   430→            >
   431→              SHOOT DEALER [E]
   432→            </button>
   433→            <button
   434→              className={`doodle-btn px-3 py-2.5 font-marker text-sm sm:px-5 sm:text-lg ${isPlayerTurn ? "" : "opacity-50"}`}
   435→              disabled={!isPlayerTurn}
   436→              onClick={() => void eng?.playerShoot("player")}
   437→            >
   438→              SHOOT SELF [Q]
   439→            </button>
   440→          </div>
   441→
   442→          {/* toasts left */}
   443→          <div className="pointer-events-none absolute left-3 top-28 z-30 flex w-64 flex-col gap-1.5">
   444→            {toasts.map((t) => (
   445→              <div
   446→                key={t.id}
   447→                className={`doodle-box pop-in px-3 py-1.5 font-hand text-[15px] leading-tight ${
   448→                  t.tone === "good"
   449→                    ? "border-[#3f6b3d]"
   450→                    : t.tone === "bad"
   451→                      ? "border-[#8f2418] text-[#8f2418]"
   452→                      : t.tone === "voodoo"
   453→                        ? "border-[#5e9c5b] text-[#3f6b3d] bg-[#e9f0e2]"
   454→                        : ""
   455→                }`}
   456→              >
   457→                {t.text}
   458→              </div>
   459→            ))}
   460→          </div>
   461→        </>
   462→      )}
   463→
   464→      {/* skill-get stamp */}
   465→      <div className="pointer-events-none absolute inset-0 z-40 flex items-center justify-center">
   466→        {assets &&
   467→          stamps.map((s) => (
   468→            <div key={s.id} className="stamp-in absolute flex flex-col items-center">
   469→              <div className="doodle-box flex items-center gap-3 px-8 py-4 rotate-[-3deg]">
   470→
   471→                <img src={snap?.skills.find((x) => x.id === s.idSkill)?.icon} alt={s.idSkill} className="h-16 w-16" />
